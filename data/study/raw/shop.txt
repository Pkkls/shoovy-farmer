<script>
const $ = (id) => document.getElementById(id);
const esc = (s) => String(s).replace(/[&<>"']/g,
  (ch) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[ch]));
const fmt = (n) => Math.floor(n).toLocaleString();

const st = {
  items: [],        // enabled items (viewer view)
  adminItems: [],   // all items incl. disabled (streamer view)
  queue: [], resolved: [],
  mine: [], logged: false, admin: false, bal: 0,
  expanded: null,   // item id with the note panel open
  note: "",
  hints: {},        // item id -> {text, ok}
};

async function refresh() {
  const d = await (await fetch("/api/shop")).json();
  st.items = d.items || [];
  st.mine = d.mine || [];
  st.logged = d.logged_in;
  st.admin = d.is_admin;
  if (d.balance !== undefined) {
    st.bal = d.balance;
    $("balance").textContent = d.balance.toLocaleString() + " credits";
  }
  if (st.admin) {
    const a = await (await fetch("/api/admin/shop")).json();
    st.adminItems = a.items || [];
    st.queue = a.queue || [];
    st.resolved = a.resolved || [];
  }
  render();
}

function ago(ts) {
  const s = Math.max(0, Math.floor(Date.now() / 1000 - ts));
  if (s < 60) return s + "s ago";
  if (s < 3600) return Math.floor(s / 60) + "m ago";
  if (s < 86400) return Math.floor(s / 3600) + "h ago";
  return Math.floor(s / 86400) + "d ago";
}

async function adminPost(url, body, hintId) {
  const r = await fetch(url, {
    method: "POST", headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  const d = await r.json();
  if (hintId) {
    const hint = $(hintId);
    hint.className = "hint" + (r.ok ? "" : " err");
    hint.textContent = r.ok ? (d.message || "Done") : (d.error || "Failed");
  }
  await refresh();
  return r.ok;
}

function render() {
  const active = document.activeElement;
  if (active && (active.id === "note-input" || active.closest?.(".modal"))) return;

  $("new-item-btn").style.display = st.admin ? "" : "none";

  // the grid: streamer sees all items (incl. disabled) to manage; viewers see enabled
  const items = st.admin ? st.adminItems : st.items;
  $("shop-empty").style.display = items.length ? "none" : "block";
  $("shop-empty").textContent = st.admin
    ? "No rewards yet — hit + New item to add one."
    : "The shop is empty right now — check back later!";
  const box = $("shop-list");
  box.innerHTML = "";
  for (const it of items) box.appendChild(rewardCard(it));

  renderQueue();

  $("mine-box").style.display = st.logged && st.mine.length ? "" : "none";
  const mine = $("mine");
  mine.innerHTML = "";
  for (const r of st.mine) {
    const tr = document.createElement("tr");
    const what = document.createElement("td");
    what.textContent = r.item_name + " ";
    const chip = document.createElement("span");
    chip.className = "chip " + r.status;
    chip.textContent = r.status.toUpperCase();
    what.appendChild(chip);
    if (r.note) {
      const note = document.createElement("span");
      note.className = "note";
      note.textContent = "“" + r.note + "”";
      what.appendChild(note);
    }
    const amt = document.createElement("td");
    amt.className = "amt";
    amt.textContent = "-" + fmt(r.cost);
    const when = document.createElement("td");
    when.className = "when";
    when.textContent = ago(r.created_at);
    tr.append(what, amt, when);
    mine.appendChild(tr);
  }
}

/* ---------- streamer: fulfillment queue + recently handled ---------- */

function renderQueue() {
  $("queue-box").style.display = st.admin ? "" : "none";
  $("resolved-box").style.display = st.admin ? "" : "none";
  if (!st.admin) return;

  const q = $("queue");
  q.innerHTML = "";
  if (!st.queue.length) {
    const tr = document.createElement("tr");
    const td = document.createElement("td");
    td.style.color = "var(--muted)";
    td.textContent = "Nothing pending — all caught up!";
    tr.appendChild(td);
    q.appendChild(tr);
  }
  for (const row of st.queue) {
    const tr = document.createElement("tr");
    const what = document.createElement("td");
    const who = document.createElement("b");
    who.textContent = row.username;
    what.append(who, document.createTextNode(" · " + row.item_name));
    if (row.note) {
      const n = document.createElement("span");
      n.className = "note";
      n.textContent = "“" + row.note + "”";
      what.appendChild(n);
    }
    const when = document.createElement("td");
    when.className = "when";
    when.textContent = ago(row.created_at);
    const act = document.createElement("td");
    const fulfill = document.createElement("button");
    fulfill.className = "btn success q-btn";
    fulfill.textContent = "Fulfill";
    fulfill.onclick = () => adminPost("/api/admin/shop/resolve", { id: row.id, action: "fulfill" }, "queue-hint");
    const refund = document.createElement("button");
    refund.className = "btn danger q-btn";
    refund.textContent = "Refund";
    refund.onclick = () => {
      if (confirm(`Refund ${row.cost.toLocaleString()} credits to ${row.username}?`))
        adminPost("/api/admin/shop/resolve", { id: row.id, action: "refund" }, "queue-hint");
    };
    act.append(fulfill, refund);
    tr.append(what, when, act);
    q.appendChild(tr);
  }

  const res = $("resolved");
  res.innerHTML = "";
  for (const row of st.resolved) {
    const tr = document.createElement("tr");
    const what = document.createElement("td");
    what.textContent = `${row.username} · ${row.item_name} — ${row.status}`;
    const amt = document.createElement("td");
    amt.className = "amt";
    amt.style.color = row.status === "refunded" ? "var(--green-text)" : "var(--red-text)";
    amt.textContent = (row.status === "refunded" ? "+" : "-") + fmt(row.cost);
    const when = document.createElement("td");
    when.className = "when";
    when.textContent = ago(row.resolved_at);
    tr.append(what, amt, when);
    res.appendChild(tr);
  }
}

/* ---------- reward cards ---------- */

function rewardCard(it) {
  const soldOut = it.stock !== null && it.stock < 1;
  const disabled = st.admin && !it.enabled;
  const card = document.createElement("section");
  card.className = "card reward" + (soldOut ? " sold-out" : "") + (disabled ? " disabled" : "");
  card.innerHTML = `
    <div class="name">${it.emoji ? esc(it.emoji) + " " : ""}${esc(it.name)}${disabled ? '<span class="off-tag">DISABLED</span>' : ""}</div>
    <div class="desc">${esc(it.description)}</div>
    <div class="meta">
      <span class="cost">${fmt(it.cost)} credits</span>
      <span class="stock">${it.stock === null ? "" : (soldOut ? "SOLD OUT" : it.stock + " left")}</span>
    </div>`;

  const meta = card.querySelector(".meta");
  const btn = document.createElement("button");
  btn.className = "btn primary";
  btn.textContent = "Redeem";
  if (!st.logged || soldOut || disabled) btn.disabled = true;
  btn.onclick = () => {
    st.expanded = st.expanded === it.id ? null : it.id;
    st.note = "";
    render();
  };
  meta.appendChild(btn);

  if (!st.logged) {
    const login = document.createElement("div");
    login.className = "hint";
    login.innerHTML = '<a href="/auth/login" style="color:var(--blue-text);">Log in with Kick</a> to redeem.';
    card.appendChild(login);
  }

  if (st.expanded === it.id) card.appendChild(notePanel(it));

  // streamer controls
  if (st.admin) card.appendChild(adminBar(it));

  const hint = st.hints[it.id];
  if (hint) {
    const el = document.createElement("div");
    el.className = "hint " + (hint.ok ? "ok" : "err");
    el.textContent = hint.text;
    card.appendChild(el);
  }
  return card;
}

function adminBar(it) {
  const bar = document.createElement("div");
  bar.className = "admin-bar";
  const edit = document.createElement("button");
  edit.className = "btn";
  edit.textContent = "Edit";
  edit.onclick = () => openModal(it);
  const toggle = document.createElement("button");
  toggle.className = "btn";
  toggle.textContent = it.enabled ? "Disable" : "Enable";
  toggle.onclick = () => adminPost("/api/admin/shop/item/toggle", { id: it.id, enabled: !it.enabled });
  const del = document.createElement("button");
  del.className = "btn danger";
  del.textContent = "Delete";
  del.onclick = () => {
    if (confirm(`Delete “${it.name}”? Viewers will no longer see it.`))
      adminPost("/api/admin/shop/item/delete", { id: it.id });
  };
  bar.append(edit, toggle, del);
  return bar;
}

/* ---------- create / edit modal ---------- */

let editingItem = null;

function openModal(item) {
  editingItem = item ? item.id : null;
  $("item-modal-title").textContent = item ? "Edit item" : "New item";
  $("m-save").textContent = item ? "Save changes" : "Create";
  $("m-emoji").value = item ? (item.emoji || "") : "";
  $("m-name").value = item ? item.name : "";
  $("m-desc").value = item ? item.description : "";
  $("m-cost").value = item ? item.cost : "";
  $("m-stock").value = item && item.stock !== null ? item.stock : "";
  $("m-hint").textContent = "";
  $("item-overlay").classList.add("show");
  $("m-name").focus();
}

$("new-item-btn").onclick = () => openModal(null);
$("m-cancel").onclick = () => $("item-overlay").classList.remove("show");
$("item-overlay").addEventListener("click", (e) => {
  if (e.target === $("item-overlay")) $("item-overlay").classList.remove("show");
});
$("m-save").onclick = async () => {
  const body = {
    name: $("m-name").value, description: $("m-desc").value,
    cost: $("m-cost").value, stock: $("m-stock").value, emoji: $("m-emoji").value,
  };
  if (editingItem !== null) body.id = editingItem;
  const r = await fetch("/api/admin/shop/item", {
    method: "POST", headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  const d = await r.json();
  if (r.ok) { $("item-overlay").classList.remove("show"); await refresh(); }
  else $("m-hint").textContent = d.error || "Failed";
};

/* ---------- viewer redeem ---------- */

function notePanel(it) {
  const panel = document.createElement("div");
  panel.className = "note-panel";
  const affordable = st.bal >= it.cost;
  panel.innerHTML = `
    <input id="note-input" maxlength="200" placeholder="Optional note for the streamer…" value="${esc(st.note)}">
    <div class="note-actions">
      <span style="margin-right:auto;color:var(--muted);font-size:12px;">
        Balance after: ${affordable ? fmt(st.bal - it.cost) : '<span style="color:var(--red-text)">not enough credits</span>'}
      </span>
      <button class="btn" id="note-cancel">Cancel</button>
      <button class="btn success" id="note-confirm" ${affordable ? "" : "disabled"}>
        Confirm — ${fmt(it.cost)}
      </button>
    </div>`;
  panel.querySelector("#note-input").addEventListener("input", (e) => { st.note = e.target.value; });
  panel.querySelector("#note-cancel").onclick = () => { st.expanded = null; st.note = ""; render(); };
  panel.querySelector("#note-confirm").onclick = async () => {
    const r = await fetch("/api/shop/redeem", {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ item_id: it.id, note: st.note }),
    });
    const data = await r.json();
    st.hints[it.id] = { ok: r.ok, text: r.ok ? data.message : data.error };
    if (r.ok && window.sfx) sfx.cash();
    if (r.ok) {
      st.expanded = null;
      st.note = "";
      if (data.balance !== undefined) {
        st.bal = data.balance;
        $("balance").textContent = data.balance.toLocaleString() + " credits";
      }
      setTimeout(() => { delete st.hints[it.id]; render(); }, 5000);
    }
    await refresh();
  };
  return panel;
}
