/* ---------------------------------------------------------------------------
   The site's look, set in Admin > Appearance.

   One hex is stored; every softer shade (glow, hairline, fill, the wash off
   the top of the page) is derived from it here, so a new accent needs no CSS.
   The surface theme is just a data-theme flag — theme.css holds the greys.

   Painted from the sessionStorage copy of /api/me first so moving between
   pages never flashes the default cyan, then again when the fetch lands.
   window.siteTheme.apply() is public so the admin panel can preview a colour
   before it saves it. */
window.siteTheme = (function () {
  const root = document.documentElement;

  const parse = (hex) => [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16));
  const rgba = (rgb, a) => `rgba(${rgb[0]}, ${rgb[1]}, ${rgb[2]}, ${a})`;
  // toward white, for the hover shade and for link text on a dark accent
  const lift = (rgb, amount) =>
    "#" + rgb.map((c) => Math.round(c + (255 - c) * amount).toString(16).padStart(2, "0")).join("");
  // sRGB relative luminance — decides whether a filled button takes dark or
  // light text, so a lime accent and a deep violet one are both readable
  const lum = (rgb) =>
    rgb.map((c) => c / 255).map((c) => (c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4))
       .reduce((a, c, i) => a + c * [0.2126, 0.7152, 0.0722][i], 0);

  function apply(accent, theme) {
    if (accent && /^#[0-9a-fA-F]{6}$/.test(accent)) {
      const rgb = parse(accent);
      const set = (k, v) => root.style.setProperty(k, v);
      set("--accent", accent);
      set("--accent-bright", lift(rgb, 0.32));
      set("--accent-glow", rgba(rgb, 0.4));
      set("--accent-line", rgba(rgb, 0.28));
      set("--accent-dim", rgba(rgb, 0.12));
      set("--accent-wash", rgba(rgb, 0.07));
      set("--accent-faint", rgba(rgb, 0.04));
      set("--on-accent", lum(rgb) > 0.4 ? "#04121a" : "#ffffff");
      // the legacy aliases the pages still use for "primary"
      set("--blue", accent);
      set("--blue-hover", lift(rgb, 0.32));
      set("--blue-text", lum(rgb) < 0.25 ? lift(rgb, 0.42) : accent);
    }
    if (theme) root.dataset.theme = theme;
  }

  try {
    const cached = JSON.parse(sessionStorage.getItem("me") || "null");
    if (cached) apply(cached.accent, cached.theme);
  } catch (e) {}

  return { apply };
})();

/* Shared site shell. Usage: put <nav class="right" id="site-nav"></nav> inside
   <header> (after the <h1> brand), then <script src="/static/nav.js"></script>
   right after the header.

   This builds two things:
     · the command bar   — brand, current route, clock, balance, daily claim,
                           log in/out
     · the nav rail      — the page list, grouped into sections, pinned down
                           the left edge (a slide-over drawer on phones)

   The #balance span is created synchronously, so page scripts further down can
   rely on document.getElementById("balance") existing. Login state (and the
   channel name shown as the brand) is cached in sessionStorage so the shell
   renders in its final shape immediately on navigation instead of shifting
   when /api/me returns. */
(function () {
  const nav = document.getElementById("site-nav");
  if (!nav) return;

  const style = document.createElement("style");
  style.textContent = `
    header h1 .beta-badge {
      font-family: var(--mono); font-size: 9px; font-weight: 600; line-height: 1;
      letter-spacing: 0.1em; text-transform: uppercase; color: var(--accent);
      background: var(--accent-dim); border: 1px solid var(--accent-line);
      padding: 3px 5px;
    }
    #site-nav { display: flex; align-items: center; gap: 12px; flex: 1; min-width: 0; }
    #site-nav .route {
      display: flex; align-items: center; gap: 7px; min-width: 0;
      font-family: var(--mono); font-size: 11px; letter-spacing: 0.14em;
      text-transform: uppercase; color: var(--dim); white-space: nowrap;
      padding-left: 12px; border-left: 1px solid var(--line);
    }
    #site-nav .route i { color: var(--muted); font-style: normal; }
    #site-nav .nav-right {
      margin-left: auto; display: flex; align-items: center; gap: 10px; min-width: 0;
    }
    #site-nav .stat {
      display: flex; align-items: center; gap: 6px;
      font-family: var(--mono); font-size: 10px; letter-spacing: 0.12em;
      text-transform: uppercase; color: var(--muted); white-space: nowrap;
    }
    #site-nav #clock { font-variant-numeric: tabular-nums; }
    #site-nav #balance {
      font-family: var(--mono); font-size: 11.5px; font-weight: 500;
      font-variant-numeric: tabular-nums; color: var(--gold);
      border: 1px solid rgba(245, 181, 68, 0.25); background: rgba(245, 181, 68, 0.07);
      padding: 3px 9px; white-space: nowrap;
    }
    #site-nav #balance:empty { display: none; }
    #site-nav #auth-link {
      font-family: var(--font); font-size: 11px; font-weight: 600;
      letter-spacing: 0.08em; text-transform: uppercase; text-decoration: none;
      color: var(--accent); border: 1px solid var(--accent-line);
      padding: 5px 11px; white-space: nowrap; transition: background 0.12s, color 0.12s;
    }
    #site-nav #auth-link:hover { background: var(--accent-dim); }
    #site-nav #claim-daily {
      font-family: var(--font); font-size: 11px; font-weight: 600;
      letter-spacing: 0.08em; text-transform: uppercase;
      color: var(--green-text); background: rgba(23, 133, 90, 0.18);
      border: 1px solid rgba(52, 211, 153, 0.35); padding: 5px 11px;
      cursor: pointer; white-space: nowrap;
    }
    #site-nav #claim-daily:hover:not(:disabled) { background: rgba(23, 133, 90, 0.32); }
    #site-nav #claim-daily:disabled { opacity: 0.6; cursor: default; }

    #nav-burger {
      display: none; background: none; color: var(--dim); border: 1px solid var(--line);
      padding: 6px; line-height: 0; flex: 0 0 auto;
    }
    #nav-burger:hover, #nav-burger:focus-visible { color: var(--accent); border-color: var(--accent-line); }
    #nav-burger svg { display: block; }

    /* long/short label pair — the bar keeps the wording it has room for */
    #site-nav .sm { display: none; }

    /* the drawer's log-out link: the bar carries it everywhere but phones */
    #rail-auth { display: none; margin-bottom: 10px; padding-left: 0; padding-right: 0; }
    #rail-auth .idx { color: var(--muted); }

    @media (max-width: 980px) { #nav-burger { display: inline-flex; } }
    @media (max-width: 860px) { #site-nav .route, #site-nav #clock { display: none; } }
    @media (max-width: 620px) {
      #site-nav { gap: 8px; }
      #site-nav .nav-right { gap: 7px; min-width: 0; }
      #site-nav .lg { display: none; }
      #site-nav .sm { display: inline; }
      #site-nav #balance { font-size: 11px; padding: 3px 7px; }
      #site-nav #claim-daily, #site-nav #auth-link { font-size: 10.5px; padding: 4px 9px; }
      /* logged in there is a balance to fit, so log out waits in the drawer */
      body.logged-in #site-nav #auth-link { display: none; }
      body.logged-in #rail-auth { display: flex; }
    }
  `;
  document.head.appendChild(style);

  // two wordings for one control: the wide one, and the one that fits a phone
  function setLabel(el, long, short) {
    el.innerHTML = "";
    const a = document.createElement("span");
    a.className = "lg";
    a.textContent = long;
    const b = document.createElement("span");
    b.className = "sm";
    b.textContent = short;
    el.append(a, b);
  }

  const header = nav.closest("header");
  const brand = document.querySelector("header h1");

  /* ------------------------------------------------------------------ rail */
  const PAGES = [
    ["Operations", [["/", "Home"], ["/casino", "Casino"], ["/stocks", "Market"]]],
    ["Economy", [["/shop", "Shop"], ["/raffles", "Raffles"], ["/fishing", "Fishing"],
                 ["/business", "Business"], ["/crime", "Crime"]]],
    ["Intel", [["/stats", "Stats"], ["/commands", "Commands"], ["/updates", "Updates"]]],
    ["Control", [["/admin", "Admin"]]],
  ];
  // the lobby is the way in; /games is a game already open, so it keeps the
  // Casino entry lit rather than leaving no rail item marked
  const ALIAS = { "/games": "/casino" };
  const here = ALIAS[location.pathname] || location.pathname;

  const rail = document.createElement("aside");
  rail.className = "op-rail";
  rail.id = "op-rail";
  const scroll = document.createElement("div");
  scroll.className = "rail-scroll";
  rail.appendChild(scroll);

  let routeName = "";
  let adminGroup = null;
  let n = 0;
  for (const [section, items] of PAGES) {
    const group = document.createElement("div");
    const label = document.createElement("div");
    label.className = "rail-label";
    label.textContent = section;
    group.appendChild(label);
    for (const [href, text] of items) {
      const a = document.createElement("a");
      a.className = "rail-item";
      a.href = href;
      const idx = document.createElement("span");
      idx.className = "idx";
      idx.textContent = String(++n).padStart(2, "0");
      const nm = document.createElement("span");
      nm.className = "nm";
      nm.textContent = text;
      a.append(idx, nm);
      if (here === href) {
        a.classList.add("current");
        a.setAttribute("aria-current", "page");
        routeName = text;
      }
      group.appendChild(a);
    }
    if (section === "Control") { adminGroup = group; group.style.display = "none"; }
    scroll.appendChild(group);
  }

  const foot = document.createElement("div");
  foot.className = "rail-foot";
  foot.innerHTML =
    '<a class="rail-item" id="rail-auth" href="/auth/logout">' +
    '<span class="idx">&#8629;</span><span class="nm">Log out</span></a>' +
    '<div class="sweep" style="margin:0 0 9px"></div>' +
    '<div class="rail-stat"><span>Session</span><b id="rail-user">Guest</b></div>' +
    '<div class="rail-stat"><span>Uplink</span><b id="rail-link" class="ok">Nominal</b></div>';
  rail.appendChild(foot);

  const scrim = document.createElement("div");
  scrim.className = "rail-scrim";

  document.body.append(rail, scrim);

  /* -------------------------------------------------------- drawer toggle */
  const burger = document.createElement("button");
  burger.id = "nav-burger";
  burger.type = "button";
  burger.setAttribute("aria-label", "Menu");
  burger.setAttribute("aria-expanded", "false");
  const icons = {
    closed: '<svg width="16" height="16" viewBox="0 0 20 20" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round"><path d="M3 5h14M3 10h14M3 15h14"/></svg>',
    open: '<svg width="16" height="16" viewBox="0 0 20 20" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round"><path d="M5 5l10 10M15 5L5 15"/></svg>',
  };
  burger.innerHTML = icons.closed;
  function setMenu(open) {
    document.body.classList.toggle("rail-open", open);
    burger.setAttribute("aria-expanded", String(open));
    burger.innerHTML = open ? icons.open : icons.closed;
  }
  burger.addEventListener("click", (e) => {
    // the icon swap replaces the clicked SVG node, which would make the
    // document listener below read this as an outside click and re-close
    e.stopPropagation();
    setMenu(!document.body.classList.contains("rail-open"));
  });
  scrim.addEventListener("click", () => setMenu(false));
  document.addEventListener("click", (e) => {
    if (document.body.classList.contains("rail-open") &&
        !rail.contains(e.target) && !header.contains(e.target)) setMenu(false);
  });
  document.addEventListener("keydown", (e) => { if (e.key === "Escape") setMenu(false); });
  header.insertBefore(burger, header.firstChild);

  /* --------------------------------------------------------- command bar */
  const route = document.createElement("span");
  route.className = "route";
  route.innerHTML = '<i>&#47;</i><span id="route-name"></span>';
  nav.appendChild(route);
  document.getElementById("route-name").textContent =
    routeName || (document.title.split("·").pop() || "").trim() || "Console";

  const right = document.createElement("span");
  right.className = "nav-right";
  nav.appendChild(right);

  const clock = document.createElement("span");
  clock.className = "stat";
  clock.id = "clock";
  right.appendChild(clock);
  function tick() {
    const d = new Date();
    const p = (x) => String(x).padStart(2, "0");
    clock.textContent =
      p(d.getUTCHours()) + ":" + p(d.getUTCMinutes()) + ":" + p(d.getUTCSeconds()) + "Z";
  }
  tick();
  setInterval(tick, 1000);

  // "Claim daily" appears only while the daily bonus is unclaimed — the
  // website counterpart of !daily in chat.
  const daily = document.createElement("button");
  daily.id = "claim-daily";
  daily.type = "button";
  daily.style.display = "none";
  daily.addEventListener("click", async () => {
    daily.disabled = true;
    try {
      const r = await fetch("/api/daily", { method: "POST" });
      const data = await r.json();
      if (r.ok) {
        balance.textContent = data.balance.toLocaleString() + " credits";
        setLabel(daily, "+" + data.amount.toLocaleString() + " claimed",
                        "+" + data.amount.toLocaleString());
        setTimeout(() => {
          daily.style.display = "none";
        }, 2500);
      } else {
        daily.style.display = "none"; // already claimed elsewhere (e.g. via chat)
      }
      // refresh the cached login state so other pages agree
      const me = await (await fetch("/api/me")).json();
      try {
        sessionStorage.setItem("me", JSON.stringify(me));
      } catch (e) {}
    } catch (e) {
      daily.disabled = false;
    }
  });
  right.appendChild(daily);

  const balance = document.createElement("span");
  balance.id = "balance";
  right.appendChild(balance);

  const auth = document.createElement("a");
  auth.id = "auth-link";
  auth.href = "/auth/login";
  setLabel(auth, "Log in with Kick", "Log in");
  auth.addEventListener("click", () => {
    try {
      sessionStorage.removeItem("me");
    } catch (e) {}
  });
  right.appendChild(auth);

  /* --------------------------------------------------------- login state */
  // small "beta" tag on the brand so people know the site is in beta
  function betaBadge() {
    const b = document.createElement("span");
    b.className = "beta-badge";
    b.textContent = "beta";
    return b;
  }
  if (brand && !brand.querySelector(".beta-badge")) brand.append(betaBadge());

  const railAuth = document.getElementById("rail-auth");
  if (railAuth) railAuth.addEventListener("click", () => {
    try {
      sessionStorage.removeItem("me");
    } catch (e) {}
  });

  const railUser = document.getElementById("rail-user");
  const railLink = document.getElementById("rail-link");

  function apply(me) {
    window.siteTheme.apply(me.accent, me.theme);
    if (brand && me.channel) {
      brand.innerHTML = "";
      const dot = document.createElement("span");
      dot.textContent = "■";
      const name = document.createElement("span");
      name.className = "brand-name";
      name.textContent = me.channel;
      brand.append(dot, name, betaBadge());
    }
    document.body.classList.toggle("logged-in", !!me.logged_in);
    if (me.logged_in) {
      setLabel(auth, "Log out", "Log out");
      auth.href = "/auth/logout";
      if (adminGroup) adminGroup.style.display = me.is_admin ? "" : "none";
      balance.textContent = me.balance.toLocaleString() + " credits";
      if (railUser) railUser.textContent = me.username || "Operator";
      if (railLink) { railLink.textContent = me.is_admin ? "Admin" : "Linked"; railLink.className = "ok"; }
      if (me.daily_ready && !daily.disabled) {
        setLabel(daily,
          "Claim " + (me.daily_amount ? me.daily_amount.toLocaleString() + " " : "") + "daily",
          me.daily_amount ? "+" + me.daily_amount.toLocaleString() : "Daily");
        daily.style.display = "";
      } else if (!me.daily_ready && !daily.disabled) {
        daily.style.display = "none";
      }
    } else {
      setLabel(auth, "Log in with Kick", "Log in");
      auth.href = "/auth/login";
      if (adminGroup) adminGroup.style.display = "none";
      balance.textContent = "";
      daily.style.display = "none";
      if (railUser) railUser.textContent = "Guest";
      if (railLink) { railLink.textContent = "Anonymous"; railLink.className = ""; }
    }
  }

  // render immediately from the cached state, then refresh in the background
  try {
    const cached = sessionStorage.getItem("me");
    if (cached) apply(JSON.parse(cached));
  } catch (e) {}

  fetch("/api/me")
    .then((r) => r.json())
    .then((me) => {
      try {
        sessionStorage.setItem("me", JSON.stringify(me));
      } catch (e) {}
      apply(me);
    })
    .catch(() => {});
})();
