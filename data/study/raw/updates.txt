const $ = (id) => document.getElementById(id);
const STATUS_LABELS = { new: "new", planned: "planned", done: "done", declined: "declined" };

function fmtDate(ts) {
  return new Date(ts * 1000).toLocaleDateString(undefined,
    { year: "numeric", month: "short", day: "numeric" });
}

/* Everything is rendered with textContent (never innerHTML) so suggestion
   text and changelog entries can't inject markup. */
function renderChangelog(entries) {
  const list = $("cl-list");
  list.innerHTML = "";
  $("cl-empty").style.display = entries.length ? "none" : "block";
  for (const e of entries) {
    const li = document.createElement("li");
    const when = document.createElement("div");
    when.className = "when";
    when.textContent = fmtDate(e.created_at);
    const title = document.createElement("h3");
    title.textContent = e.title;
    li.append(when, title);
    if (e.body) {
      const body = document.createElement("div");
      body.className = "body";
      body.textContent = e.body;
      li.appendChild(body);
    }
    list.appendChild(li);
  }
}

function renderMine(mine) {
  const list = $("sg-mine");
  list.innerHTML = "";
  for (const s of mine || []) {
    const li = document.createElement("li");
    const txt = document.createElement("div");
    txt.className = "txt";
    txt.textContent = s.text;
    const tag = document.createElement("span");
    tag.className = "tag st-" + s.status;
    tag.textContent = STATUS_LABELS[s.status] || s.status;
    txt.appendChild(tag);
    const meta = document.createElement("div");
    meta.className = "meta";
    meta.textContent = "sent " + fmtDate(s.created_at);
    li.append(txt, meta);
    list.appendChild(li);
  }
}

function paintInfo(d) {
  const info = $("sg-info");
  if (!d.enabled) {
    info.textContent = "Feature requests are closed right now — check back later.";
    $("sg-box").style.display = "none";
    return;
  }
  if (!d.logged_in) {
    info.innerHTML = '<a href="/auth/login" style="color:var(--blue-text);">Log in with Kick</a> to suggest a feature — up to ' + d.per_day + " per day.";
    return;
  }
  $("sg-box").style.display = "block";
  const left = d.remaining_today;
  info.textContent = left > 0
    ? `You can send ${left} more suggestion${left > 1 ? "s" : ""} today.`
    : "You've used today's suggestions — come back tomorrow.";
  $("sg-send").disabled = left <= 0;
  $("sg-input").disabled = left <= 0;
}

async function load() {
  const d = await (await fetch("/api/updates")).json();
  renderChangelog(d.changelog);
  paintInfo(d);
  renderMine(d.mine);

  const input = $("sg-input");
  input.maxLength = d.max_chars;
  const count = $("sg-count");
  const updateCount = () => { count.textContent = input.value.length + " / " + d.max_chars; };
  input.addEventListener("input", updateCount);
  updateCount();

  $("sg-send").onclick = async () => {
    const hint = $("sg-hint");
    const btn = $("sg-send");
    btn.disabled = true;
    try {
      const r = await fetch("/api/suggestions", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ text: input.value })
      });
      const data = await r.json();
      hint.className = "hint " + (r.ok ? "ok" : "err");
      if (r.ok) {
        hint.textContent = data.message;
        input.value = "";
        updateCount();
        renderMine(data.mine);
        paintInfo({ ...d, logged_in: true, remaining_today: data.remaining_today });
      } else {
        hint.textContent = data.error;
        btn.disabled = false;
      }
    } catch (e) {
      hint.className = "hint err";
      hint.textContent = "Something went wrong — try again.";
      btn.disabled = false;
    }
  };
}
load();
</script>
