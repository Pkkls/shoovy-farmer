/* same hash-based tabs as the casino and market pages */
(() => {
  const buttons = document.querySelectorAll(".tabs button");
  // a tab that is still [hidden] (admin-only, before /api/me confirms) can't
  // be selected — not even via an #admin deep link
  const names = () => [...buttons].filter(b => !b.hidden).map(b => b.dataset.tab);
  function show(name) {
    if (!names().includes(name)) name = "all";
    buttons.forEach(b => b.classList.toggle("active", b.dataset.tab === name));
    document.querySelectorAll("[data-tab-panel]").forEach(s => {
      s.style.display = (name === "all" || s.dataset.tabPanel === name) ? "" : "none";
    });
    history.replaceState(null, "", "#" + name);
  }
  buttons.forEach(b => b.onclick = () => show(b.dataset.tab));

  // Admin-only commands: shown only when the streamer or an admin is logged
  // in. Reveal from the sessionStorage copy nav.js keeps — before the first
  // render, so an admin's #admin deep link works — then confirm via /api/me.
  function reveal(me) {
    if (!me || !me.is_admin) return;
    document.querySelectorAll('[data-tab="admin"], [data-tab-panel="admin"]')
      .forEach(el => { el.hidden = false; });
    const active = document.querySelector(".tabs button.active");
    if (active) show(active.dataset.tab); // refresh the current view
  }
  try {
    const cached = sessionStorage.getItem("me");
    if (cached) reveal(JSON.parse(cached));
  } catch (e) {}
  fetch("/api/me").then(r => r.json()).then(reveal).catch(() => {});

  show(location.hash.slice(1));
  // hash edited while already on the page (e.g. a #tab link) still switches
  window.addEventListener("hashchange", () => {
    const name = location.hash.slice(1);
    if (names().includes(name)) show(name);
  });
})();
</script>
