<script>
const $ = (id) => document.getElementById(id);
const fmt = (n) => Math.floor(n).toLocaleString();

let robRemaining = 0, loggedIn = false, crimeOff = false;
let cooldownTimer = null, loadedOnce = false, retryTimer = null;

// live !rob countdown — ticked once a second, independent of the 20s data
// refresh (which reseeds robRemaining from the server)
function tickCooldown() {
  clearTimeout(cooldownTimer);
  const el = $("rob-status");
  if (!loggedIn || crimeOff) return;
  if (robRemaining <= 0) {
    el.innerHTML = "🦹 Ready! Type <b>!rob [user]</b> in chat to make your move.";
    return;
  }
  const m = Math.floor(robRemaining / 60), s = robRemaining % 60;
  el.innerHTML = "⏳ Next <b>!rob</b> attempt in <b>" + (m ? m + "m " : "") + s + "s</b>";
  robRemaining--;
  cooldownTimer = setTimeout(tickCooldown, 1000);
}

function fmtLeft(s) {
  const h = Math.floor(s / 3600), m = Math.floor((s % 3600) / 60);
  return h ? `${h}h ${m}m` : `${m}m`;
}

async function load() {
  let d;
  clearTimeout(retryTimer);
  try {
    const ctl = new AbortController();
    const timeout = setTimeout(() => ctl.abort(), 15000);
    const r = await fetch("/api/crime", { signal: ctl.signal });
    clearTimeout(timeout);
    if (!r.ok) throw new Error("HTTP " + r.status);
    d = await r.json();
  } catch (e) {
    if (!loadedOnce) {
      $("rob-status").innerHTML = "⚠️ Couldn't load crime data — retrying…";
      // before anything has rendered a blank page is all the user sees, so
      // retry quickly instead of waiting out the 20s poll
      retryTimer = setTimeout(load, 5000);
    }
    return;
  }
  loadedOnce = true;
  loggedIn = d.logged_in;
  crimeOff = d.enabled === false;
  if (crimeOff) {
    $("rob-status").innerHTML = "🚫 Robbing is currently disabled.";
  } else if (!loggedIn) {
    $("rob-status").innerHTML =
      '<a href="/auth/login" style="color:var(--blue-text);">Log in with Kick</a> to rob and buy gear.';
  } else {
    robRemaining = d.rob_remaining || 0;
    tickCooldown();
  }

  if (d.balance !== undefined)
    $("balance").textContent = d.balance.toLocaleString() + " credits";

  // standing effects from what this user owns, right under the timer
  const bits = [];
  if (d.bodyguard_seconds > 0) bits.push(`🕴️ bodyguard active — ${fmtLeft(d.bodyguard_seconds)} left`);
  if (d.alarm_seconds > 0) bits.push(`🔔 silent alarm armed — ${fmtLeft(d.alarm_seconds)} left`);
  if (d.cooldown_reduction_pct) bits.push(`ride: -${d.cooldown_reduction_pct}% !rob cooldown`);
  if (d.steal_bonus_pct) bits.push(`🕵️ fence: +${d.steal_bonus_pct}% steal cap`);
  if (d.safe_pct) bits.push(`🧰 ${d.safe_pct}% of your balance locked away`);
  if (d.shelter_pct) bits.push(`🏢 ${d.shelter_pct}% of your robbable stocks sheltered`);
  if (d.deterrent_pct) bits.push(`🐕 guard dog: -${d.deterrent_pct} points off robbers' odds`);
  $("effects").textContent = bits.join(" · ");

  renderGear(d);
}

function renderGear(d) {
  const card = $("gear-card");
  if (crimeOff || !d.catalog || !d.catalog.length) { card.style.display = "none"; return; }
  card.style.display = "";

  const body = $("gear");
  body.innerHTML = "";
  for (const g of d.catalog) {
    const tr = document.createElement("tr");
    const what = document.createElement("td");
    const nm = document.createElement("b");
    nm.textContent = `${g.emoji} ${g.name}`;
    const desc = document.createElement("span");
    desc.style.cssText = "display:block;color:var(--muted);font-size:12px;";
    desc.textContent = g.description;
    what.append(nm, desc);
    const cost = document.createElement("td");
    cost.style.cssText = "text-align:right;font-family:var(--mono);white-space:nowrap;color:var(--gold);";
    cost.textContent = fmt(g.price);
    const act = document.createElement("td");
    act.style.cssText = "text-align:right;white-space:nowrap;";
    const oneTime = g.kind !== "consumable" && g.kind !== "timed";
    if (oneTime && g.owned) {
      const tag = document.createElement("span");
      tag.className = "chip fulfilled";
      tag.textContent = "OWNED";
      act.appendChild(tag);
    } else {
      if (g.kind === "consumable" && g.owned) {
        const n = document.createElement("span");
        n.style.cssText = "color:var(--muted);font-size:12px;margin-right:8px;";
        n.textContent = g.owned + "× held";
        act.appendChild(n);
      }
      const timedLeft = g.kind === "timed" ? (g.seconds_left || 0) : 0;
      if (timedLeft > 0) {
        const n = document.createElement("span");
        n.style.cssText = "color:var(--muted);font-size:12px;margin-right:8px;";
        n.textContent = fmtLeft(timedLeft) + " left";
        act.appendChild(n);
      }
      const b = document.createElement("button");
      b.className = "btn primary";
      b.style.cssText = "font-size:12px;padding:4px 10px;";
      b.textContent = timedLeft > 0 ? "Extend" : "Buy";
      b.disabled = !loggedIn;
      b.onclick = async () => {
        const r = await fetch("/api/crime/gear", {
          method: "POST", headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ item: g.key }),
        });
        const res = await r.json();
        if (r.ok && window.sfx) sfx.cash();
        if (r.ok && res.balance !== undefined)
          $("balance").textContent = res.balance.toLocaleString() + " credits";
        // the reload repaints the table, so the outcome message goes up
        // last or it'd be racing a render that clears the row it sits under
        await load();
        const hint = $("gear-hint");
        hint.className = "hint " + (r.ok ? "ok" : "err");
        hint.textContent = r.ok ? res.message : res.error;
      };
      act.appendChild(b);
    }
    tr.append(what, cost, act);
    body.appendChild(tr);
  }
}

load();
setInterval(load, 20000);
// phones suspend timers in background tabs — refresh as soon as we're back
document.addEventListener("visibilitychange", () => {
  if (!document.hidden) load();
});
</script>
