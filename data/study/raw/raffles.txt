const $ = (id) => document.getElementById(id);
const fmt = (n) => Math.floor(n).toLocaleString();

const st = {
  open: [], past: [], logged: false, admin: false, bal: 0,
  enabled: true, maxPerBuy: 10000,
  counts: {},   // raffle id -> what's typed in its ticket box
  hints: {},    // raffle id -> {text, ok}
  boards: new Set(),  // raffle ids whose odds board the viewer opened
  loaded: false,
  sig: null,    // last rendered payload — an unchanged poll repaints nothing
  seen: {},     // raffle id -> {status, drawn_at} as of the last poll
  seenReady: false,   // false until the first poll lands (see watchForDraws)
  revealing: false,   // a winner reveal is on screen — hold the repaints
};

// Prize images are cached by node, not just by URL: the poll rebuilds the
// cards, and a fresh <img> would re-request and flash on every refresh.
// Moving the same element into the new card keeps the decoded image.
const artCache = new Map();

function prizeImage(r) {
  const key = r.id + "|" + r.image_url;
  let img = artCache.get(key);
  if (!img) {
    img = document.createElement("img");
    img.className = "art";
    img.alt = "";
    img.loading = "lazy";
    // a dead link drops the frame instead of leaving a broken-image box —
    // remembered, so the next render doesn't put it back
    img.onerror = () => { img.dataset.broken = "1"; img.remove(); };
    // .src, never innerHTML — the link is streamer-supplied text
    img.src = r.image_url;
    artCache.set(key, img);
  }
  return img.dataset.broken ? null : img;
}

async function refresh() {
  let d;
  try {
    const r = await fetch("/api/raffles");
    if (!r.ok) throw new Error("HTTP " + r.status);
    d = await r.json();
  } catch (e) {
    if (!st.loaded) $("open-empty").textContent = "⚠️ Couldn't load raffles — retrying…";
    return;
  }
  st.loaded = true;
  st.open = d.open || [];
  st.past = d.past || [];
  st.logged = d.logged_in;
  st.admin = d.is_admin;
  st.enabled = d.enabled !== false;
  st.maxPerBuy = d.max_per_buy || 10000;
  if (d.balance !== undefined) {
    st.bal = d.balance;
    $("balance").textContent = d.balance.toLocaleString() + " credits";
  }
  watchForDraws();
  render();
}

// A raffle that just flipped to 'drawn' — or a drawn one whose winner changed,
// which is a redraw — gets the reel instead of a name quietly appearing in the
// card. This runs off the poll, so viewers who never touched the draw button
// see it too, a few seconds behind the streamer at worst.
function watchForDraws() {
  const now = {};
  for (const r of st.open.concat(st.past)) now[r.id] = { status: r.status, drawn_at: r.drawn_at };
  const before = st.seen;
  st.seen = now;
  if (!st.seenReady) { st.seenReady = true; return; }  // first load reveals nothing
  if (st.revealing || document.hidden) return;         // rAF is frozen in a hidden tab

  for (const r of st.past) {
    const was = before[r.id];
    if (r.status !== "drawn" || !was) continue;
    const fresh = was.status === "open";
    const redrawn = was.status === "drawn" && was.drawn_at !== r.drawn_at;
    // a draw we only learn about minutes later (laptop asleep, tab in the
    // background) is history, not a moment — show it in the card instead
    if ((fresh || redrawn) && Date.now() / 1000 - r.drawn_at < 45) {
      startReveal(r, redrawn);
      return;
    }
  }
}

function ago(ts) {
  const s = Math.max(0, Math.floor(Date.now() / 1000 - ts));
  if (s < 60) return s + "s ago";
  if (s < 3600) return Math.floor(s / 60) + "m ago";
  if (s < 86400) return Math.floor(s / 3600) + "h ago";
  return Math.floor(s / 86400) + "d ago";
}

function until(ts) {
  const s = Math.max(0, ts - Math.floor(Date.now() / 1000));
  if (s < 60) return s + "s";
  if (s < 3600) return Math.floor(s / 60) + "m " + (s % 60) + "s";
  if (s < 86400) return Math.floor(s / 3600) + "h " + Math.floor((s % 3600) / 60) + "m";
  return Math.floor(s / 86400) + "d " + Math.floor((s % 86400) / 3600) + "h";
}

function render() {
  // never repaint out from under someone mid-type (the poll runs every 8s)
  const active = document.activeElement;
  if (active && (active.classList?.contains("tickets") || active.closest?.(".modal"))) return;
  // the reveal is the one telling the story — painting the winner into the
  // card behind it would spoil the reel that's still spinning
  if (st.revealing) return;

  // Most polls bring back exactly what's already on screen. Rebuilding the
  // cards for those is pure churn the viewer sees, so compare first and bail.
  // The minute bucket is in the signature because the "entries close in…"
  // countdown is rendered text — it has to repaint as often as it changes,
  // and no more.
  const sig = JSON.stringify([st.open, st.past, st.logged, st.admin, st.bal,
                              st.enabled, st.hints,
                              st.open.some((r) => r.ends_at) ? Math.floor(Date.now() / 60000) : 0]);
  if (sig === st.sig) return;
  st.sig = sig;

  $("new-btn").style.display = st.admin ? "" : "none";

  const openBox = $("open-list");
  openBox.innerHTML = "";
  for (const r of st.open) openBox.appendChild(card(r));
  $("open-empty").style.display = st.open.length ? "none" : "block";
  $("open-empty").textContent = !st.enabled
    ? "Raffles are switched off right now — check back later!"
    : st.admin
      ? "No raffles running — hit + New raffle to put a prize up."
      : "No raffle running right now. Next one will show up here!";

  $("past-box").style.display = st.past.length ? "" : "none";
  const pastBox = $("past-list");
  pastBox.innerHTML = "";
  for (const r of st.past) pastBox.appendChild(card(r));
}

function fact(label, value, cls) {
  const d = document.createElement("div");
  const l = document.createElement("div");
  l.className = "f-label";
  l.textContent = label;
  const v = document.createElement("div");
  v.className = "f-value" + (cls ? " " + cls : "");
  v.textContent = value;
  d.append(l, v);
  return d;
}

function card(r) {
  const card = document.createElement("section");
  card.className = "card raffle";

  const top = document.createElement("div");
  top.className = "top";
  if (r.image_url) {
    const img = prizeImage(r);
    if (img) top.appendChild(img);
  }
  const head = document.createElement("div");
  head.className = "head";
  const prize = document.createElement("div");
  prize.className = "prize";
  prize.textContent = "🎟️ " + r.prize;
  if (r.status === "open") {
    const tag = document.createElement("span");
    tag.className = "tag " + (r.closed ? "shut" : "live");
    tag.textContent = r.closed ? "ENTRIES CLOSED" : "LIVE";
    tag.style.marginLeft = "8px";
    prize.appendChild(tag);
  }
  head.appendChild(prize);
  if (r.description) {
    const desc = document.createElement("div");
    desc.className = "desc";
    desc.textContent = r.description;
    head.appendChild(desc);
  }

  const facts = document.createElement("div");
  facts.className = "facts";
  facts.appendChild(fact("Ticket price", fmt(r.ticket_price), "gold"));
  // a disqualification pulls tickets out of the hat, so this is what's live
  facts.appendChild(fact(r.past_draws?.length ? "Tickets in play" : "Tickets sold",
                         fmt(r.total_tickets)));
  facts.appendChild(fact("Entrants", fmt(r.entrants)));
  if (r.your_tickets > 0) {
    facts.appendChild(fact("Your tickets", fmt(r.your_tickets)));
    facts.appendChild(fact(r.status === "open" ? "Your odds" : "Your odds were",
                           r.your_pct.toFixed(1) + "%", "odds"));
  }
  head.appendChild(facts);

  if (r.your_tickets > 0) {
    const bar = document.createElement("div");
    bar.className = "oddsbar";
    const i = document.createElement("i");
    i.style.width = Math.min(100, r.your_pct) + "%";
    bar.appendChild(i);
    head.appendChild(bar);
  }
  top.appendChild(head);
  card.appendChild(top);

  if (r.status === "open") card.appendChild(buyRow(r));
  if (r.status === "drawn") card.appendChild(winnerBanner(r));
  if (r.past_draws && r.past_draws.length) card.appendChild(redrawRecord(r));
  if (r.you_disqualified) {
    const n = document.createElement("div");
    n.className = "dq-note";
    n.textContent = "Your entry was disqualified from this raffle — your tickets "
      + "were taken out of the draw. Ask the streamer if you think that's wrong.";
    card.appendChild(n);
  }
  if (r.status === "cancelled") {
    const c = document.createElement("div");
    c.className = "cancelled";
    c.textContent = "This raffle was called off — every ticket was refunded.";
    card.appendChild(c);
  }

  if (r.top && r.top.length) card.appendChild(board(r));
  if (st.admin) card.appendChild(adminBar(r));

  const hint = st.hints[r.id];
  if (hint) {
    const el = document.createElement("div");
    el.className = "hint " + (hint.ok ? "ok" : "err");
    el.textContent = hint.text;
    card.appendChild(el);
  }
  return card;
}

function buyRow(r) {
  const wrap = document.createElement("div");

  if (r.closed) {
    const line = document.createElement("div");
    line.className = "status-line";
    line.textContent = "🔒 Entries are closed — the winner gets drawn on stream.";
    wrap.appendChild(line);
    return wrap;
  }
  if (!st.logged) {
    const line = document.createElement("div");
    line.className = "status-line";
    line.innerHTML = '<a href="/auth/login" style="color:var(--blue-text);">Log in with Kick</a>'
      + " to buy tickets — or type <b>!raffle [tickets]</b> in chat.";
    wrap.appendChild(line);
    return wrap;
  }

  const row = document.createElement("div");
  row.className = "buy";

  const box = document.createElement("input");
  box.type = "number";
  box.min = "1";
  box.className = "tickets";
  box.placeholder = "Tickets";
  box.value = st.counts[r.id] || "";

  const cost = document.createElement("span");
  cost.className = "cost";

  // how many more this viewer may hold (a per-viewer cap, what they can
  // afford, and the per-purchase ceiling all bite)
  const capLeft = r.max_per_user ? Math.max(0, r.max_per_user - r.your_tickets) : Infinity;
  const affordable = Math.floor(st.bal / r.ticket_price);
  const most = Math.min(capLeft, affordable, st.maxPerBuy);

  function paint() {
    const n = parseInt(box.value, 10);
    if (!n || n < 1) {
      cost.innerHTML = r.max_per_user
        ? `Limit ${fmt(r.max_per_user)} per viewer — you can buy ${fmt(capLeft)} more`
        : "Every ticket is one more entry in the hat";
      return;
    }
    const total = n * r.ticket_price;
    const after = r.total_tickets + n;
    const odds = ((r.your_tickets + n) / after) * 100;
    cost.innerHTML = total > st.bal
      ? `<b style="color:var(--red-text)">${fmt(total)}</b> — more than you have`
      : `<b>${fmt(total)}</b> credits → ${odds.toFixed(1)}% odds`;
  }
  box.addEventListener("input", () => { st.counts[r.id] = box.value; paint(); });

  const buy = document.createElement("button");
  buy.className = "btn primary";
  buy.textContent = "Buy tickets";
  buy.onclick = async () => {
    const n = parseInt(box.value, 10);
    if (!n || n < 1) { box.focus(); return; }
    buy.disabled = true;
    const res = await fetch("/api/raffles/buy", {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ raffle_id: r.id, tickets: n }),
    });
    const d = await res.json();
    st.hints[r.id] = { ok: res.ok, text: res.ok ? d.message : d.error };
    if (res.ok) {
      if (window.sfx) sfx.cash();
      st.counts[r.id] = "";
      if (d.balance !== undefined) {
        st.bal = d.balance;
        $("balance").textContent = d.balance.toLocaleString() + " credits";
      }
      setTimeout(() => { delete st.hints[r.id]; render(); }, 6000);
    }
    box.blur();  // let the repaint through
    await refresh();
  };

  row.append(box);
  for (const n of [1, 5, 10, 25]) {
    if (n > most) continue;
    const q = document.createElement("button");
    q.className = "btn quick";
    q.textContent = "+" + n;
    q.onclick = () => {
      box.value = Math.min(most, (parseInt(box.value, 10) || 0) + n);
      st.counts[r.id] = box.value;
      paint();
    };
    row.appendChild(q);
  }
  if (most >= 1 && most !== Infinity) {
    const max = document.createElement("button");
    max.className = "btn quick";
    max.textContent = "Max";
    max.title = "As many as you can afford";
    max.onclick = () => { box.value = most; st.counts[r.id] = box.value; paint(); };
    row.appendChild(max);
  }
  row.append(buy, cost);
  paint();
  wrap.appendChild(row);

  if (r.ends_at) {
    const line = document.createElement("div");
    line.className = "status-line";
    line.innerHTML = "⏳ Entries close in <b>" + until(r.ends_at) + "</b>";
    wrap.appendChild(line);
  }
  return wrap;
}

/* ---------- the reveal ----------
   The winner is already decided server-side before any of this runs — the
   reel is theatre over a settled result, not part of picking it. It can be
   skipped or missed entirely without changing anything. */

const ROW = 56;          // height of one .rv-name — keep in step with the CSS
const SPIN_MS = 5200;    // reel spins for this long before it settles
const SPIN_STEPS = 34;   // names that go past on the way
const FAST_TIME = 0.45;  // fraction of the spin at full speed…
const FAST_STEPS = 0.7;  // …which covers this fraction of the names

const rv = { raf: 0, timer: 0, idx: -1, tick: 0, r: null, redrawn: false, spinning: false };

// Names for the reel, weighted by tickets so the people with the real odds
// flash past most often. The odds board leaves out viewers who hid
// themselves, so they stay off the reel too — but the winner is always in it.
function reelNames(r) {
  const pool = [];
  for (const e of r.top || []) {
    const weight = Math.max(1, Math.round(e.pct / 4));
    for (let i = 0; i < weight; i++) pool.push(e.username);
  }
  const unique = new Set(pool);
  const items = [];
  let last = null;
  const next = () => {
    let pick;
    if (unique.size > 1) {
      // two of the same name in a row reads as a stuck reel, not a spin
      do { pick = pool[Math.floor(Math.random() * pool.length)]; } while (pick === last);
    } else {
      // everyone who entered is hidden, or there's only one of them — spin
      // ticket numbers instead and let the name land at the end
      pick = "🎟️ #" + fmt(Math.floor(Math.random() * Math.max(1, r.total_tickets)) + 1);
    }
    last = pick;
    return pick;
  };
  for (let i = 0; i < SPIN_STEPS; i++) items.push(next());
  items.push(r.winner_name);   // index SPIN_STEPS — where the reel settles
  last = r.winner_name;
  // a couple of names past the winner, so the band isn't sitting on an empty
  // reel once it stops
  items.push(next(), next());
  return items;
}

const tapWord = () => (matchMedia("(hover: none)").matches ? "Tap" : "Click");

function reelPos(t) {
  // full speed, then a long tail: the last two names crawl into the band,
  // which is the whole point of watching
  if (t >= 1) return SPIN_STEPS;
  if (t < FAST_TIME) return SPIN_STEPS * FAST_STEPS * (t / FAST_TIME);
  const u = (t - FAST_TIME) / (1 - FAST_TIME);
  return SPIN_STEPS * (FAST_STEPS + (1 - FAST_STEPS) * (1 - (1 - u) * (1 - u)));
}

function setHot(idx) {
  const strip = $("rv-strip");
  strip.querySelector(".hot")?.classList.remove("hot");
  strip.children[idx]?.classList.add("hot");
}

function startReveal(r, redrawn) {
  st.revealing = true;
  rv.r = r;
  rv.redrawn = redrawn;
  rv.idx = -1;
  rv.tick = 0;

  const strip = $("rv-strip");
  strip.innerHTML = "";
  strip.style.transform = "translateY(" + ROW + "px)";
  for (const name of reelNames(r)) {
    const row = document.createElement("div");
    row.className = "rv-name";
    row.textContent = name;   // .textContent — entrant names are viewer input
    strip.appendChild(row);
  }
  $("rv-confetti").innerHTML = "";
  $("reveal").classList.remove("landed");
  $("rv-eyebrow").textContent = redrawn
    ? "Redrawing — the last ticket was disqualified" : "Drawing the winner";
  $("rv-prize").textContent = "🎟️ " + r.prize;
  $("rv-sub").textContent = fmt(r.total_tickets) + " tickets · "
    + fmt(r.entrants) + " entrant" + (r.entrants === 1 ? "" : "s") + " in the hat";
  $("rv-ticket").textContent = "";
  $("rv-foot").textContent = "";
  $("rv-dismiss").textContent = tapWord() + " anywhere to skip";
  $("reveal").classList.add("show");

  if (matchMedia("(prefers-reduced-motion: reduce)").matches) { land(); return; }

  rv.spinning = true;
  const t0 = performance.now();
  const step = (now) => {
    const t = Math.min(1, (now - t0) / SPIN_MS);
    const p = reelPos(t);
    strip.style.transform = "translateY(" + (ROW - p * ROW) + "px)";
    const idx = Math.min(SPIN_STEPS, Math.floor(p));
    if (idx !== rv.idx) {
      rv.idx = idx;
      setHot(idx);
      // motion blur eases off with the reel; the last names are sharp
      strip.style.filter = "blur(" + (1.5 * (1 - idx / SPIN_STEPS)).toFixed(2) + "px)";
      $("rv-ticket").textContent = "🎟️ #"
        + fmt(Math.floor(Math.random() * Math.max(1, r.total_tickets)) + 1);
      if (window.sfx && now - rv.tick > 55) { rv.tick = now; sfx.tick(); }
    }
    if (t < 1) rv.raf = requestAnimationFrame(step); else land();
  };
  rv.raf = requestAnimationFrame(step);
}

function land() {
  cancelAnimationFrame(rv.raf);
  rv.spinning = false;
  const r = rv.r;
  const strip = $("rv-strip");
  strip.style.filter = "";
  strip.style.transform = "translateY(" + (ROW - SPIN_STEPS * ROW) + "px)";
  setHot(SPIN_STEPS);

  $("reveal").classList.add("landed");
  $("rv-eyebrow").textContent = rv.redrawn ? "🏆 Redrawn winner" : "🏆 Winner";
  $("rv-ticket").textContent = "Ticket #" + fmt(r.winning_ticket) + " of " + fmt(r.total_tickets);
  const mine = (r.top || []).find((e) => e.username === r.winner_name);
  $("rv-foot").textContent = mine
    ? `Held ${fmt(mine.tickets)} ticket${mine.tickets === 1 ? "" : "s"} — a ${mine.pct.toFixed(1)}% shot at “${r.prize}”`
    : `Wins “${r.prize}”`;
  $("rv-dismiss").textContent = tapWord() + " anywhere to close";
  confetti();
  if (window.sfx) sfx.bigwin();
  // long enough to read and screenshot, short enough that a forgotten tab
  // isn't stuck behind it
  rv.timer = setTimeout(closeReveal, 12000);
}

function confetti() {
  if (matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  const box = $("rv-confetti");
  const colors = ["#5ed69a", "#2f9e5f", "#e9e6df", "#ff8a8d", "#c87619"];
  for (let i = 0; i < 46; i++) {
    const bit = document.createElement("i");
    bit.style.left = (Math.random() * 100).toFixed(2) + "%";
    bit.style.background = colors[i % colors.length];
    bit.style.animationDuration = (1.5 + Math.random() * 1.4).toFixed(2) + "s";
    bit.style.animationDelay = (Math.random() * 0.5).toFixed(2) + "s";
    box.appendChild(bit);
  }
}

function closeReveal() {
  clearTimeout(rv.timer);
  cancelAnimationFrame(rv.raf);
  rv.spinning = false;
  $("reveal").classList.remove("show");
  $("rv-confetti").innerHTML = "";
  st.revealing = false;
  st.sig = null;   // the card changed while the repaints were held — paint it
  render();
}

$("reveal").onclick = () => { if (rv.spinning) land(); else closeReveal(); };

async function adminPost(url, body, raffleId) {
  const r = await fetch(url, {
    method: "POST", headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  const d = await r.json();
  st.hints[raffleId] = { ok: r.ok, text: r.ok ? (d.message || "Done") : (d.error || "Failed") };
  await refresh();
}

function adminBar(r) {
  const bar = document.createElement("div");
  bar.className = "admin-bar";
  if (r.status === "drawn") {
    // audit the winner's profile, and if they don't hold up, take them out
    // of the hat and pull again from everyone else
    const again = document.createElement("button");
    again.className = "btn";
    again.textContent = "⚠️ Disqualify winner & redraw";
    again.onclick = () => openRedraw(r);
    const note = document.createElement("span");
    note.style.cssText = "color:var(--muted);font-size:12px;margin-left:auto;";
    note.textContent = "Drawn — the ticket credits stay spent.";
    bar.append(again, note);
    return bar;
  }
  if (r.status !== "open") {
    const note = document.createElement("span");
    note.style.cssText = "color:var(--muted);font-size:12px;";
    note.textContent = "Cancelled and refunded.";
    bar.appendChild(note);
    return bar;
  }

  const edit = document.createElement("button");
  edit.className = "btn";
  edit.textContent = "Edit";
  edit.onclick = () => openModal(r);

  const toggle = document.createElement("button");
  toggle.className = "btn";
  toggle.textContent = r.closed ? "Reopen entries" : "Close entries";
  toggle.onclick = () => adminPost("/api/admin/raffles/close",
                                   { raffle_id: r.id, closed: !r.closed }, r.id);

  const draw = document.createElement("button");
  draw.className = "btn success";
  draw.textContent = "🎲 Draw winner";
  draw.disabled = r.total_tickets < 1;
  draw.onclick = async () => {
    if (!confirm(`Draw the winner of “${r.prize}” from ${fmt(r.total_tickets)} tickets? This is final.`))
      return;
    draw.disabled = true;
    // deliberately not adminPost: that hint line spells the winner out, which
    // would give it away before the reel has spun a single name. The refresh
    // below is what starts the reveal, for the streamer and chat alike.
    const res = await fetch("/api/admin/raffles/draw", {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ raffle_id: r.id }),
    });
    const d = await res.json();
    if (!res.ok) st.hints[r.id] = { ok: false, text: d.error || "Failed" };
    await refresh();
  };

  const cancel = document.createElement("button");
  cancel.className = "btn danger spacer";
  cancel.textContent = "Cancel & refund";
  cancel.onclick = () => {
    if (confirm(`Call off “${r.prize}” and refund all ${fmt(r.total_tickets)} tickets?`))
      adminPost("/api/admin/raffles/cancel", { raffle_id: r.id }, r.id);
  };

  bar.append(edit, toggle, draw, cancel);
  return bar;
}

// Entries being shut means the draw is the next thing that happens, so the
// poll tightens up — the reel should land on chat's screens while the
// streamer is still saying the name, not eight seconds later.
let pollTimer = null;

function schedulePoll() {
  clearTimeout(pollTimer);
  pollTimer = setTimeout(pollNow, st.open.some((r) => r.closed) ? 3000 : 8000);
