// When the admin turns the casino off, lock every game's action button so it
// can't be clicked (e.g. Plinko's Drop Ball) and grey the boards out. Each
// game's own updateUI() keeps re-computing its button's disabled state from
// bet validity, so a one-time disable won't stick — we watch each button and
// force it back to disabled whenever something tries to re-enable it.
let _casinoLocked = false;
function applyCasinoState(enabled) {
  const off = enabled === false;
  document.body.classList.toggle("casino-off", off);
  if (!off || _casinoLocked) return;
  _casinoLocked = true;
  for (const id of ["dropBtn", "sl-spin", "mn-btn", "dt-btn", "cs-btn", "kn-btn",
                    "bj-deal", "crash-btn", "cf-btn", "wh-btn", "rp-btn", "rp-open"]) {
    const b = $(id);
    if (!b) continue;
    b.disabled = true;
    new MutationObserver(() => { if (!b.disabled) b.disabled = true; })
      .observe(b, { attributes: true, attributeFilter: ["disabled"] });
  }
}

/* Every game's 2× button. Doubling past what you actually have just puts the
   whole balance in the box instead of a bet that would be rejected. The cap is
   skipped when there's no balance to cap against — not logged in, or the page
   hasn't heard back yet — so the button still behaves like plain doubling. */
function doubleBet(input) {
  const min = Math.max(1, Number(input.min) || 1);
  const want = Math.max(min, Math.floor(Number(input.value) || 0) * 2);
  const cap = Math.floor(BALANCE);
  input.value = cap >= min ? Math.min(want, cap) : want;
}

/* ================= SLOTS — "Le Shoovy" =================
   A 6×5 machine drawn with Pixi. The server rolls the whole grid before a reel
   moves; the reels spin on random filler and are handed their real symbols the
   moment they start braking, so what lands is exactly what was already paid. */
(() => {
  const stageEl = $("sl-stage"), btn = $("sl-spin"), msg = $("sl-msg"), hint = $("sl-hint");
  const DEFAULT_HINT = hint.textContent;
  const fmt = (n) => Math.round(n).toLocaleString();
  const credits = (n) => fmt(n) + (Math.round(n) === 1 ? " credit" : " credits");

    hint.textContent = "Graphics engine failed to load — check your connection.";

  const REELS = 6, ROWS = 5, BAND = ROWS + 1, CELL = 108;
  const GRID_W = REELS * CELL, GRID_H = ROWS * CELL;
  const STAGE_W = 760, STAGE_H = 768;
  const MACHINE_Y = 384;   // grid centre; title sits above, the readouts below

  /* Near-black cabinet with red kept for the premium symbols and the accents. */
  const RED = 0xc62828, RED_HOT = 0xe03131, INK = 0x0a0a0c;
  const PANEL = 0x141418, PANEL_DK = 0x0d0d10, HAIRLINE = 0x26262c;
  const BONE = "#e9e9ee";

  const app = new PIXI.Application({
    width: STAGE_W, height: STAGE_H, backgroundColor: INK, antialias: true,
    autoDensity: true, resolution: Math.min(window.devicePixelRatio || 1, 2),
  });
  stageEl.appendChild(app.view);
  // autoDensity writes the canvas size as an inline style, which no stylesheet
  // rule can beat — hand it back to the layout so the machine scales with the panel
  app.view.style.width = "100%";
  app.view.style.height = "auto";
  const BlurFilter = PIXI.BlurFilter || PIXI.filters.BlurFilter;  // v7 moved the filters out of PIXI.filters

  const st = {
    ready: false, loggedIn: false, running: false, mode: "manual",
    auto: false, autoLeft: null,
    pending: null, error: null, spinStart: 0, round: 0,
    watchdog: null, chain: null, pulse: null,
    visible: false, free: 0, stake: 0,
    spins: 0, wins: 0, best: 0, bestX: 0, profit: 0,

  /* ---------- the machine's own voice ----------
     A lever clack on the spin and a thunk per reel, synthesised like the rest
     of the page's audio. Wins reuse the shared casino sfx so the whole casino
     still sounds like itself. */
  const audio = (() => {
    let ctx = null, master = null, noiseBuf = null;
    function ac() {
      if (!ctx) {
        const AC = window.AudioContext || window.webkitAudioContext;
        if (!AC) return null;
        ctx = new AC();
        master = ctx.createGain();
        master.gain.value = 0.22;
        master.connect(ctx.destination);
      }
      if (ctx.state === "suspended") ctx.resume();
      return ctx;
    }
    function envelope(t0, dur, peak, attack) {
      const g = ctx.createGain();
      g.gain.setValueAtTime(0.0001, t0);
      g.gain.exponentialRampToValueAtTime(peak, t0 + (attack || 0.006));
      g.gain.exponentialRampToValueAtTime(0.0001, t0 + dur);
      g.connect(master);
      return g;
    }
    function tone(freq, dur, type, peak, glideTo) {
      const t0 = ctx.currentTime;
      const o = ctx.createOscillator();
      o.type = type || "sine";
      o.frequency.setValueAtTime(freq, t0);
      if (glideTo) o.frequency.exponentialRampToValueAtTime(glideTo, t0 + dur);
      o.connect(envelope(t0, dur, peak || 0.4));
      o.start(t0);
      o.stop(t0 + dur + 0.03);
    }
    function burst(dur, peak, freq, q) {
      const t0 = ctx.currentTime;
      if (!noiseBuf) {
        const len = Math.floor(ctx.sampleRate * 0.5);
        noiseBuf = ctx.createBuffer(1, len, ctx.sampleRate);
        const d = noiseBuf.getChannelData(0);
        for (let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1;
      }
      const src = ctx.createBufferSource();
      src.buffer = noiseBuf;
      const bp = ctx.createBiquadFilter();
      bp.type = "bandpass";
      bp.frequency.value = freq;
      bp.Q.value = q || 1;
      src.connect(bp);
      bp.connect(envelope(t0, dur, peak, 0.002));
      src.start(t0);
      src.stop(t0 + dur + 0.03);
    }
    const safe = (fn) => { try { if (ac()) fn(); } catch (e) {} };
    return {
      spin: () => safe(() => { burst(0.09, 0.5, 1800, 2); tone(180, 0.28, "sawtooth", 0.2, 620); }),
      reelStop: (i) => safe(() => { burst(0.07, 0.4, 320, 1.4); tone(120 - i * 6, 0.13, "triangle", 0.3, 62); }),
    };
  })();

  /* ---------- symbol art ---------- */
  let symDefs = [], textures = [], totalWeight = 0, SCATTER_ID = 10;

  const hexNum = (c) => parseInt(c.slice(1), 16);
  const isRed = (c) => { const n = hexNum(c); return ((n >> 16) & 255) > 0x90 && ((n >> 8) & 255) < 0x60; };

  function randomSymbolId() {
    let r = Math.random() * totalWeight;
    for (const d of symDefs) { r -= d.weight; if (r <= 0) return d.id; }
    return symDefs[0].id;
  }

  const isAnimated = (url) => /\.gif(\?|#|$)/i.test(url);

  // Every symbol wears a channel emote, so the art has to be in hand before the
  // reels are baked. A symbol whose emote never arrives falls back to the
  // letterform or suit it is named after, and the machine is built regardless.
  function loadSymbolArt(defs) {
    return Promise.all(defs.map((def) => new Promise((resolve) => {
      if (!def.image) return resolve();
      const img = new Image();
      img.crossOrigin = "anonymous";   // the tile is baked into a WebGL texture
      if (isAnimated(def.image)) {
        // a gif only advances while it is laid out, and the reels read their
        // frames straight off this element — so it lives, invisible, on the page
        img.setAttribute("aria-hidden", "true");
        img.style.cssText = "position:fixed;left:0;top:0;width:1px;height:1px;"
          + "opacity:0;pointer-events:none;z-index:-1";
        document.body.appendChild(img);
      }
      // a host that never answers must not hold the machine hostage
      const giveUp = setTimeout(resolve, 6000);
      const done = (art) => { clearTimeout(giveUp); def.art = art; resolve(); };
      img.onload = () => done(img);
      img.onerror = () => done(null);
      img.src = def.image;
    })));
  }

  // an emote inset into a tile, scaled to fit `box` without distorting it
  function emoteSprite(art, S, box) {
    const sp = new PIXI.Sprite(PIXI.Texture.from(art));
    sp.scale.set(Math.min(box / art.width, box / art.height));
    sp.anchor.set(0.5);
    sp.position.set(S / 2, S / 2);
    return sp;
  }

  // the plate the premium symbols sit on
  function symbolPlate(S, red) {
    const g = new PIXI.Graphics();
    g.beginFill(red ? 0x150c0e : 0x121216, 0.95).drawRoundedRect(8, 8, S - 16, S - 16, 14).endFill();
    g.lineStyle(1.5, red ? RED : 0x3a3a44, red ? 0.5 : 0.9).drawRoundedRect(8, 8, S - 16, S - 16, 14);
    return g;
  }

  // The tile a symbol is drawn on. `kind` decides how it is mounted — bare for
  // the lows, a plate for the premiums — and the emote takes the place of the
  // letterform inside that mount, so the tiers still read apart at a glance.
  function buildTile(def, S) {
    const tile = new PIXI.Container();
    if (def.kind === "rank") {
      // low symbols: bare engraved letterforms, no tile behind them
      if (def.art) { tile.addChild(emoteSprite(def.art, S, S - 16)); return tile; }
      const t = new PIXI.Text(def.label, new PIXI.TextStyle({
        fontFamily: 'Georgia, "Times New Roman", serif',
        fontSize: def.label.length > 1 ? 58 : 72, fontWeight: "700",
        fill: def.colour, stroke: "#000000", strokeThickness: 5, lineJoin: "round",
        letterSpacing: 1, dropShadow: true, dropShadowColor: "#000000",
        dropShadowAlpha: 0.8, dropShadowBlur: 3, dropShadowDistance: 3,
      }));
      t.anchor.set(0.5);
      t.position.set(S / 2, S / 2);
      tile.addChild(t);
    } else if (def.kind === "scatter") {
      const g = new PIXI.Graphics();
      g.beginFill(0x18080a).drawRoundedRect(7, 7, S - 14, S - 14, 12).endFill();
      g.lineStyle(2, RED, 0.9).drawRoundedRect(7, 7, S - 14, S - 14, 12);
      g.lineStyle(1, RED, 0.28).drawRoundedRect(13, 13, S - 26, S - 26, 8);
      tile.addChild(g);

        dropShadow: true, dropShadowColor: "#e03131", dropShadowAlpha: 0.55,
        dropShadowBlur: 8, dropShadowDistance: 0,
      }));
      t.anchor.set(0.5);
      t.position.set(S / 2, S / 2 + 1);
      tile.addChild(t);
    } else {
      // premiums: a dark plate with a hairline, glyph in bone or red
      const red = isRed(def.colour);
      tile.addChild(symbolPlate(S, red));
      if (def.art) { tile.addChild(emoteSprite(def.art, S, S - 26)); return tile; }
      const t = new PIXI.Text(def.label, new PIXI.TextStyle({
        fontFamily: 'Georgia, "Times New Roman", serif',
        fontSize: def.label === "7" ? 72 : 62, fontWeight: "700", fill: def.colour,
        dropShadow: true, dropShadowColor: red ? "#e03131" : "#000000",
        dropShadowAlpha: red ? 0.45 : 0.7, dropShadowBlur: red ? 10 : 4,
        dropShadowDistance: red ? 0 : 3,
      }));
      t.anchor.set(0.5);
      t.position.set(S / 2, S / 2 + 2);
      tile.addChild(t);
    }
    return tile;
  }

  // Tiles that hold a gif can't be baked once and forgotten — they keep their
  // container and are re-rendered into their own texture while the emote plays.
  const livingTiles = [];
  const GIF_FPS = 20;

  function bakeSymbols(defs) {
    const S = 100;
    return defs.map((def) => {
      const tile = buildTile(def, S);
      if (def.art && isAnimated(def.image)) {
        const tex = PIXI.RenderTexture.create({ width: S, height: S, resolution: 2 });
        app.renderer.render(tile, { renderTexture: tex });
        livingTiles.push({ tile, tex, base: PIXI.Texture.from(def.art).baseTexture });
        return tex;
      }
      const tex = app.renderer.generateTexture(tile, {
        resolution: 2, region: new PIXI.Rectangle(0, 0, S, S),
      });
      tile.destroy({ children: true });
      return tex;
    });
  }

  // Re-uploading the <img> hands us whichever frame the browser is showing, so
  // the reels play the gif without anything having to decode it. This rides the
  // machine's own ticker, which is parked whenever the cabinet is off screen.
  function playAnimatedSymbols() {
    if (!livingTiles.length) return;
    let since = 0;
    app.ticker.add(() => {
      since += app.ticker.elapsedMS;
      if (since < 1000 / GIF_FPS) return;
      since = 0;
      for (const t of livingTiles) {
        t.base.update();
        app.renderer.render(t.tile, { renderTexture: t.tex });
      }
    });
  }

  /* ---------- the room ---------- */
  function paintScene() {
    const g = new PIXI.Graphics();
    g.beginFill(INK).drawRect(0, 0, STAGE_W, STAGE_H).endFill();
    // one soft ember of red behind the cabinet — the only colour in the room
    const steps = 26, maxR = Math.max(STAGE_W, STAGE_H) * 0.58;
    for (let i = steps; i > 0; i--) {
      const t = i / steps;
      g.beginFill(0x6e1219, 0.03 * (1 - t) + 0.004);
      g.drawCircle(STAGE_W / 2, STAGE_H * 0.42, maxR * t);
      g.endFill();
    }
    // vignette: darkness closing in from every edge
    const v = 16;
    for (let i = 0; i < v; i++) {
      const a = 0.055 * (1 - i / v), d = i / v;
      g.beginFill(0x000000, a);
      g.drawRect(0, 0, STAGE_W, STAGE_H * 0.16 * (1 - d) + 2);
      g.drawRect(0, STAGE_H - (STAGE_H * 0.2 * (1 - d)) - 2, STAGE_W, STAGE_H * 0.2 * (1 - d) + 2);
      g.drawRect(0, 0, STAGE_W * 0.14 * (1 - d) + 2, STAGE_H);
      g.drawRect(STAGE_W - (STAGE_W * 0.14 * (1 - d)) - 2, 0, STAGE_W * 0.14 * (1 - d) + 2, STAGE_H);
      g.endFill();
    }
    g.lineStyle(1, RED, 0.1).moveTo(0, STAGE_H * 0.74).lineTo(STAGE_W, STAGE_H * 0.74);
    // The room is 26 stacked circles the size of the stage and 64 translucent
    // rects over the top of them — a wall of blended overdraw that never
    // changes by a pixel. Bake it once and hang the result up as a single quad
    // rather than re-blending all ninety shapes on every frame of every spin.
    // One screen resolution is plenty: it is nothing but soft gradients.
    const baked = app.renderer.generateTexture(g, {
      resolution: 1, region: new PIXI.Rectangle(0, 0, STAGE_W, STAGE_H),
    });

    app.stage.addChild(new PIXI.Sprite(baked));

  /* ---------- reels ---------- */

  function buildMachine() {
    const machine = new PIXI.Container();
    machine.pivot.set(GRID_W / 2, GRID_H / 2);
    machine.position.set(STAGE_W / 2, MACHINE_Y);
    app.stage.addChild(machine);

    const cabinet = new PIXI.Graphics();
    cabinet.beginFill(PANEL).drawRoundedRect(-22, -22, GRID_W + 44, GRID_H + 44, 16).endFill();
    cabinet.lineStyle(1, HAIRLINE).drawRoundedRect(-22, -22, GRID_W + 44, GRID_H + 44, 16);
    cabinet.lineStyle(1, RED, 0.42).drawRoundedRect(-13, -13, GRID_W + 26, GRID_H + 26, 11);
    cabinet.lineStyle(0);
    cabinet.beginFill(PANEL_DK).drawRect(-4, -4, GRID_W + 8, GRID_H + 8).endFill();
    machine.addChild(cabinet);

    const reelContainer = new PIXI.Container();
    machine.addChild(reelContainer);

    for (let i = 0; i < REELS; i++) {
      const rc = new PIXI.Container();
      rc.x = i * CELL;
      reelContainer.addChild(rc);
      const reel = {
        container: rc, symbols: [], position: 0, previousPosition: 0,
        velocity: 0, state: "idle", spinUntil: Infinity, blur: new BlurFilter(8, 1),
      };
      // One vertical pass is all a motion-blurred reel needs. Quality 3 buys
      // three ping-ponged render targets per reel per frame — eighteen across
      // the machine — for a smoothness nothing travelling that fast can show.
      // Clamping the edge pixels pins the padding at zero, which matters more
      // than it looks: every write to blurY recomputes padding from the blur
      // strength, so without this the filter texture grew by up to 48px on all
      // four sides at exactly the moment the reels were at full speed.
      reel.blur.blurX = 0;
      reel.blur.blurY = 0;
      reel.blur.repeatEdgePixels = true;
      reel.blur.autoFit = false;
      reel.blur.resolution = 1;
      for (let j = 0; j < BAND; j++) {
        const s = new PIXI.Sprite(textures[0]);
        s.anchor.set(0.5);
        s.x = CELL / 2;
        s.y = j * CELL;
        setSymbol(s, randomSymbolId());
        reel.symbols.push(s);
        rc.addChild(s);

      reels.push(reel);

    mask.beginFill(0xffffff).drawRect(0, 0, GRID_W, GRID_H).endFill();
    machine.addChild(mask);

    lines.lineStyle(1, 0xffffff, 0.055);
    for (let c = 1; c < REELS; c++) lines.moveTo(c * CELL, 0).lineTo(c * CELL, GRID_H);
    for (let r = 1; r < ROWS; r++) lines.moveTo(0, r * CELL).lineTo(GRID_W, r * CELL);

      fontSize: 58, fontWeight: "700", fill: "#d92c2c",

    message.anchor.set(0.5);
    message.position.set(STAGE_W / 2, 700);

  function setSymbol(sprite, id) {
    sprite.symbolId = id;
    sprite.texture = textures[id];
    sprite.scale.set((CELL - 14) / sprite.texture.width);
  }
  function spriteAt(reel, row) {
    const p = Math.round(reel.position);
    return reel.symbols[(((row + 1 - p) % BAND) + BAND) % BAND];
  }

  /* ---------- tweens + spin physics (ported from the standalone machine) ----------
     Everything is timed off the wall clock rather than off accumulated frame
     deltas: a slow machine then drops animation smoothness instead of
     stretching the spin out. Only the integration step is clamped. */
  const tweening = [];
  const now = () => performance.now();
  const lerp = (a, b, t) => a * (1 - t) + b * t;
  const backout = (amount) => (t) => --t * t * ((amount + 1) * t + amount) + 1;
  const MAX_SPEED = 17, ACCEL = 55, SETTLE_MS = 620;

  function tweenTo(object, property, target, time, easing, oncomplete) {
    tweening.push({ object, property, target, easing, time,
                    from: object[property], complete: oncomplete, start: now() });
  }

  // The landing symbols are written the instant a reel starts braking: at rest
  // the sprite showing row r is the one where (position + index) % BAND == r+1,
  // so the whole band can be filled from the server's column in one pass. The
  // spare band slot is the one parked just above the window.
  function dressReel(reel, index, position) {
    const column = st.pending ? st.pending.grid[index] : null;
    for (let j = 0; j < BAND; j++) {
      const slot = (((position + j) % BAND) + BAND) % BAND;
      setSymbol(reel.symbols[j], slot === 0 || !column ? randomSymbolId() : column[slot - 1]);
    }
  }

  app.ticker.add(() => {
    const t = now();
    const dt = Math.min(app.ticker.elapsedMS, 50) / 1000;

    reels.forEach((r, i) => {
      if (r.state === "accel") {
        r.velocity = Math.min(MAX_SPEED, r.velocity + ACCEL * dt);
        if (r.velocity === MAX_SPEED) r.state = "spin";
      }
      if (r.state === "accel" || r.state === "spin") {
        r.position += r.velocity * dt;
        if (r.state === "spin" && t >= r.spinUntil) {
          r.state = "stop";
          const target = Math.ceil(r.position + r.velocity * 0.32);
          dressReel(r, i, target);
          tweenTo(r, "position", target, SETTLE_MS, backout(0.35), () => {
            r.state = "idle";
            r.velocity = 0;
            audio.reelStop(i);
            if (reels.every((x) => x.state === "idle")) reelsComplete();
          });
        }
      }
    });

    for (let i = tweening.length - 1; i >= 0; i--) {
      const tw = tweening[i];
      const phase = Math.min(1, (t - tw.start) / tw.time);
      tw.object[tw.property] = lerp(tw.from, tw.target, tw.easing(phase));
      if (phase === 1) {
        tw.object[tw.property] = tw.target;
        tweening.splice(i, 1);
        if (tw.complete) tw.complete();
      }
    }

    for (const r of reels) {
      const speed = (r.position - r.previousPosition) / dt;
      r.previousPosition = r.position;
      const blur = Math.min(24, Math.abs(speed) * 1.4);
      if (blur > 0.4) {
        r.blur.blurY = blur;
        if (!r.container.filters) r.container.filters = [r.blur];
      } else if (r.container.filters) {
        r.container.filters = null;
      }
      const wrapped = ((r.position % BAND) + BAND) % BAND;
      for (let j = 0; j < BAND; j++) {
        const s = r.symbols[j];
        const prevY = s.y;
        s.y = ((wrapped + j) % BAND) * CELL - CELL * 0.5;
        // A symbol that wraps back to the top gets a fresh face — but only
        // while the reel is still free-running, never once it's been dressed.
        if (s.y < prevY - CELL && (r.state === "accel" || r.state === "spin")) {
          setSymbol(s, randomSymbolId());
        }
      }
    }
  });

  // Park the renderer while the machine is off screen — or simply at rest.
  // A cabinet nobody has spun is a still picture, and redrawing it sixty times
  // a second costs the same blur passes, background overdraw and 36 sprite
  // transforms as a real spin does.
  //
  // A spin already in flight keeps its frames wherever the page is looking:
  // the reels coming down is what applies the payout, so it has to finish even
  // if you tab away mid-spin. The win pulse and the animated tiles are only
  // decoration and can wait until the cabinet is back on screen.
  function moving() {
    if (st.running || tweening.length > 0 || reels.some((r) => r.state !== "idle")) return true;
    return st.visible && (st.pulse !== null || livingTiles.length > 0);
  }
  // Anything that changes the stage while the ticker is parked has to come back
  // through here — it paints one last frame on the way down, so the machine is
  // never left showing a stale grid.
  function syncTicker() {
    if (moving()) { app.ticker.start(); return; }
    app.ticker.stop();
    app.render();
  }
  new IntersectionObserver((entries) => {
    st.visible = entries[0].isIntersecting;
    syncTicker();
  }).observe(stageEl);

  /* ---------- presentation ---------- */
  function setMessage(text) { if (message) message.text = text; }
  function updateBanner() {
    $("sl-free").textContent = st.free;
    if (!fsBanner) return;
    fsBanner.visible = st.free > 0;
    fsBanner.text = "FREE SPINS  " + st.free + " LEFT   ×" + (SLOT_INFO ? SLOT_INFO.fs_mult : 2);
  }
  function resetSymbols() {
    stopPulse();
    for (const r of reels) {
      for (const s of r.symbols) {
        s.alpha = 1;
        s.scale.set((CELL - 14) / s.texture.width);
      }
    }
  }
  function stopPulse() {
    if (!st.pulse) return;
    app.ticker.remove(st.pulse);
    st.pulse = null;
  }
  function highlight(cells) {
    const winners = new Set(cells);
    const sprites = [];
    for (let c = 0; c < REELS; c++) {
      for (let row = 0; row < ROWS; row++) {
        const s = spriteAt(reels[c], row);
        const win = winners.has(c + ":" + row);
        s.alpha = win ? 1 : 0.28;
        if (win) sprites.push(s);
      }
    }
    const base = sprites.map((s) => s.scale.x);
    const from = now();
    st.pulse = () => {
      const elapsed = now() - from;
      const k = 1 + Math.sin(elapsed / 90) * 0.07;
      sprites.forEach((s, i) => s.scale.set(base[i] * k));
      if (elapsed > 1500 || st.running) {
        sprites.forEach((s, i) => s.scale.set(base[i]));
        stopPulse();
        syncTicker();
      }
    };
    app.ticker.add(st.pulse);
  }

  function setControls() {

  /* One paid spin off the counter, once a spin has settled. Free spins are the
     feature playing itself out — nothing is staked, so they don't count against
     the run, and a feature won on the last bet still plays out in full. */
  function countAuto(d) {
    if (!st.auto || d.free || st.autoLeft === null) return;
    $("sl-auto-count").value = --st.autoLeft;
    if (st.autoLeft <= 0) stopAuto();
  }
  function queue(ms, fn) {
    st.chain = setTimeout(() => { st.chain = null; setControls(); fn(); }, ms);
    setControls();
  }

  /* ---------- a spin ---------- */
  function brake() {
    // even an instant answer gets a full spin out of the reels
    const t0 = Math.max(now(), st.spinStart + (st.error ? 250 : 700));
    reels.forEach((r, i) => { r.spinUntil = t0 + i * 230 + Math.random() * 120; });
  }

  async function spin() {
    if (st.running || !st.ready) return;
    clearTimeout(st.chain);
    st.chain = null;
    const free = st.free > 0;
    const bet = Math.floor(Number($("sl-bet").value) || 0);
    if (!free) {
      if (!(bet >= 1)) { msg.textContent = "Bet at least 1 credit."; stopAuto(); return; }
      if (bet > BALANCE) { msg.textContent = "Can't bet more than your balance!"; stopAuto(); return; }
    }
    msg.textContent = "";
    const round = ++st.round;   // a late answer from an abandoned spin is dropped
    st.running = true;
    st.pending = null;
    st.error = null;
    st.spinStart = now();
    setControls();
    syncTicker();
    resetSymbols();
    setMessage(free ? "Free spin" : "Good luck");
    hint.className = "hint";
    hint.textContent = free ? "Free spin — no stake deducted." : "Spinning…";
    audio.spin();
    for (const r of reels) { r.state = "accel"; r.spinUntil = Infinity; }
    // if the answer never arrives the reels still have to come down
    st.watchdog = setTimeout(() => {
      if (round === st.round && st.running && !st.pending && !st.error) {
        st.error = "No answer from the server — nothing was staked.";
        brake();
      }
    }, 9000);

    try {
      const r = await fetch("/api/slots/spin", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ bet }),
      });
      const d = await r.json();
      if (round !== st.round) return;
      if (!r.ok) { st.error = d.error; brake(); return; }
      st.pending = d;
      if (!d.free) setBalance(BALANCE - d.stake);   // the payout lands with the reels
      brake();
    } catch (e) {
      if (round !== st.round) return;
      st.error = "Connection lost — try again.";
      brake();
    } finally {
      clearTimeout(st.watchdog);
    }
  }

  function symbolName(id) {
    const d = symDefs.find((s) => s.id === id);
    return d ? d.label : "?";
  }
  function describe(d) {
    return d.wins.map((w) => w.symbol === SCATTER_ID
      ? w.reels + " scatters · " + w.multiplier + "×"
      : symbolName(w.symbol) + " on " + w.reels + " reels · " + w.multiplier + "×").join(", ");
  }

  function reelsComplete() {
    st.running = false;
    const d = st.pending;
    st.pending = null;

    if (!d) {   // the spin never happened — say so instead of pretending it did
      setMessage("—");
      hint.className = "hint loss";
      hint.textContent = st.error || "Spin cancelled — nothing was staked.";
      msg.textContent = st.error || "";
      st.error = null;
      stopAuto();
      setControls();
      syncTicker();
      return;
    }

    st.spins++;
    st.free = d.free_spins;
    st.stake = d.stake;
    st.profit += d.payout - (d.free ? 0 : d.stake);
    if (d.payout > 0) {
      st.wins++;
      st.best = Math.max(st.best, d.payout);
      st.bestX = Math.max(st.bestX, d.multiplier);
    }
    setBalance(d.balance);
    if (window.plinkoRefreshUI) plinkoRefreshUI();
    paintStats();
    updateBanner();
    if (d.cells.length) highlight(d.cells);

    if (window.sfx) {
      if (d.payout > 0) (d.awarded > 0 || d.multiplier >= 10 ? sfx.bigwin() : sfx.win());
      else if (!d.free) sfx.lose();
    }

    if (d.awarded > 0) {
      setMessage(d.scatters + " SCATTERS — " + d.awarded + " FREE SPINS");
      hint.className = "hint win";
      hint.textContent = d.scatters + " scatters — " + d.awarded + " free spins at "
        + fmt(d.stake) + " a spin, every win doubled!"
        + (d.payout > 0 ? " Scatter pay +" + fmt(d.payout) + "." : "");
    } else if (d.payout > 0) {
      setMessage((d.free ? "FREE SPIN WIN " : "WIN ") + fmt(d.payout));
      hint.className = "hint win";
      hint.textContent = "Paid " + credits(d.payout) + " — " + describe(d)
        + (d.free ? " (doubled)" : "") + " 🎉";
    } else {
      setMessage(d.free ? "No win — free spin" : "No win");
      hint.className = "hint loss";
      hint.textContent = d.free ? "No win on this free spin." : "No win — spin again 💀";
    }

    countAuto(d);
    setControls();
    syncTicker();

    if (st.free > 0) {                     // the feature drives itself
      queue(d.awarded > 0 ? 1900 : 1300, () => { if (!st.running) spin(); });
      return;
    }
    if (d.feature_done) {                  // let the last spin read before the summary
      queue(1700, () => {
        if (st.running) return;
        setMessage("FEATURE PAID " + fmt(d.feature_win));
        hint.className = "hint win";
        hint.textContent = "Free spins finished — the feature paid "
          + credits(d.feature_win) + " in total. 🎉";
        syncTicker();   // the ticker is parked by now — paint the summary
        if (window.sfx) sfx.bigwin();
        if (st.auto) queue(900, () => { if (st.auto && !st.running) spin(); });
      });
      return;
    }
    if (st.auto) {
      if (Math.floor(Number($("sl-bet").value) || 0) > BALANCE) {
        stopAuto();
        msg.textContent = "Autobet stopped — not enough credits for the next spin.";
        return;
      }
      queue(1200, () => { if (st.auto && !st.running) spin(); });
    }
  }

  /* ---------- paytable ---------- */
  let SLOT_INFO = null;

  function chip(s) {
    const cls = s.kind === "scatter" ? "scatter"
      : s.kind === "glyph" ? "glyph" + (isRed(s.colour) ? " red" : "") : "rank";
    const face = s.image
      ? '<img src="' + s.image + '" alt="' + s.label + '">'
      : s.label;
    return '<span class="sl-sym ' + cls + '" style="color:' + s.colour + '">' + face + "</span>";
  }
  function buildPaytable(info) {
    const groups = [];
    for (const s of info.symbols) {
      if (!s.pays) continue;
      const key = JSON.stringify(s.pays);
      const found = groups.find((g) => g.key === key);
      if (found) found.syms.push(s);
      else groups.push({ key, syms: [s], pays: s.pays });
    }
    const cols = [3, 4, 5, 6];
    let html = '<table class="sl-table"><thead><tr><th>Symbol</th>'
      + cols.map((c) => "<th>" + c + " reels</th>").join("") + "</tr></thead><tbody>";
    for (const g of groups) {
      html += "<tr><td>" + g.syms.map(chip).join("") + "</td>"
        + cols.map((c) => "<td>" + (g.pays[c] !== undefined ? g.pays[c] + "×" : "—") + "</td>").join("")
        + "</tr>";
    }
    html += "</tbody></table>";
    const scatter = info.symbols.find((s) => s.id === info.scatter_id);
    const pays = Object.keys(info.scatter_pays)
      .map((n) => n + (n === "5" ? "+" : "") + " pays " + info.scatter_pays[n] + "×").join(", ");
    const award = Object.keys(info.fs_award)
      .map((n) => n + (n === "5" ? "+" : "") + " → " + info.fs_award[n]).join(", ");
    html += '<div class="sl-fs">' + chip(scatter)
      + "<p><b>FREE SPINS</b>Scatters pay from anywhere on the grid (" + pays
      + ") and award free spins (" + award + "). The feature replays your triggering bet, "
      + "every win is multiplied by " + info.fs_mult
      + ", and 3 more scatters retrigger it. Free spins you're owed are kept safe — "
      + "they wait for you through a reload.</p></div>";
    $("sl-paytable").innerHTML = html;
  }

  /* ---------- wiring ---------- */
  btn.onclick = () => {
    if (st.mode === "manual") spin();
    else if (st.auto) stopAuto();
    else startAuto();
  };
  $("sl-manual").onclick = () => { if (window.sfx) sfx.click(); setMode("manual"); };
  $("sl-auto").onclick = () => { if (window.sfx) sfx.click(); setMode("auto"); };
  const betMin = () => Math.max(1, Number($("sl-bet").min) || 1);
  $("sl-half").onclick = () => {
    $("sl-bet").value = Math.max(betMin(), Math.floor(Number($("sl-bet").value || 0) / 2));
  };
  $("sl-double").onclick = () => doubleBet($("sl-bet"));

  btn.disabled = true;
  (async () => {
    const info = await (await fetch("/api/games/info")).json();
    applyCasinoState(info.casino_enabled);
    if (info.bet_min) { $("sl-bet").value = info.bet_min; $("sl-bet").min = info.bet_min; }
    if (!info.slots) { hint.textContent = "Slots are unavailable right now."; return; }

    SLOT_INFO = info.slots;
    SCATTER_ID = info.slots.scatter_id;
    symDefs = info.slots.symbols;
    totalWeight = symDefs.reduce((a, d) => a + d.weight, 0);
    await loadSymbolArt(symDefs);
    textures = bakeSymbols(symDefs);
    playAnimatedSymbols();
    paintScene();
    buildMachine();
    buildPaytable(info.slots);
    st.ready = true;
    st.loggedIn = info.logged_in;
    setControls();
    syncTicker();

    if (!info.logged_in) {
      hint.textContent = "Log in on the home page to play.";
      return;
    }
    hint.textContent = DEFAULT_HINT;
    try {
      const d = await (await fetch("/api/slots/state")).json();
      if (d.feature) {   // free spins owed from an earlier session

        st.stake = d.feature.stake;
        $("sl-bet").value = d.feature.stake;
        updateBanner();
        setControls();
        setMessage(st.free + " FREE SPINS WAITING");
        hint.className = "hint win";
        hint.textContent = "You still have " + st.free + " free spins at "
          + fmt(st.stake) + " a spin — press spin to play them out.";
        syncTicker();   // the banner landed on a machine that is already at rest
      }
    } catch (e) {}
  })();
})();

/* ================= BLACKJACK =================
   Felt-table blackjack: cards fly out of the shoe and the hole card flips in
   3D. The server is still the only authority — every deal/hit/stand round
   trips to /api/blackjack/*, and the animation just replays what came back. */
(() => {
  const shoeEl = $("bj-shoe"), toastEl = $("bj-toast"), hint = $("bj-hint");
  const dealerEl = $("bj-dealer"), dealerBadge = $("bj-dealer-badge"), playerRow = $("bj-player");
  const DEFAULT_HINT = hint.textContent;
  const fmt = (n) => Math.round(n).toLocaleString();
  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
  const reduced = window.matchMedia && matchMedia("(prefers-reduced-motion: reduce)").matches;
  const wait = (ms) => sleep(reduced ? 0 : ms);

  /* Card voices: a soft slide across the felt plus a struck wooden bar.
     Round outcomes keep using the shared sfx so the casino sounds alike. */
  const audio = (() => {
    let ctx = null, bus = null;
    function ac() {
      if (!ctx) {
        ctx = new (window.AudioContext || window.webkitAudioContext)();
        bus = ctx.createGain(); bus.gain.value = 0.5;
        const soften = ctx.createBiquadFilter();  // shave off any brittle top end
        soften.type = "lowpass"; soften.frequency.value = 4800; soften.Q.value = 0.3;
        bus.connect(soften); soften.connect(ctx.destination);
      }
      if (ctx.state === "suspended") ctx.resume();
      return ctx;
    }
    function wood(freq, t0, gain, decay) {
      const c = ac(), t = c.currentTime + t0;
      const o = c.createOscillator(), g = c.createGain();
      o.type = "sine";
      o.frequency.setValueAtTime(freq, t);
      o.frequency.exponentialRampToValueAtTime(freq * 0.88, t + Math.min(0.06, decay));
      g.gain.setValueAtTime(0.0001, t);
      g.gain.exponentialRampToValueAtTime(gain, t + 0.006);
      g.gain.exponentialRampToValueAtTime(0.0001, t + decay);
      o.connect(g); g.connect(bus);
      o.start(t); o.stop(t + decay + 0.05);
    }
    function slide(dur, cutoff, gain) {
      const c = ac(), t = c.currentTime;
      const len = Math.floor(c.sampleRate * dur);
      const buf = c.createBuffer(1, len, c.sampleRate), d = buf.getChannelData(0);
      for (let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1;
      const src = c.createBufferSource(); src.buffer = buf;
      const lp = c.createBiquadFilter(); lp.type = "lowpass";
      lp.frequency.setValueAtTime(cutoff, t);
      lp.frequency.exponentialRampToValueAtTime(cutoff * 0.45, t + dur);
      const g = c.createGain();
      g.gain.setValueAtTime(0.0001, t);
      g.gain.linearRampToValueAtTime(gain, t + 0.035);
      g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
      src.connect(lp); lp.connect(g); g.connect(bus);
      src.start(t); src.stop(t + dur + 0.02);
    }
    const safe = (fn) => { try { fn(); } catch (e) {} };
    return {
      deal: () => safe(() => { slide(0.16, 950, 0.045); wood(392, 0.05, 0.045, 0.16); }),
      flip: () => safe(() => { slide(0.12, 1400, 0.035); wood(523, 0.03, 0.035, 0.14); }),
      chip: () => safe(() => { wood(587, 0, 0.05, 0.18); wood(880, 0.055, 0.035, 0.16); }),
      bust: () => safe(() => [330, 262, 196].forEach((f, i) => wood(f, i * 0.075, 0.075, 0.35))),
    };
  })();

  // st.view is the last server state; `shown` mirrors what's on the table, so
  // each response only has to animate the cards that aren't out there yet.
  // Every flow that paints takes the next ticket (st.gen); an older flow parked
  // on an await sees its ticket go stale and stops touching the table. Two
  // writers on one row is how a resumed hand landing mid-deal used to leave
  // duplicate cards, a hole card that never flipped, and a badge adding up
  // cards that were never on the table.
  const st = { view: null, busy: false, loggedIn: false, hands: [], holeEl: null, gen: 0 };
  const shown = { dealer: [], hands: [] };
  const claim = () => ++st.gen;
  const stale = (ticket) => ticket !== st.gen;

  function handTotal(cards) {
    let total = 0, aces = 0;
    for (const code of cards) {
      const rank = code.slice(0, -1);
      if (rank === "A") { total += 11; aces++; }
      else if (["J", "Q", "K"].includes(rank)) total += 10;
      else total += Number(rank) || 0;
    }
    while (total > 21 && aces) { total -= 10; aces--; }
    return total;
  }

  // The animations only ever append, so they assume the table already holds a
  // prefix of the hand the server just described. Anything else — a round that
  // moved on in another tab, a repaint that landed mid-animation — has to be
  // painted from the server's view rather than animated onto a stale table.
  const isPrefix = (mirror, cards) =>
    mirror.length <= cards.length && mirror.every((code, i) => code === cards[i]);
  function mirrorMatches(game) {
    // mid-hand only the up card is comparable: the hole card is still "??"
    if (!isPrefix(shown.dealer, game.done ? game.dealer : [game.dealer[0]])) return false;
    if (shown.hands.length !== game.hands.length) return false;
    return game.hands.every((h, i) => isPrefix(shown.hands[i], h.cards));
  }

  /* ---------- cards ---------- */
  function setFace(el, code) {
    const front = el.querySelector(".bj-front");
    const suit = code && code !== "??" ? code.slice(-1) : "";
    front.classList.toggle("red", suit === "♥" || suit === "♦");
    front.querySelector(".bj-rank").textContent = suit ? code.slice(0, -1) : "";
    front.querySelector(".bj-suit").textContent = suit;
  }
  function makeCard(code, faceDown) {
    const el = document.createElement("div");
    el.className = "bj-card" + (faceDown ? " down" : "");
    el.innerHTML =
      '<div class="bj-card-inner">' +
        '<div class="bj-face bj-front"><div class="bj-rank"></div><div class="bj-suit"></div></div>' +
        '<div class="bj-face bj-back"><div class="bj-diamond">21</div></div>' +
      "</div>";
    setFace(el, code);
    return el;
  }
  // FLIP-style deal: start the card over the shoe, then let it settle in place.
  function flyFromShoe(el) {
    if (reduced) return;
    const from = shoeEl.getBoundingClientRect(), to = el.getBoundingClientRect();
    const dx = from.left + from.width / 2 - (to.left + to.width / 2);
    const dy = from.top + from.height / 2 - (to.top + to.height / 2);
    el.style.transition = "none";
    el.style.opacity = "0";
    el.style.transform = `translate(${dx}px, ${dy}px) rotate(-22deg) scale(.72)`;
    el.getBoundingClientRect();  // reflow
    requestAnimationFrame(() => {
      el.style.transition = "transform .46s cubic-bezier(.18,.85,.28,1), opacity .2s linear";
      el.style.opacity = "1";
      el.style.transform = "translate(0,0) rotate(0) scale(1)";
    });
  }
  async function dealCard(container, code, faceDown) {
    const el = makeCard(code, faceDown);
    container.appendChild(el);
    flyFromShoe(el);
    shoeEl.classList.add("pulse");
    audio.deal();
    setTimeout(() => shoeEl.classList.remove("pulse"), 280);
    await wait(330);
    return el;
  }

  /* ---------- seats & badges ---------- */
  function buildSeats(count) {
    playerRow.innerHTML = "";
    st.hands = []; shown.hands = [];
    for (let i = 0; i < count; i++) {
      const wrap = document.createElement("div");
      wrap.className = "bj-hand-wrap" + (count > 1 ? " split" : "");
      const cards = document.createElement("div"); cards.className = "bj-cards";
      const badge = document.createElement("div"); badge.className = "bj-badge"; badge.textContent = "0";
      wrap.append(cards, badge);  // column-reverse puts the badge above the cards
      playerRow.appendChild(wrap);
      st.hands.push({ wrap, cards, badge });
      shown.hands.push([]);
    }
  }
  function setBadge(el, text, cls) {
    el.className = "bj-badge" + (cls ? " " + cls : "");
    if (el.textContent !== String(text)) {
      el.textContent = text;
      el.classList.remove("pop"); void el.offsetWidth; el.classList.add("pop");
    }
  }
  function paintBadges() {
    const g = st.view, done = !!(g && g.done);
    if (!shown.dealer.length) {
      dealerBadge.className = "bj-badge hidden";
    } else {
      const v = handTotal(shown.dealer);
      let cls = "";
      if (done && v > 21) cls = "bust";
      else if (done && g.dealer.length === 2 && v === 21) cls = "bj";
      setBadge(dealerBadge, v, cls);
    }
    shown.hands.forEach((cards, i) => {
      const seat = st.hands[i];
      if (!seat) return;
      const h = g && g.hands[i];
      const result = h && h.result;
      let cls = "";
      if (result === "bust") cls = "bust";
      else if (result === "blackjack") cls = "bj";
      else if (result === "win") cls = "win";
      setBadge(seat.badge, handTotal(cards), cls);
      const multi = shown.hands.length > 1;
      seat.wrap.classList.toggle("active",
        multi && ((!done && !!g && i === g.active) || (done && (result === "win" || result === "blackjack"))));
      seat.wrap.classList.toggle("settled", multi && !done && !!h && h.done);
    });
  }

  /* ---------- toast & hint ---------- */
  let toastTimer = null;
  function showToast(text, kind) {
    toastEl.textContent = text;
    toastEl.className = "bj-toast show" + (kind ? " " + kind : "");
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => toastEl.classList.remove("show"), 2600);
  }
  function clearToast() { clearTimeout(toastTimer); toastEl.classList.remove("show"); }

  function setHint(game) {
    if (!game) { hint.className = "hint"; hint.textContent = DEFAULT_HINT; return; }
    if (game.done) {
      const net = (game.payout ?? 0) - game.bet;
      const msgs = { blackjack: "BLACKJACK! Paid 3:2 🎉", win: "You win! 🎉",
                     push: "Push — bet returned", lose: "Dealer wins 💀", bust: "Bust 💀" };
      hint.className = "hint " + (net > 0 ? "win" : net < 0 ? "loss" : "");
      hint.textContent = game.hands.length > 1
        ? (net > 0 ? "You win! +" + fmt(net) + " 🎉"
           : net === 0 ? "Push — bets returned"
           : "Dealer wins " + fmt(net) + " 💀")
        : (msgs[game.result] || game.result);
      return;
    }
    hint.className = "hint";
    const parts = ["Hit", "stand"];
    if (game.can_double) parts.push("double");
    if (game.can_split) parts.push("split");
    const opts = parts.length === 2 ? parts.join(" or ")
      : parts.slice(0, -1).join(", ") + ", or " + parts[parts.length - 1];
    hint.textContent = (game.hands.length > 1 ? "Hand " + (game.active + 1) + " of " + game.hands.length + ": " : "") + opts + ".";
  }

  /* ---------- stats ---------- */
  const stats = { hands: 0, wins: 0, bjs: 0, profit: 0 };
  function tally(game) {
    stats.hands++;
    stats.profit += (game.payout ?? 0) - game.bet;
    if (game.result === "win" || game.result === "blackjack") stats.wins++;
    if (game.hands.some((h) => h.result === "blackjack")) stats.bjs++;
    $("bj-hands").textContent = stats.hands;
    $("bj-wins").textContent = stats.wins;
    $("bj-bjs").textContent = stats.bjs;
    const p = $("bj-profit");
    p.textContent = (stats.profit < 0 ? "-" : "") + fmt(Math.abs(stats.profit));
    p.style.color = stats.profit > 0 ? "#62d96b" : stats.profit < 0 ? "var(--pk-danger)" : "";
  }

  /* ---------- controls ---------- */
  function setControls() {
    const g = st.view, playing = !!(g && !g.done), lock = st.busy;
    $("bj-hit").disabled = $("bj-stand").disabled = !playing || lock;
    $("bj-double").disabled = !playing || lock || !g.can_double;
    $("bj-split").disabled = !playing || lock || !g.can_split;
    $("bj-deal").disabled = playing || lock || !st.loggedIn;
    $("bj-bet").disabled = $("bj-half").disabled = $("bj-double-bet").disabled = playing || lock;
  }
  const betMin = () => Math.max(1, Number($("bj-bet").min) || 1);
  $("bj-half").addEventListener("click", () => {
    $("bj-bet").value = Math.max(betMin(), Math.floor(Number($("bj-bet").value || 0) / 2));
  });
  $("bj-double-bet").addEventListener("click", () => doubleBet($("bj-bet")));

  function pushBalance(balance) {
    if (balance === undefined) return;
    setBalance(balance);
    if (window.plinkoRefreshUI) plinkoRefreshUI();
  }

  /* ---------- animation flows ---------- */
  async function animateDeal(game, ticket) {
    dealerEl.innerHTML = "";
    shown.dealer = [];
    st.holeEl = null;
    dealerBadge.className = "bj-badge hidden";
    buildSeats(1);
    st.view = null;

    const cards = game.hands[0].cards, seat = st.hands[0];
    shown.hands[0].push(cards[0]);
    await dealCard(seat.cards, cards[0]);
    if (stale(ticket)) return false;
    paintBadges();
    shown.dealer.push(game.dealer[0]);
    await dealCard(dealerEl, game.dealer[0]);
    if (stale(ticket)) return false;
    paintBadges();
    shown.hands[0].push(cards[1]);
    await dealCard(seat.cards, cards[1]);
    if (stale(ticket)) return false;
    paintBadges();
    // the hole card carries its real face already when the round ended on a
    // natural — either way it stays flipped over until the reveal
    const hole = game.dealer[1];
    st.holeEl = await dealCard(dealerEl, hole === "??" ? "" : hole, true);
    if (stale(ticket)) return false;

    st.view = game;
    paintBadges();
    return true;
  }

  async function revealHole(game, ticket) {
    // no hole card on the table to turn over: paint the dealer's hand from the
    // view instead of dealing its cards on top of whatever is up there
    if (!st.holeEl) { paintDealer(game); return; }
    setFace(st.holeEl, game.dealer[1]);
    st.holeEl.classList.remove("down");
    st.holeEl = null;
    audio.flip();
    shown.dealer.push(game.dealer[1]);
    await wait(450);
    if (stale(ticket)) return;
    paintBadges();
  }

  // Deal out every card the server knows about that isn't on the table yet.
  async function dealNewCards(game, ticket) {
    for (let i = 0; i < game.hands.length; i++) {
      const seat = st.hands[i];
      if (!seat) continue;
      const cards = game.hands[i].cards;
      while (shown.hands[i].length < cards.length) {
        const code = cards[shown.hands[i].length];
        shown.hands[i].push(code);
        const el = await dealCard(seat.cards, code);
        if (stale(ticket)) return false;
        paintBadges();
        if (game.hands[i].result === "bust" && shown.hands[i].length === cards.length) {
          el.classList.add("busted");
          seat.wrap.classList.add("shake");
          setTimeout(() => seat.wrap.classList.remove("shake"), 450);
          audio.bust();
          await wait(500);
          if (stale(ticket)) return false;
        }
      }
    }
    return true;
  }

  // A split inserts a hand: rebuild the row with the cards that were already
  // dealt, so dealNewCards can fly in the fresh card for each half.
  function rebuildForSplit(prev, game) {
    const at = prev.active;
    const before = shown.hands.map((c) => c.slice());
    buildSeats(game.hands.length);
    game.hands.forEach((h, i) => {
      const base = (i === at || i === at + 1) ? [h.cards[0]]
        : (i < at ? before[i] : before[i - 1]) || [];
      base.forEach((code) => st.hands[i].cards.appendChild(makeCard(code, false)));
      shown.hands[i] = base.slice();
    });
    paintBadges();
  }

  async function settle(game, ticket) {
    st.view = game;
    await revealHole(game, ticket);
    if (stale(ticket)) return;
    for (let i = shown.dealer.length; i < game.dealer.length; i++) {
      await wait(300);
      if (stale(ticket)) return;
      shown.dealer.push(game.dealer[i]);
      await dealCard(dealerEl, game.dealer[i]);
      if (stale(ticket)) return;
      paintBadges();
    }
    paintBadges();
    await wait(300);
    if (stale(ticket)) return;
    finish(game);
    await wait(400);
  }

  // Closing out the round — kept apart from the animation so a table that had
  // to be repainted from the view still books the result.
  function finish(game) {
    const net = (game.payout ?? 0) - game.bet;
    const gotBJ = game.hands.some((h) => h.result === "blackjack");
    const allBust = game.hands.every((h) => h.result === "bust");
    if (gotBJ && net > 0) { showToast("BLACKJACK  +" + fmt(net), "bj"); if (window.sfx) sfx.bigwin(); }
    else if (net > 0) { showToast("Win  +" + fmt(net), ""); if (window.sfx) sfx.win(); }
    else if (net < 0) { showToast((allBust ? "Bust  −" : "Lose  −") + fmt(-net), "lose"); if (window.sfx) sfx.lose(); }
    else { showToast("Push", "push"); if (window.sfx) sfx.click(); }

    tally(game);
    paintBadges();
    setHint(game);
  }

  // Lay the dealer's row out exactly as the server describes it.
  function paintDealer(game) {
    dealerEl.innerHTML = "";
    shown.dealer = [];
    st.holeEl = null;
    game.dealer.forEach((code) => {
      const down = code === "??";
      const el = makeCard(down ? "" : code, down);
      dealerEl.appendChild(el);
      if (down) st.holeEl = el; else shown.dealer.push(code);
    });
    paintBadges();
  }

  // A full repaint from the server's view — takes the table off whatever flow
  // was animating it, so it is always safe to call.
  function renderStatic(game) {
    claim();
    st.view = game;
    paintDealer(game);
    buildSeats(game.hands.length);
    game.hands.forEach((h, i) => {
      h.cards.forEach((code) => st.hands[i].cards.appendChild(makeCard(code, false)));
      shown.hands[i] = h.cards.slice();
    });
    paintBadges();
    setHint(game);
    setControls();
  }

  /* ---------- server round trips ---------- */
  async function resume() {
    try {
      const d = await (await fetch("/api/blackjack/state")).json();
      if (d.game) { renderStatic(d.game); return true; }
    } catch (e) {}
    return false;
  }

  async function deal() {
    if (st.busy || (st.view && !st.view.done)) return;
    st.busy = true; setControls();
    let data = null, ok = false;
    try {
      const r = await fetch("/api/blackjack/start", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ bet: $("bj-bet").value }),
      });
      data = await r.json(); ok = r.ok;
    } catch (e) {}
    if (!ok || !data || !data.game) {
      hint.className = "hint loss";
      hint.textContent = (data && data.error) || "Couldn't deal — try again.";
      st.busy = false;
      await resume();  // a live hand (other tab, old reload) blocks dealing — pull it up
      setControls();
      return;
    }
    audio.chip();
    clearToast();
    pushBalance(data.balance);
    const ticket = claim();
    if (await animateDeal(data.game, ticket)) {
      if (data.game.done) await settle(data.game, ticket);
      else { paintBadges(); setHint(data.game); }
    }
    st.busy = false;
    setControls();
  }

  async function act(action) {
    if (st.busy || !st.view || st.view.done) return;
    const prev = st.view;
    st.busy = true; setControls();
    let data = null, ok = false;
    try {
      const r = await fetch("/api/blackjack/action", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ action }),
      });
      data = await r.json(); ok = r.ok;
    } catch (e) {}
    if (!ok || !data || !data.game) { st.busy = false; setControls(); return; }

    const game = data.game;
    pushBalance(data.balance);
    if (action === "split" && game.hands.length > prev.hands.length) {
      audio.chip();
      rebuildForSplit(prev, game);
      // prev.active can point past the hand list if the round moved on
      // elsewhere — the repaint below sorts that out, the chip just stays quiet
    } else if (action === "double" && game.hands[prev.active]?.bet > prev.hands[prev.active].bet) {
      audio.chip();
    }
    // The table has to be this hand's for the append-only animations to make
    // sense of it; if it isn't (a round played on in another tab, say), the
    // server's view is the truth and gets painted whole.
    if (!mirrorMatches(game)) {
      renderStatic(game);
      if (game.done) finish(game);
      st.busy = false;
      setControls();
      return;
    }
    const ticket = claim();
    st.view = game;
    if (await dealNewCards(game, ticket)) {
      if (game.done) await settle(game, ticket);
      else { paintBadges(); setHint(game); }
    }
    st.busy = false;
    setControls();
  }

  $("bj-deal").addEventListener("click", deal);
  for (const action of ["hit", "stand", "double", "split"]) {
    $("bj-" + action).addEventListener("click", () => act(action));
  }

  // keyboard shortcuts, only while the blackjack tab is the one on screen
  document.addEventListener("keydown", (e) => {
    if (e.ctrlKey || e.metaKey || e.altKey) return;
    const panel = document.querySelector('[data-tab-panel="blackjack"]');
    if (!panel || panel.style.display === "none") return;
    const tag = (e.target.tagName || "").toLowerCase();
    if (tag === "input" || tag === "select" || tag === "textarea") return;
    const key = { h: "bj-hit", s: "bj-stand", d: "bj-double", p: "bj-split" }[e.key.toLowerCase()];
    if (key && !$(key).disabled) { e.preventDefault(); $(key).click(); }
  });

  // resume a live hand after a reload
  (async () => {
    try {
      const info = await (await fetch("/api/games/info")).json();
      st.loggedIn = !!info.logged_in;
      if (info.bet_min) { $("bj-bet").value = info.bet_min; $("bj-bet").min = info.bet_min; }
      if (!st.loggedIn) {
        hint.textContent = "Log in on the home page to play.";
        setControls();
        return;
      }
      setControls();
      await resume();
    } catch (e) {
      // Deal starts out disabled so a click can't race this handshake; if the
      // handshake itself fails, hand the button back and let the server have
      // the final say on whether the round is allowed.
      st.loggedIn = true;
      setControls();
    }
  })();
})();

/* ================= MINES ================= */
(() => {
  const grid = $("mn-grid"), btn = $("mn-btn"), msg = $("mn-msg"), hint = $("mn-hint");
  const fmt = (n) => Math.floor(n).toLocaleString();
  const st = { active: false, bet: 0, profit: 0, busy: false };

  /* Tile artwork comes from the #mn-sprite symbols rather than emoji, so the
     gems and bombs render identically on every platform and can be sized off
     the tile. */
  const ICON = { gem: "gem-ico", bomb: "bomb-ico", boom: "boom-ico" };
  function setIcon(tile, name) {
    tile.innerHTML =
      `<svg class="mn-ico ${ICON[name]}" aria-hidden="true" focusable="false">` +
      `<use href="#mn-${name}"></use></svg>`;
  }

  const tiles = [];
  for (let i = 0; i < 25; i++) {
    const t = document.createElement("button");
    t.className = "mn-tile idle";
    t.disabled = true;
    t.onclick = () => pick(i);
    grid.appendChild(t);
    tiles.push(t);
  }
  const sel = $("mn-count");
  for (let m = 1; m <= 24; m++) {
    const o = document.createElement("option");
    o.value = m; o.textContent = m;
    if (m === 3) o.selected = true;
    sel.appendChild(o);
  }

  function setStats(g) {
    $("mn-mult").textContent = g && g.picks ? "×" + g.multiplier.toFixed(2) : "—";
    $("mn-payout").textContent = g && g.picks ? fmt(g.payout) : "—";
    $("mn-next").textContent = g && g.next_multiplier ? "×" + g.next_multiplier.toFixed(2) : "—";
    $("mn-gems").textContent = g ? g.gems_left : "—";
    $("mn-safe").textContent = g && g.gems_left > 0 ? g.safe_pct.toFixed(1) + "%" : g ? "0%" : "—";
  }
  function updateProfit() {
    const el = $("mn-profit");
    el.textContent = (st.profit < 0 ? "-" : "") + fmt(Math.abs(st.profit));
    el.style.color = st.profit > 0 ? "#62d96b" : st.profit < 0 ? "var(--pk-danger)" : "";
  }
  function setControls(inGame) {
    for (const id of ["mn-bet", "mn-half", "mn-double", "mn-count"]) $(id).disabled = inGame;
    btn.textContent = inGame ? "Cash out" : "Bet";
    btn.classList.toggle("stop", inGame);
  }

  function enterGame(g) {
    st.active = true;
    st.bet = g.bet;
    tiles.forEach((t) => { t.className = "mn-tile"; t.textContent = ""; t.disabled = false; });
    for (const i of g.picked) {
      tiles[i].classList.add("gem");
      setIcon(tiles[i], "gem");
      tiles[i].disabled = true;
    }
    sel.value = g.mines;
    $("mn-bet").value = g.bet;
    setControls(true);
    setStats(g);
    hint.className = "hint";
    hint.textContent = g.picks ? "Cash out any time — or push your luck." : "Pick a tile…";
  }

  function endGame() {
    st.active = false;
    tiles.forEach((t) => { t.disabled = true; if (!t.firstChild) t.classList.add("idle"); });
    setControls(false);
  }

  function revealMines(minesAt, boomTile) {
    for (const i of minesAt) {
      setIcon(tiles[i], i === boomTile ? "boom" : "bomb");
      tiles[i].classList.add(i === boomTile ? "boom" : "mine-dim");
    }
  }

  async function start() {
    const bet = Math.floor(Number($("mn-bet").value));
    btn.disabled = true;
    try {
      const r = await fetch("/api/mines/start", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ bet, mines: Number(sel.value) }),
      });
      const d = await r.json();
      if (!r.ok) { msg.textContent = d.error; return; }
      msg.textContent = "";
      setBalance(d.balance);
      if (window.plinkoRefreshUI) plinkoRefreshUI();
      enterGame(d.game);
    } finally { btn.disabled = false; }
  }

  function finishWin(d) {
    if (window.sfx) sfx.cash();
    revealMines(d.mines_at);
    st.profit += d.payout - st.bet;
    updateProfit();
    setBalance(d.balance);
    if (window.plinkoRefreshUI) plinkoRefreshUI();
    hint.className = "hint win";
    hint.textContent = `Cashed out ×${d.multiplier.toFixed(2)} for ${fmt(d.payout)} credits!`;
    setStats(null);
    endGame();
  }

  async function pick(i) {
    if (!st.active || st.busy || tiles[i].disabled) return;
    st.busy = true;
    tiles[i].disabled = true;
    try {
      const r = await fetch("/api/mines/pick", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ tile: i }),
      });
      const d = await r.json();
      if (!r.ok) { tiles[i].disabled = false; return; }
      if (!d.safe) {
        if (window.sfx) sfx.boom();
        revealMines(d.mines_at, i);
        st.profit -= st.bet;
        updateProfit();
        setBalance(d.balance);
        if (window.plinkoRefreshUI) plinkoRefreshUI();
        hint.className = "hint loss";
        hint.textContent = "Boom — bet lost.";
        setStats(null);
        endGame();
        return;
      }
      if (d.cleared) {  // every safe tile found — auto cash out
        tiles[i].classList.add("gem");
        setIcon(tiles[i], "gem");
        finishWin(d);
        return;
      }
      if (window.sfx) sfx.gem();
      tiles[i].classList.add("gem");
      setIcon(tiles[i], "gem");
      setStats(d.game);
      hint.textContent = "Cash out any time — or push your luck.";
    } finally { st.busy = false; }
  }

  async function cashout() {
    btn.disabled = true;
    try {
      const r = await fetch("/api/mines/cashout", { method: "POST" });
      const d = await r.json();
      if (!r.ok) { hint.className = "hint loss"; hint.textContent = d.error; return; }
      finishWin(d);
    } finally { btn.disabled = false; }
  }

  btn.onclick = () => (st.active ? cashout() : start());
  $("mn-half").onclick = () => {
    $("mn-bet").value = Math.max(1, Math.floor(Number($("mn-bet").value || 0) / 2));
  };
  $("mn-double").onclick = () => doubleBet($("mn-bet"));

  // resume a live grid after a reload
  (async () => {
    const info = await (await fetch("/api/games/info")).json();
    applyCasinoState(info.casino_enabled);
    if (info.bet_min) { $("mn-bet").value = info.bet_min; $("mn-bet").min = info.bet_min; }
    if (!info.logged_in) {
      btn.disabled = true;
      hint.textContent = "Log in on the home page to play.";
      return;
    }
    try {
      const d = await (await fetch("/api/mines/state")).json();
      if (d.game) enterGame(d.game);
    } catch (e) {}
  })();
})();

/* ================= DRAGON TOWER ================= */
(() => {
  const tower = $("dt-tower"), btn = $("dt-btn"), msg = $("dt-msg"), hint = $("dt-hint");
  const sel = $("dt-difficulty");
  const fmt = (n) => Math.floor(n).toLocaleString();
  // Master tops out over ×250,000 — two decimals on that would never fit the
  // ladder column, and they say nothing at that size.
  const mx = (m) => "×" + (m >= 1000 ? Math.round(m).toLocaleString() : m.toFixed(2));
  const LABELS = { easy: "Easy", medium: "Medium", hard: "Hard",
                   expert: "Expert", master: "Master" };
  const st = { active: false, bet: 0, profit: 0, busy: false };
  let INFO = null;                       // /api/games/info → dragon
  let rows = [];                         // [{ el, multEl, tiles: [button] }]

  function conf() { return INFO.difficulties[sel.value]; }

  /* The tower is rebuilt whenever the difficulty changes — each difficulty
     has its own number of tiles per row. */
  function buildTower() {
    const d = conf();
    tower.textContent = "";
    rows = [];
    for (let r = 0; r < INFO.rows; r++) {
      const row = document.createElement("div");
      row.className = "dt-row";
      const tiles = document.createElement("div");
      tiles.className = "dt-tiles";
      tiles.style.gridTemplateColumns = `repeat(${d.tiles}, 1fr)`;
      const buttons = [];
      for (let i = 0; i < d.tiles; i++) {
        const t = document.createElement("button");
        t.className = "dt-tile";
        t.disabled = true;
        t.onclick = () => pick(r, i);
        tiles.appendChild(t);
        buttons.push(t);
      }
      const mult = document.createElement("div");
      mult.className = "dt-mult";
      mult.textContent = mx(d.multipliers[r]);
      row.append(tiles, mult);
      tower.appendChild(row);
      rows.push({ el: row, multEl: mult, tiles: buttons });
    }
  }

  function setStats(g) {
    $("dt-mult").textContent = g && g.level ? mx(g.multiplier) : "—";
    $("dt-payout").textContent = g && g.level ? fmt(g.payout) : "—";
    $("dt-next").textContent = g && g.next_multiplier ? mx(g.next_multiplier) : "—";
    $("dt-level").textContent = g ? `${g.level}/${g.rows}` : "—";
    $("dt-safe").textContent = g ? g.safe_pct.toFixed(1) + "%"
                                 : (INFO ? (conf().eggs / conf().tiles * 100).toFixed(1) + "%" : "—");
  }
  function updateProfit() {
    const el = $("dt-profit");
    el.textContent = (st.profit < 0 ? "-" : "") + fmt(Math.abs(st.profit));
    el.style.color = st.profit > 0 ? "#62d96b" : st.profit < 0 ? "var(--pk-danger)" : "";
  }
  function setControls(inGame) {
    for (const id of ["dt-bet", "dt-half", "dt-double", "dt-difficulty"]) $(id).disabled = inGame;
    btn.textContent = inGame ? "Cash out" : "Bet";
    btn.classList.toggle("stop", inGame);
  }

  /* Mid-game: lock a row the player just climbed, showing only the egg they
     found. What the other tiles were held stays a secret until the run ends —
     showing them would give away the rows' contents a pick at a time. */
  function markPick(r, tile) {
    const row = rows[r];
    row.el.classList.remove("here");
    row.el.classList.add("cleared");
    row.tiles.forEach((t, i) => {
      t.disabled = true;
      t.textContent = i === tile ? "🥚" : "";
      t.className = "dt-tile" + (i === tile ? " egg" : "");
    });
  }

  /* Game over: paint one row in full. `eggs` are that row's safe tiles,
     `picked` the tile the player chose there (if any), `boom` the one that
     ended the run. */
  function paintRow(r, eggs, picked, boom) {
    const row = rows[r];
    row.el.classList.remove("here");
    row.el.classList.toggle("cleared", picked != null && boom == null);
    row.tiles.forEach((t, i) => {
      const isEgg = eggs.includes(i);
      t.disabled = true;
      t.textContent = isEgg ? "🥚" : "❌";
      t.className = "dt-tile " + (i === boom ? "boom" : i === picked ? "egg"
                                  : isEgg ? "egg-dim" : "skull");
    });
  }

  function focusRow(r) {
    rows.forEach((row, i) => {
      row.el.classList.toggle("here", i === r);
      if (i === r) row.tiles.forEach((t) => { t.disabled = false; });
    });
  }

  /* The run is over, so the whole tower stops being a secret: every row is
     laid bare, with the picks that got the player this far still marked. */
  function revealTower(eggsAt, picked, boomRow, boomTile) {
    for (let r = 0; r < rows.length; r++) {
      paintRow(r, eggsAt[r], picked[r] != null ? picked[r] : null,
               r === boomRow ? boomTile : null);
    }
  }

  function enterGame(g) {
    st.active = true;
    st.bet = g.bet;
    sel.value = g.difficulty;
    buildTower();
    $("dt-bet").value = g.bet;
    g.picked.forEach((tile, r) => markPick(r, tile));
    focusRow(g.level);
    setControls(true);
    setStats(g);
    hint.className = "hint";
    hint.textContent = g.level ? "Cash out any time — or climb higher."
                               : "Pick a tile on the bottom row…";
  }

  function endGame() {
    st.active = false;
    rows.forEach((row) => {
      row.el.classList.remove("here");
      row.tiles.forEach((t) => { t.disabled = true; });
    });
    setControls(false);
  }

  async function start() {
    const bet = Math.floor(Number($("dt-bet").value));
    btn.disabled = true;
    try {
      const r = await fetch("/api/dragon/start", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ bet, difficulty: sel.value }),
      });
      const d = await r.json();
      if (!r.ok) { msg.textContent = d.error; return; }
      msg.textContent = "";
      setBalance(d.balance);
      if (window.plinkoRefreshUI) plinkoRefreshUI();
      enterGame(d.game);
    } finally { btn.disabled = false; }
  }

  function finishWin(d) {
    if (window.sfx) (d.multiplier >= 10 ? sfx.bigwin() : sfx.cash());
    revealTower(d.eggs_at, d.picked);
    st.profit += d.payout - st.bet;
    updateProfit();
    setBalance(d.balance);
    if (window.plinkoRefreshUI) plinkoRefreshUI();
    hint.className = "hint win";
    hint.textContent = `Cashed out ${mx(d.multiplier)} for ${fmt(d.payout)} credits!`;
    setStats(null);
    endGame();
  }

  async function pick(row, i) {
    if (!st.active || st.busy || rows[row].tiles[i].disabled) return;
    st.busy = true;
    rows[row].tiles.forEach((t) => { t.disabled = true; });
    try {
      const r = await fetch("/api/dragon/pick", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ tile: i }),
      });
      const d = await r.json();
      if (!r.ok) { focusRow(row); return; }
      if (!d.safe) {
        if (window.sfx) sfx.boom();
        revealTower(d.eggs_at, d.picked, row, i);
        st.profit -= st.bet;
        updateProfit();
        setBalance(d.balance);
        if (window.plinkoRefreshUI) plinkoRefreshUI();
        hint.className = "hint loss";
        hint.textContent = "The dragon got you ❌ — bet lost.";
        setStats(null);
        endGame();
        return;
      }
      markPick(row, i);
      if (d.cleared) {  // all nine rows climbed — auto cash out
        finishWin(d);
        hint.textContent = `Tower cleared! ${mx(d.multiplier)} for ${fmt(d.payout)} credits!`;
        return;
      }
      if (window.sfx) sfx.gem();
      focusRow(d.game.level);
      setStats(d.game);
      hint.textContent = "Cash out any time — or climb higher.";
    } finally { st.busy = false; }
  }

  async function cashout() {
    btn.disabled = true;
    try {
      const r = await fetch("/api/dragon/cashout", { method: "POST" });
      const d = await r.json();
      if (!r.ok) { hint.className = "hint loss"; hint.textContent = d.error; return; }
      finishWin(d);
    } finally { btn.disabled = false; }
  }

  btn.onclick = () => (st.active ? cashout() : start());
  sel.onchange = () => { buildTower(); setStats(null); };
  $("dt-half").onclick = () => {
    $("dt-bet").value = Math.max(1, Math.floor(Number($("dt-bet").value || 0) / 2));
  };
  $("dt-double").onclick = () => doubleBet($("dt-bet"));

  // build from the server's tables, then resume a live tower after a reload
  (async () => {
    const info = await (await fetch("/api/games/info")).json();
    applyCasinoState(info.casino_enabled);
    INFO = info.dragon;
    for (const name of Object.keys(INFO.difficulties)) {
      const d = INFO.difficulties[name];
      const o = document.createElement("option");
      o.value = name;
      o.textContent = `${LABELS[name] || name} — ${d.eggs} in ${d.tiles}`;
      if (name === "medium") o.selected = true;
      sel.appendChild(o);
    }
    buildTower();
    setStats(null);
    if (info.bet_min) { $("dt-bet").value = info.bet_min; $("dt-bet").min = info.bet_min; }
    if (!info.logged_in) {
      btn.disabled = true;
      hint.textContent = "Log in on the home page to play.";
      return;
    }
    try {
      const d = await (await fetch("/api/dragon/state")).json();
      if (d.game) enterGame(d.game);
    } catch (e) {}
  })();
})();

/* ================= CASES ================= */
// The server decides the tier before the reel moves; everything here is the
// animation of a result that already happened. The filler cases either side of
// it are drawn from the same odds table, so a spin never shows a mix of cases
// the difficulty couldn't produce.
(() => {
  const stage = $("cs-stage"), reel = $("cs-reel"), btn = $("cs-btn");
  const msg = $("cs-msg"), hint = $("cs-hint"), sel = $("cs-difficulty");
  const LABELS = { easy: "Easy", medium: "Medium", hard: "Hard",
                   expert: "Expert", master: "Master" };
  const SPIN_MS = 2600;   // matches --cs-ms on the reel
  const RUN = 42;         // cases scrolled past before the winning one
  const TRAIL = 6;        // cases left beyond it, so the strip never looks like it ends
  const MAX_ITEMS = 160;  // trimmed from the front past this, invisibly
  const st = { busy: false, idx: 0, jitter: 0,
               opened: 0, wins: 0, best: 0, bestWin: 0, profit: 0, history: [] };
  let INFO = null, COLOURS = [];

  const fmt = (n) => Math.round(n).toLocaleString();
  const mx = (m) => "×" + (m >= 1000 ? Math.round(m).toLocaleString() : m.toFixed(2));
  const conf = () => INFO.difficulties[sel.value];
  const topPrize = (d) => d.multipliers[d.multipliers.length - 1];

  /* One case width drives all the reel maths, so it lives in JS and is pushed
     into CSS — not the other way round. */
  function caseSize() { return window.innerWidth <= 720 ? 62 : 92; }
  function step() {
    const size = caseSize(), gap = size >= 90 ? 14 : 10;
    reel.style.setProperty("--cs-size", size + "px");
    reel.style.setProperty("--cs-gap", gap + "px");
    return size + gap;
  }

  /* Slide the strip so the case at st.idx sits under the marker. */
  function place(animate) {
    const x = stage.clientWidth / 2 - caseSize() / 2 - st.idx * step() - st.jitter;
    reel.classList.toggle("spin", !!animate);
    reel.style.transform = `translateX(${x}px)`;
    if (!animate) void reel.offsetWidth;  // flush, so the next spin still transitions
  }

  function randomTier() {
    const w = conf().weights;
    let total = 0;
    for (const x of w) total += x;
    let r = Math.random() * total;
    for (let i = 0; i < w.length; i++) {
      r -= w[i];
      if (r < 0) return i;
    }
    return 0;
  }
  function makeCase(tier) {
    const el = document.createElement("div");
    el.className = "cs-case";
    el.style.setProperty("--c", COLOURS[tier]);
    el.appendChild(document.createElement("b"));  // the latch under the seam
    return el;
  }
  function fill(n) {
    for (let i = 0; i < n; i++) reel.appendChild(makeCase(randomTier()));
  }
  function resetReel() {
    reel.textContent = "";
    st.idx = 8;
    st.jitter = 0;
    fill(20);
    place(false);
  }

  /* Drop the cases already scrolled past — the strip would otherwise grow by
     50 nodes a spin. Nothing moves on screen: the offset is corrected in the
     same frame the nodes leave. */
  function trim() {
    const drop = st.idx - 6;
    if (reel.children.length <= MAX_ITEMS || drop <= 0) return;
    for (let i = 0; i < drop; i++) reel.removeChild(reel.firstChild);
    st.idx -= drop;
    place(false);
  }

  function spinTo(tier) {
    const prev = reel.querySelector(".cs-case.win");
    if (prev) {
      prev.classList.remove("win");
      const tag = prev.querySelector(".cs-tag");
      if (tag) tag.remove();
    }
    trim();
    const start = reel.children.length;
    fill(RUN);
    reel.appendChild(makeCase(tier));
    fill(TRAIL);
    st.idx = start + RUN;                              // the case we just added
    st.jitter = (Math.random() * 2 - 1) * caseSize() * 0.26;  // stops dead-centre landings
    place(true);
    return new Promise((res) => setTimeout(res, SPIN_MS + 80));
  }

  function paintStats() {
    $("cs-opened").textContent = st.opened;
    $("cs-wins").textContent = st.wins;
    $("cs-best").textContent = st.opened ? mx(st.best) : "—";
    $("cs-bestwin").textContent = fmt(st.bestWin);
    $("cs-top").textContent = INFO ? mx(topPrize(conf())) : "—";
    const p = $("cs-profit");
    p.textContent = (st.profit < 0 ? "-" : "") + fmt(Math.abs(st.profit));
    p.style.color = st.profit > 0 ? "#62d96b" : st.profit < 0 ? "var(--pk-danger)" : "";
  }

  function paintRecent() {
    const box = $("cs-recent");
    box.textContent = "";
    for (const h of st.history.slice(-5)) {
      const pill = document.createElement("div");
      pill.className = "cs-pill";
      pill.style.background = COLOURS[h.tier];
      pill.textContent = mx(h.multiplier);
      box.appendChild(pill);
    }
  }

  const pct = (v) => (v >= 1 ? v.toFixed(2) : v.toFixed(4)) + "%";

  function paintTables() {
    const c = conf();
    const legend = $("cs-legend");
    legend.textContent = "";
    c.multipliers.forEach((m, i) => {
      const s = document.createElement("div");
      s.className = "cs-swatch";
      s.style.setProperty("--c", COLOURS[i]);
      s.textContent = m >= 1000 ? Math.round(m / 1000) + "k×" : m + "×";
      s.title = `${mx(m)} — ${pct(c.chances[i])} chance`;
      legend.appendChild(s);
    });
    const rows = c.multipliers.map((m, i) =>
      `<tr><td><i style="--c:${COLOURS[i]}"></i></td><td>${mx(m)}</td>`
      + `<td>${pct(c.chances[i])}</td><td>1 in ${fmt(100 / c.chances[i])}</td></tr>`).join("");
    $("cs-odds").innerHTML =
      '<table class="cs-odds"><thead><tr><th>Case</th><th>Pays</th><th>Chance</th>'
      + `<th>Odds</th></tr></thead><tbody>${rows}</tbody></table>`
      + `<div class="hint" style="margin-top:8px;">Every difficulty pays back ${c.rtp}% of what's `
      + "staked on it over the long run — the difficulty only changes how those wins are spread.</div>";
  }

  function setControls(playing) {
    for (const id of ["cs-bet", "cs-half", "cs-double", "cs-difficulty"]) $(id).disabled = playing;
  }

  function finish(d, bet) {
    const winEl = reel.children[st.idx];
    if (winEl) {
      winEl.classList.add("win");
      const tag = document.createElement("div");
      tag.className = "cs-tag";
      tag.textContent = mx(d.multiplier);
      winEl.appendChild(tag);
    }
    const won = d.payout > bet;
    st.opened++;
    st.profit += d.payout - bet;
    if (won) st.wins++;
    st.best = Math.max(st.best, d.multiplier);
    st.bestWin = Math.max(st.bestWin, d.payout);
    st.history.push({ tier: d.tier, multiplier: d.multiplier });
    paintStats();
    paintRecent();
    setBalance(d.balance);
    if (window.plinkoRefreshUI) plinkoRefreshUI();
    if (window.sfx) (d.multiplier >= 10 ? sfx.bigwin() : won ? sfx.win() : sfx.lose());
    hint.className = "hint " + (won ? "win" : "loss");
    hint.textContent = won
      ? `${mx(d.multiplier)} — ${fmt(d.payout)} credits! 🎉`
      : d.payout > 0
        ? `${mx(d.multiplier)} — only ${fmt(d.payout)} of your ${fmt(bet)} back.`
        : `${mx(d.multiplier)} — nothing back this time 💀`;
  }

  async function play() {
    if (st.busy) return;
    const bet = Math.floor(Number($("cs-bet").value));
    st.busy = true;
    btn.disabled = true;
    setControls(true);
    try {
      const r = await fetch("/api/cases", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ bet, difficulty: sel.value }),
      });
      const d = await r.json();
      if (!r.ok) { msg.textContent = d.error; return; }
      msg.textContent = "";
      setBalance(BALANCE - bet);   // the payout lands when the reel stops
      if (window.sfx) sfx.click();
      hint.className = "hint";
      hint.textContent = "Opening…";
      await spinTo(d.tier);
      finish(d, bet);
    } catch (e) {
    } finally { st.busy = false; btn.disabled = false; setControls(false); }
  }

  btn.onclick = play;
  sel.onchange = () => { resetReel(); paintTables(); paintStats(); };
  $("cs-half").onclick = () => {
    $("cs-bet").value = Math.max(1, Math.floor(Number($("cs-bet").value || 0) / 2));
  };
  $("cs-double").onclick = () => doubleBet($("cs-bet"));
  // The panel is hidden until its tab is opened, so the stage has no width to
  // centre against at load — re-place whenever it gets one (and on resize).
  new ResizeObserver(() => { if (!st.busy) place(false); }).observe(stage);

  (async () => {
    const info = await (await fetch("/api/games/info")).json();
    applyCasinoState(info.casino_enabled);
    INFO = info.cases;
    COLOURS = INFO.colours;
    for (const name of Object.keys(INFO.difficulties)) {
      const o = document.createElement("option");
      o.value = name;
      o.textContent = `${LABELS[name] || name} — up to ${mx(topPrize(INFO.difficulties[name]))}`;
      if (name === "medium") o.selected = true;
      sel.appendChild(o);
    }
    resetReel();
    paintTables();
    paintStats();
    if (info.bet_min) { $("cs-bet").value = info.bet_min; $("cs-bet").min = info.bet_min; }
    if (!info.logged_in) {
      btn.disabled = true;
      hint.textContent = "Log in on the home page to play.";
    }
  })();
})();

/* ================= KENO ================= */
// The whole round settles in one request: the server draws its ten numbers and
// pays before it answers, so everything below is the reveal of a result that
// already happened. The payout row is rebuilt from the server's own table
// every time the selection or the difficulty changes — what's printed on the
// row is exactly what the server pays out of.
(() => {
  const grid = $("kn-grid"), btn = $("kn-btn"), msg = $("kn-msg"), hint = $("kn-hint");
  const sel = $("kn-difficulty");
  const LABELS = { classic: "Classic", low: "Low", medium: "Medium", high: "High" };
  const BALL_MS = 130;   // between drawn numbers landing on the board
  const AUTO_GAP_MS = 380;   // breather between autobet rounds, on top of the draw
  const st = { busy: false, picks: new Set(), drawn: false, locked: false,
               mode: "manual", autoOn: false, autoLeft: null,
               bestWin: 0, profit: 0 };
  const IDLE_HINT = hint.textContent;
  let INFO = null, MAX = 10;

  const fmt = (n) => Math.round(n).toLocaleString();
  const mx = (m) => "×" + (m >= 1000 ? Math.round(m).toLocaleString() : m.toFixed(2));
  /* The payout row is eleven columns wide at ten picks, so it gets the short
     form: trailing zeros dropped, thousands as "k". Everywhere with room —
     the stats, the odds table — keeps the full figure. */
  const mxShort = (m) => m >= 1000 ? "×" + +(m / 1000).toFixed(1) + "k" : "×" + +m.toFixed(2);
  // the row for the current selection; null while the board is empty
  const conf = () => (st.picks.size ? INFO.difficulties[sel.value][st.picks.size] : null);

  const cells = [];
  for (let n = 1; n <= 40; n++) {
    const c = document.createElement("button");
    c.className = "kn-cell";
    c.textContent = n;
    c.onclick = () => toggle(n);
    grid.appendChild(c);
    cells.push(c);
  }
  const cell = (n) => cells[n - 1];

  /* Wipe the last round's draw off the board. Called before anything that
     changes the selection, so a stale draw is never read as this one. */
  function clearDraw() {
    if (!st.drawn) return;
    st.drawn = false;
    for (const c of cells) c.classList.remove("drawn", "pop");
    $("kn-hitcount").textContent = "—";
    $("kn-mult").textContent = "—";
    if (!st.locked) { hint.className = "hint"; hint.textContent = IDLE_HINT; }
    paintPays();
  }

  function toggle(n) {
    if (st.busy || st.autoOn) return;
    clearDraw();
    const c = cell(n);
    if (st.picks.has(n)) {
      st.picks.delete(n);
      c.classList.remove("sel");
    } else {
      if (st.picks.size >= MAX) {
        msg.textContent = `${MAX} numbers is the most you can pick`;
        return;
      }
      st.picks.add(n);
      c.classList.add("sel");
    }
    msg.textContent = "";
    afterPickChange();
  }

  function setPicks(numbers) {
    clearDraw();
    st.picks = new Set(numbers);
    for (let n = 1; n <= 40; n++) cell(n).classList.toggle("sel", st.picks.has(n));
    msg.textContent = "";
    afterPickChange();
  }

  /* Fill the board out to the maximum, keeping what's already selected —
     and reroll the lot once it's full, so the button always does something. */
  function randomPick() {
    if (st.busy || st.autoOn) return;
    const keep = st.picks.size >= MAX ? [] : [...st.picks];
    const pool = [];
    for (let n = 1; n <= 40; n++) if (!keep.includes(n)) pool.push(n);
    for (let i = pool.length - 1; i > 0; i--) {           // Fisher-Yates
      const j = Math.floor(Math.random() * (i + 1));
      [pool[i], pool[j]] = [pool[j], pool[i]];
    }
    setPicks(keep.concat(pool.slice(0, MAX - keep.length)));
    if (window.sfx) sfx.click();
  }

  function afterPickChange() {
    paintPays();
    paintOdds();
    paintStats();
    paintDifficulties();
    paintButton();
  }

  /* The payout row: one column per possible hit count, 0 on the left. Its
     length is the selection size + 1, so picking more numbers visibly
     stretches the row and pushes the top prize further out. */
  function paintPays(landed) {
    const pays = $("kn-pays"), hits = $("kn-hits");
    pays.textContent = "";
    hits.textContent = "";
    const c = conf();
    if (!c) {
      pays.style.display = hits.style.display = "none";
      return;
    }
    pays.style.display = hits.style.display = "grid";
    const cols = `repeat(${c.multipliers.length}, minmax(0, 1fr))`;
    pays.style.gridTemplateColumns = hits.style.gridTemplateColumns = cols;
    c.multipliers.forEach((m, k) => {
      const dead = m ? "" : " dead";
      const here = k === landed ? " landed" : "";
      const p = document.createElement("div");
      p.className = "kn-pay" + dead + here;
      p.textContent = mxShort(m);
      p.title = mx(m);
      pays.appendChild(p);

      const h = document.createElement("div");
      h.className = "kn-hit" + dead + here;
      h.innerHTML = `${k}× <i></i>`;
      h.title = `${k} of your ${st.picks.size} numbers`;
      hits.appendChild(h);
    });
  }

  // below a ten-thousandth of a percent the figure stops meaning anything —
  // the "1 in N" column next to it is what's readable at those odds
  const pct = (v) => v >= 1 ? v.toFixed(2) + "%"
                   : v >= 0.01 ? v.toFixed(3) + "%"
                   : v >= 0.0001 ? v.toFixed(5) + "%"
                   : "< 0.0001%";

  function paintOdds() {
    const box = $("kn-odds"), c = conf();
    if (!c) {
      box.innerHTML = '<div class="hint">Select some numbers to see what they pay.</div>';
      return;
    }
    const rows = c.multipliers.map((m, k) =>
      `<tr${m ? "" : ' class="dead"'}><td>${k} of ${st.picks.size}</td><td>${mx(m)}</td>`
      + `<td>${pct(c.chances[k])}</td><td>1 in ${fmt(100 / c.chances[k])}</td></tr>`).join("");
    box.innerHTML =
      '<table class="kn-odds"><thead><tr><th>Hits</th><th>Pays</th><th>Chance</th>'
      + `<th>Odds</th></tr></thead><tbody>${rows}</tbody></table>`
      + `<div class="hint" style="margin-top:8px;">Ten of the forty numbers are drawn every `
      + `round. This board pays back ${c.rtp}% of what's staked on it over the long run — `
      + "picking more numbers raises the top prize, not the house's cut.</div>";
  }

  /* Each difficulty's option carries the top prize for the current selection,
     so the trade — a fatter jackpot for deader near misses — is visible
     before you switch. */
  function paintDifficulties() {
    for (const o of sel.options) {
      const d = INFO.difficulties[o.value][st.picks.size];
      o.textContent = LABELS[o.value] + (d ? ` — up to ${mx(d.multipliers[d.multipliers.length - 1])}` : "");
    }
  }

  function paintStats() {
    const c = conf();
    $("kn-picks").textContent = `${st.picks.size} / ${MAX}`;
    $("kn-top").textContent = c ? mx(c.multipliers[c.multipliers.length - 1]) : "—";
    $("kn-bestwin").textContent = fmt(st.bestWin);
    const p = $("kn-profit");
    p.textContent = (st.profit < 0 ? "-" : "") + fmt(Math.abs(st.profit));
    p.style.color = st.profit > 0 ? "#62d96b" : st.profit < 0 ? "var(--pk-danger)" : "";
  }

  /* One place decides what the primary button says and whether it's live.
     In manual it plays a round; in auto it starts and stops the run, and it
     has to stay clickable mid-round or a run could never be stopped. */
  function paintButton() {
    const auto = st.mode === "auto";
    btn.textContent = !auto ? "Play" : st.autoOn ? "Stop Autobet" : "Start Autobet";
    btn.classList.toggle("stop", st.autoOn);
    btn.disabled = st.autoOn ? false : st.locked || st.busy || st.picks.size === 0;
  }

  /* Everything that would change the wager mid-round, or mid-run: locked
     while a draw is landing, and for the whole run once autobet starts. */
  function lockUI() {
    const busy = st.busy || st.autoOn;
    for (const id of ["kn-bet", "kn-half", "kn-double", "kn-difficulty",
                      "kn-random", "kn-clear"]) $(id).disabled = busy;
    for (const c of cells) c.disabled = busy;
    for (const id of ["kn-manual", "kn-auto", "kn-auto-count"]) $(id).disabled = st.autoOn;
    paintButton();
  }

  /* Land the ten drawn numbers one at a time. Nothing here decides anything —
     the order is the order the server drew them in. */
  function reveal(draw) {
    st.drawn = true;
    return new Promise((done) => {
      draw.forEach((n, i) => setTimeout(() => {
        const c = cell(n);
        c.classList.add("drawn", "pop");
        if (window.sfx) (c.classList.contains("sel") ? sfx.gem() : sfx.click());
        if (i === draw.length - 1) setTimeout(done, 260);
      }, i * BALL_MS));
    });
  }

  function finish(d, bet) {
    const hits = d.hits.length;
    st.profit += d.payout - bet;
    st.bestWin = Math.max(st.bestWin, d.payout);
    $("kn-hitcount").textContent = `${hits} / ${d.picks.length}`;
    $("kn-mult").textContent = mx(d.multiplier);
    paintPays(hits);
    paintStats();
    setBalance(d.balance);
    if (window.plinkoRefreshUI) plinkoRefreshUI();
    if (window.sfx) (d.multiplier >= 10 ? sfx.bigwin() : d.payout > bet ? sfx.win() : sfx.lose());
    const of = `${hits} of ${d.picks.length}`;
    hint.className = "hint " + (d.payout > bet ? "win" : "loss");
    hint.textContent = d.payout > bet
      ? `${of} — ${mx(d.multiplier)} for ${fmt(d.payout)} credits! 🎉`
      : d.payout > 0
        ? `${of} — ${mx(d.multiplier)}, only ${fmt(d.payout)} of your ${fmt(bet)} back.`
        : `${of} — nothing back this time 💀`;
  }

  /* Returns true only if a round actually settled — autobet stops on
     anything else, so a rejected bet can't spin the loop against the server. */
  async function play() {
    if (st.busy || !st.picks.size) return false;
    const bet = Math.floor(Number($("kn-bet").value));
    const picks = [...st.picks];
    st.busy = true;
    lockUI();
    try {
      const r = await fetch("/api/keno", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ bet, picks, difficulty: sel.value }),
      });
      const d = await r.json();
      if (!r.ok) { msg.textContent = d.error; return false; }
      msg.textContent = "";
      setBalance(BALANCE - bet);   // the payout lands when the draw finishes
      clearDraw();
      hint.className = "hint";
      hint.textContent = "Drawing…";
      await reveal(d.draw);
      finish(d, bet);
      return true;
    } catch (e) {
      return false;
    } finally {
      st.busy = false;
      lockUI();
    }
  }

  /* ---- autobet ---- */
  // Rounds are chained rather than put on an interval: a keno round takes as
  // long as its draw takes to land, and a timer would just stack requests
  // behind a reveal that hasn't finished.
  const pause = (ms) => new Promise((r) => setTimeout(r, ms));

  function stopAuto(reason) {
    if (!st.autoOn) return;
    st.autoOn = false;
    st.autoLeft = null;
    lockUI();
    if (reason) { hint.className = "hint"; hint.textContent = reason; }
  }

  async function runAuto() {
    while (st.autoOn) {
      const bet = Math.floor(Number($("kn-bet").value));
      if (!(bet >= 1) || bet > BALANCE) {
        stopAuto("Autobet stopped — not enough credits for the next bet.");
        return;
      }
      if (!(await play()) || !st.autoOn) { stopAuto(); return; }
      if (st.autoLeft !== null) {
        $("kn-auto-count").value = --st.autoLeft;
        if (st.autoLeft <= 0) { stopAuto(); return; }
      }
      await pause(AUTO_GAP_MS);
    }
  }

  function startAuto() {
    const n = parseInt($("kn-auto-count").value || "0", 10);
    const bad = !isFinite(n) || n < 0;
    $("kn-auto-count").classList.toggle("bad", bad);
    $("kn-auto-msg").textContent = bad ? "This must be greater than or equal to 0." : "";
    if (bad || !st.picks.size) return;
    st.autoOn = true;
    st.autoLeft = n === 0 ? null : n;   // 0 means "until I say stop"
    lockUI();
    runAuto();
  }

  function setMode(mode) {
    st.mode = mode;
    $("kn-manual").classList.toggle("active", mode === "manual");
    $("kn-auto").classList.toggle("active", mode === "auto");
    $("kn-auto-block").classList.toggle("pk-hidden", mode !== "auto");
    paintButton();
  }

  btn.onclick = () => {
    if (st.mode === "manual") play();
    else if (st.autoOn) stopAuto();
    else startAuto();
  };
  $("kn-manual").onclick = () => setMode("manual");
  $("kn-auto").onclick = () => setMode("auto");
  $("kn-random").onclick = randomPick;
  $("kn-clear").onclick = () => { if (!st.busy && !st.autoOn) setPicks([]); };
  sel.onchange = () => { clearDraw(); paintPays(); paintOdds(); paintStats(); paintDifficulties(); };
  $("kn-half").onclick = () => {
    $("kn-bet").value = Math.max(1, Math.floor(Number($("kn-bet").value || 0) / 2));
  };
  $("kn-double").onclick = () => doubleBet($("kn-bet"));

  (async () => {
    const info = await (await fetch("/api/games/info")).json();
    applyCasinoState(info.casino_enabled);
    INFO = info.keno;
    MAX = INFO.max_picks;
    for (const name of Object.keys(INFO.difficulties)) {
      const o = document.createElement("option");
      o.value = name;
      o.textContent = LABELS[name] || name;
      if (name === "classic") o.selected = true;
      sel.appendChild(o);
    }
    afterPickChange();
    // autobet won't start a round it can't afford, so the balance has to be
    // known here rather than left to whichever game's init runs first
    setBalance(info.balance);
    if (info.bet_min) { $("kn-bet").value = info.bet_min; $("kn-bet").min = info.bet_min; }
    if (!info.logged_in) {
      st.locked = true;
      paintButton();
      hint.textContent = "Log in on the home page to play.";
    }
  })();
})();

/* ================= COIN FLIP ================= */
(() => {
  const coin = $("cf-coin"), btn = $("cf-btn"), msg = $("cf-msg"), hint = $("cf-hint");
  // rot only ever grows, so the coin always spins forward; the server's
  // result decides which face it lands on (heads = 0°, tails = 180°).

               flips: 0, wins: 0, streak: 0, profit: 0, history: [] };
  const fmt = (n) => Math.round(n).toLocaleString();

    box.innerHTML = "";
    for (const h of st.history.slice(-10)) {

    st.busy = true;
    btn.disabled = true;
    try {
      const r = await fetch("/api/coinflip", {

/* ================= WHEEL ================= */
(() => {
  const canvas = $("wh-canvas"), ctx = canvas.getContext("2d");
  const btn = $("wh-btn"), msg = $("wh-msg"), hint = $("wh-hint");
  // Rust bandit-camp colors, one per number
  const COLORS = { 1: "#fbd065", 3: "#3fa76b", 5: "#3e7bd0", 10: "#b558d6", 20: "#e0592b" };
  const TEXT = { 1: "#5c3d0c", 3: "#eaf6ee", 5: "#eef3fa", 10: "#f6ecfa", 20: "#fbeae2" };
  const SPIN_MS = 4200;  // matches the #wh-canvas.spin transition
  // rot only ever grows, so the wheel always spins forward; the server's
  // result index decides which segment stops under the pointer.
  const st = { layout: null, payouts: null, stakes: {}, rot: 0, busy: false,
               spins: 0, wins: 0, best: 0, profit: 0, history: [] };
  const fmt = (n) => Math.round(n).toLocaleString();
  const totalStaked = () => Object.values(st.stakes).reduce((a, b) => a + b, 0);

  function drawWheel() {
    if (!st.layout) return;
    const n = st.layout.length, seg = (2 * Math.PI) / n;
    const cx = canvas.width / 2, cy = canvas.height / 2, r = cx - 4;
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    for (let i = 0; i < n; i++) {
      // segment i is centered at the pointer (top) when the canvas rotation is -i*seg
      const mid = -Math.PI / 2 + i * seg;
      ctx.beginPath();
      ctx.moveTo(cx, cy);
      ctx.arc(cx, cy, r, mid - seg / 2, mid + seg / 2);
      ctx.closePath();
      ctx.fillStyle = COLORS[st.layout[i]];
      ctx.fill();
      ctx.strokeStyle = "rgba(244, 245, 247, .55)";
      ctx.lineWidth = 1.5;
      ctx.stroke();
      ctx.save();  // number near the rim, rotated to read along the radius
      ctx.translate(cx + Math.cos(mid) * r * 0.84, cy + Math.sin(mid) * r * 0.84);
      ctx.rotate(mid + Math.PI / 2);
      ctx.font = "800 24px system-ui, sans-serif";
      ctx.textAlign = "center";
      ctx.textBaseline = "middle";
      ctx.fillStyle = TEXT[st.layout[i]];
      ctx.fillText(st.layout[i], 0, 0);
      ctx.restore();
    }
    ctx.beginPath();  // outer rim
    ctx.arc(cx, cy, r, 0, 2 * Math.PI);
    ctx.strokeStyle = "#e8e4d8";
    ctx.lineWidth = 3;
    ctx.stroke();
    ctx.beginPath();  // weathered hub with cross braces, like the real thing
    ctx.arc(cx, cy, r * 0.48, 0, 2 * Math.PI);
    ctx.fillStyle = "#cfc4a5";
    ctx.fill();
    ctx.strokeStyle = "#8a7f63";
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.strokeStyle = "rgba(90, 80, 60, .55)";
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.moveTo(cx - r * 0.48, cy); ctx.lineTo(cx + r * 0.48, cy);
    ctx.moveTo(cx, cy - r * 0.48); ctx.lineTo(cx, cy + r * 0.48);
    ctx.stroke();
    ctx.beginPath();
    ctx.arc(cx, cy, 6, 0, 2 * Math.PI);
    ctx.fillStyle = "#3a352a";
    ctx.fill();
  }

  function place(n) {
    if (st.busy) return;
    const chip = Math.floor(Number($("wh-bet").value) || 0);
    if (chip <= 0) return;
    st.stakes[n] = (st.stakes[n] || 0) + chip;
    paintPicks();
  }
  function clearStake(n) {
    if (st.busy) return;
    delete st.stakes[n];
    paintPicks();
  }
  function clearAll() {
    if (st.busy) return;
    st.stakes = {};
    paintPicks();
  }
  function buildPicks() {
    const box = $("wh-picks");
    box.innerHTML = "";
    for (const [nStr, pays] of Object.entries(st.payouts)) {
      const n = Number(nStr);
      const b = document.createElement("button");
      b.className = "wh-pick";
      b.dataset.n = n;
      b.style.background = COLORS[n];
      b.style.color = TEXT[n];
      b.title = "Left click: add chip · Right click: clear";
      b.innerHTML = "<span class='wh-stake' style='display:none'></span>" +
                    n + "<small>pays " + pays + "×</small>";
      b.onclick = () => place(n);
      b.oncontextmenu = (e) => { e.preventDefault(); clearStake(n); };
      box.appendChild(b);
    }
    paintPicks();
  }
  function paintPicks() {
    [...$("wh-picks").children].forEach((b) => {
      const n = Number(b.dataset.n), amt = st.stakes[n] || 0;
      const badge = b.querySelector(".wh-stake");
      b.classList.toggle("staked", amt > 0);
      badge.style.display = amt > 0 ? "" : "none";
      badge.textContent = amt > 0 ? fmt(amt) : "";
    });
    const total = totalStaked();
    $("wh-total").textContent = fmt(total);
    $("wh-clear").disabled = st.busy || total <= 0;
  }
  const setPicks = (on) => {
    [...$("wh-picks").children].forEach((b) => (b.disabled = !on));
    $("wh-clear").disabled = !on || totalStaked() <= 0;
  };

  function paintStats() {
    $("wh-spins").textContent = st.spins;
    $("wh-wins").textContent = st.wins;
    $("wh-best").textContent = fmt(st.best);
    const p = $("wh-profit");
    p.textContent = (st.profit < 0 ? "-" : "") + fmt(Math.abs(st.profit));
    p.style.color = st.profit > 0 ? "#62d96b" : st.profit < 0 ? "var(--pk-danger)" : "";
  }
  function paintHistory() {
    const box = $("wh-history");
    box.innerHTML = "";
    for (const h of st.history.slice(-10)) {
      const d = document.createElement("div");
      d.className = "wh-chip " + (h.won ? "win" : "loss");
      d.style.background = COLORS[h.value];
      d.textContent = h.value;
      box.appendChild(d);
    }
  }

  async function spin() {
    if (st.busy || !st.layout) return;
    const total = totalStaked();
    if (total <= 0) { msg.textContent = "place a bet on at least one number"; return; }
    // snapshot the stakes for this spin — the server settles them all at once
    const bets = { ...st.stakes };
    st.busy = true;
    btn.disabled = true;
    setPicks(false);
    try {
      const r = await fetch("/api/wheel", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ bets }),
      });
      const d = await r.json();
      if (!r.ok) { msg.textContent = d.error; return; }
      msg.textContent = "";
      setBalance(BALANCE - total);        // payout lands when the wheel does
      if (window.sfx) sfx.click();
      hint.className = "hint";
      hint.textContent = "Spinning…";
      const segDeg = 360 / st.layout.length;
      // land inside segment d.index with a little jitter so stops vary
      const target = -d.index * segDeg + (Math.random() - 0.5) * segDeg * 0.7;
      st.rot += 3 * 360 + (((target - st.rot) % 360) + 360) % 360;
      canvas.classList.add("spin");
      canvas.style.transform = `rotate(${st.rot}deg)`;
      await new Promise((res) => setTimeout(res, SPIN_MS + 150));
      st.spins++;
      st.profit += d.payout - total;
      if (d.won) { st.wins++; st.best = Math.max(st.best, d.payout); }
      st.history.push({ value: d.value, won: d.won });
      paintStats();
      paintHistory();
      setBalance(d.balance);
      if (window.plinkoRefreshUI) plinkoRefreshUI();
      if (window.sfx) (d.won ? (d.value >= 10 ? sfx.bigwin() : sfx.win()) : sfx.lose());
      hint.className = "hint " + (d.won ? "win" : "loss");
      const staked = bets[d.value] || 0;
      hint.textContent = d.won
        ? `Landed on ${d.value} — you staked ${fmt(staked)} and win ${fmt(d.payout)} credits! 🎉`
        : `Landed on ${d.value} — you had no chips there, ${fmt(total)} lost 💀`;
    } catch (e) {
    } finally { st.busy = false; btn.disabled = false; setPicks(true); paintPicks(); }
  }

  btn.onclick = spin;
  $("wh-clear").onclick = clearAll;
  $("wh-half").onclick = () => {
    $("wh-bet").value = Math.max(1, Math.floor(Number($("wh-bet").value || 0) / 2));
  };
  $("wh-double").onclick = () => doubleBet($("wh-bet"));

  (async () => {
    const info = await (await fetch("/api/games/info")).json();
    applyCasinoState(info.casino_enabled);
    if (info.bet_min) { $("wh-bet").value = info.bet_min; $("wh-bet").min = info.bet_min; }
    if (info.wheel) {
      st.layout = info.wheel.layout;
      st.payouts = info.wheel.payouts;
      buildPicks();
      drawWheel();
    }
    if (!info.logged_in) {
      btn.disabled = true;
      setPicks(false);
      hint.textContent = "Log in on the home page to play.";
      return;
    }
    hint.textContent = "Stake chips on any numbers you like, then spin — the number it lands on pays that many times your stake, plus the stake back.";
  })();
})();

/* ================= CRASH ================= */
(() => {
  const cv = $("crash-canvas"), ctx = cv.getContext("2d");
  const multEl = $("crash-mult"), btn = $("crash-btn"), hint = $("crash-hint");

  // The client mirrors the server curve for the animation; the server clock
  // is the only one that counts, so a poll confirms the bust and every
  // cash-out is scored server-side. The animation is therefore only ever a
  // guess at what the server is doing, and it is only allowed to keep
  // guessing while the server keeps confirming it: STALE_AFTER_MS without a
  // successful poll and the counter freezes at the last elapsed the server
  // actually reported. Without that, dropping the connection let the curve
  // free-run to millions of × — a number no flight can reach, since every
  // bust point is capped at st.max.
  const STALE_AFTER_MS = 1500;   // ~4 missed polls at the 350ms interval

  const st = { flying: false, growth: 0.07, max: 1000, start: 0, round: 0,
               raf: null, poll: null, results: [], bet: 0,
               lastSync: 0, syncElapsed: 0, stale: false,
               flights: 0, cashed: 0, best: 0, profit: 0 };

  function paintCrashStats() {
    $("cr-flights").textContent = st.flights;
    $("cr-cashed").textContent = st.cashed;
    $("cr-best").textContent = st.best ? "×" + st.best.toFixed(2) : "—";
    const p = $("cr-profit");
    p.textContent = (st.profit < 0 ? "-" : "") + Math.abs(st.profit).toLocaleString();
    p.style.color = st.profit > 0 ? "#62d96b" : st.profit < 0 ? "var(--pk-danger)" : "";
  }

  const mult = (elapsed) => Math.min(Math.exp(st.growth * elapsed), st.max);
  const elapsedNow = () => (performance.now() - st.start) / 1000;

  /* The flight is still running on the server — we just can't see it. Freeze
     the readout and say so, rather than inventing multipliers nobody can
     actually cash out at. */
  function setStale(stale) {
    if (stale === st.stale) return;   // called every frame; only touch the DOM on a change
    st.stale = stale;
    multEl.classList.toggle("stale", stale);
    hint.className = "hint" + (stale ? " loss" : "");
    hint.textContent = stale
      ? "Connection lost — the flight is still running on the server. Reconnect to cash out."
      : "Flying — cash out any time.";
  }

  function drawCurve(elapsed, crashed) {
    ctx.clearRect(0, 0, cv.width, cv.height);
    const m = mult(elapsed);
    const tMax = Math.max(8, elapsed * 1.15), mMax = Math.max(2, m * 1.15);
    ctx.strokeStyle = "#24242a";
    ctx.lineWidth = 1;
    ctx.strokeRect(0.5, 0.5, cv.width - 1, cv.height - 1);
    ctx.strokeStyle = crashed ? "#ff6b70" : accentColour();
    ctx.lineWidth = 3;
    ctx.beginPath();
    for (let i = 0; i <= 80; i++) {
      const t = elapsed * i / 80;
      const x = 12 + (t / tMax) * (cv.width - 24);
      const y = cv.height - 14 - ((mult(t) - 1) / (mMax - 1)) * (cv.height - 60);
      i ? ctx.lineTo(x, y) : ctx.moveTo(x, y);
    }
    ctx.stroke();
  }

  function setIdle(text, cls) {
    st.flying = false;
    cancelAnimationFrame(st.raf);
    clearInterval(st.poll);
    st.poll = null;
    btn.textContent = "Launch";
    btn.classList.remove("cashout");
    $("crash-auto").disabled = false;
    st.stale = false;
    multEl.className = "crash-mult " + (cls || "");
    if (text) multEl.innerHTML = text;
  }

  function bust(crashAt, round) {
    if (round !== st.round || !st.flying) return;
    drawCurve(Math.log(crashAt) / st.growth, true);
    setIdle(`${crashAt.toFixed(2)}×<small>CRASHED</small>`, "bust");
    hint.className = "hint loss";
    hint.textContent = "Crashed at " + crashAt.toFixed(2) + "× — bet lost.";
    st.profit -= st.bet;
    paintCrashStats();
    if (window.sfx) sfx.boom();
  }

  function win(d, round) {
    if (round !== st.round || !st.flying) return;
    drawCurve(Math.log(d.multiplier) / st.growth, false);
    setIdle(`${d.multiplier.toFixed(2)}×<small>+${d.payout.toLocaleString()} CREDITS</small>`, "win");
    hint.className = "hint win";
    hint.textContent = (d.cashed ? "Auto cashed out at " : "Cashed out at ")
      + d.multiplier.toFixed(2) + "× for " + d.payout.toLocaleString() + " credits!";
    st.cashed++;
    st.best = Math.max(st.best, d.multiplier);
    st.profit += d.payout - st.bet;
    paintCrashStats();
    if (window.sfx) (d.multiplier >= 5 ? sfx.bigwin() : sfx.cash());
    setBalance(d.balance);
    if (window.plinkoRefreshUI) plinkoRefreshUI();
  }

  function animate(round) {
    if (round !== st.round || !st.flying) return;
    const live = performance.now() - st.lastSync <= STALE_AFTER_MS;
    setStale(!live);
    // Unconfirmed, hold at the last elapsed the server vouched for.
    const e = live ? elapsedNow() : st.syncElapsed;
    multEl.textContent = mult(e).toFixed(2) + "×";
    drawCurve(e, false);
    st.raf = requestAnimationFrame(() => animate(round));
  }

  function startFlight(elapsed) {
    st.flying = true;
    st.round++;
    const round = st.round;
    st.start = performance.now() - elapsed * 1000;
    st.lastSync = performance.now();
    st.syncElapsed = elapsed;
    st.stale = false;
    btn.textContent = "Cash out";
    btn.classList.add("cashout");
    $("crash-auto").disabled = true;
    multEl.className = "crash-mult";
    hint.className = "hint";
    hint.textContent = "Flying — cash out any time.";
    animate(round);
    st.poll = setInterval(async () => {
      try {
        const r = await fetch("/api/crash/state");
        if (!r.ok) return;
        const d = await r.json();
        if (!d.state) {
          // The server has no flight for us: it settled somewhere we didn't
          // see it — another tab, or a response lost on the way back. Stop
          // animating a flight that is already over.
          setIdle("", "");
          hint.className = "hint";
          hint.textContent = "Flight already settled — see the history for the result.";
          return;
        }
        if (d.state.crashed) { bust(d.state.crash_at, round); return; }
        if (d.state.cashed) { win(d.state, round); return; }
        // Re-anchor on the server clock. performance.now() runs on while the
        // tab is throttled or the machine sleeps, so the client's own elapsed
        // drifts ahead of the truth if it is never corrected.
        st.lastSync = performance.now();
        st.syncElapsed = d.state.elapsed;
        st.start = st.lastSync - d.state.elapsed * 1000;
        if (d.state.max) st.max = d.state.max;
      } catch (e) {
        // Offline. animate() freezes the readout once the sync goes stale.
      }
    }, 350);
  }

  async function launch() {
    const bet = Math.floor(Number($("crash-bet").value));
    btn.disabled = true;
    try {
      const r = await fetch("/api/crash/start", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ bet, auto_cashout: $("crash-auto").value || null }),
      });
      const data = await r.json();
      if (!r.ok) { hint.className = "hint loss"; hint.textContent = data.error; return; }
      st.growth = data.growth;
      if (data.max) st.max = data.max;
      st.bet = bet;
      st.flights++;
      paintCrashStats();
      setBalance(data.balance);
      if (window.plinkoRefreshUI) plinkoRefreshUI();
      startFlight(0);
    } finally { btn.disabled = false; }
  }

  async function cashout() {
    const round = st.round;
    btn.disabled = true;
    try {
      const r = await fetch("/api/crash/cashout", { method: "POST" });
      const data = await r.json();
      if (!r.ok) {
        hint.className = "hint loss";
        hint.textContent = data.error || "Cash out failed — try again.";
        return;
      }
      if (data.crashed) { bust(data.crash_at, round); return; }
      win(data, round);
    } catch (e) {
      // The request never reached the server, so nothing was cashed out and
      // the flight is still busting on the server clock. Say that plainly
      // instead of leaving a dead "Cash out" button. Once the polls have gone
      // stale too the frozen-counter notice already says it, and setStale
      // owns the hint from there.
      if (round === st.round && st.flying && !st.stale) {
        hint.className = "hint loss";
        hint.textContent = "Cash out didn't reach the server — try again.";
      }
    } finally { btn.disabled = false; }
  }

  btn.onclick = () => (st.flying ? cashout() : launch());
  $("crash-half").onclick = () => {
    $("crash-bet").value = Math.max(1, Math.floor(Number($("crash-bet").value || 0) / 2));
  };
  $("crash-double").onclick = () => doubleBet($("crash-bet"));
  drawCurve(0, false);

  // resume a flight that survived a page reload
  (async () => {
    const info = await (await fetch("/api/games/info")).json();
    applyCasinoState(info.casino_enabled);
    if (info.bet_min) { $("crash-bet").value = info.bet_min; $("crash-bet").min = info.bet_min; }
    if (!info.logged_in) {
      btn.disabled = true;
      hint.textContent = "Log in on the home page to play.";
      return;
    }
    try {
      const d = await (await fetch("/api/crash/state")).json();
      if (d.state && !d.state.crashed) {
        st.growth = d.state.growth;
        if (d.state.max) st.max = d.state.max;
        startFlight(d.state.elapsed);
      }
      else if (d.state && d.state.crashed) {
        setIdle(`${d.state.crash_at.toFixed(2)}×<small>CRASHED</small>`, "bust");
      }
    } catch (e) {}
  })();
})();

(() => {

    c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));

  const house = (() => {

        if (!r.ok) { msg.textContent = d.error; return; }

            ? `${NAME[d.house]} beats ${NAME[d.you]} — bet lost 💀`
            : `Both threw ${NAME[d.you]} — push, your ${fmt(bet)} is back`;
      } catch (e) {
      } finally { st.busy = false; btn.disabled = false; }

        multiplier = info.rps_multiplier || 0;
        if (!info.logged_in) {

          hint.textContent = "Log in on the home page to play.";
          return;
        }
        if (multiplier) {
          hint.textContent = `Beat the house and win ${multiplier}× your bet — a tie gives your stake back.`;
        }
      },
    };
  })();

  /* ---------- multiplayer ---------- */
  const pvp = (() => {
    const stage = $("rp-pvp-stage"), matchBox = $("rp-match"), lobbyBox = $("rp-lobby");
    const openBtn = $("rp-open"), leaveBtn = $("rp-leave"), msg = $("rp-pvp-msg"), hint = $("rp-pvp-hint");
    // `pending` is the throw we've posted but not yet seen confirmed — it
    // keeps the button lit through the poll that lands in between.
    const st = { info: null, state: null, tables: [], history: [], loggedIn: false,
                 pending: null, busy: false, lastMatch: null, timer: null, poll: null };

    const active = () => st.state && st.state.status !== "done";
    const live = () => st.state && st.state.status === "live";

    /* --- rendering --- */
    function matchHTML(s) {
      const target = s.best_of / 2 + 0.5;
      const pips = (n, them) => Array.from({ length: target }, (_, i) =>
        `<i class="rp-pip${them ? " them" : ""}${i < n ? " on" : ""}"></i>`).join("");

      if (s.status === "open") {

      const r = s.result;
      const head = `<div class="rp-match-head">
          <b>You vs ${esc(s.opponent || "?")}</b>
          <span class="rp-pot">pot <b>${fmt(s.stake * 2)}</b></span>
        </div>
        <div class="rp-score">
          <span class="rp-pips">${pips(s.wins.you, false)}</span>
          <span class="rp-vs">${s.wins.you} — ${s.wins.them}</span>

    function lobbyHTML() {
      const rows = st.tables.length ? st.tables.map(t => `
        <div class="rp-table-row${t.mine ? " mine" : ""}">
          <span>${esc(t.host)}</span>

      return `<h4 style="margin:${top} 0 4px; font-size:10px; letter-spacing:.12em; text-transform:uppercase; color:var(--pk-muted);">Open tables</h4>

    function render() {
      const s = st.state;
      matchBox.style.display = s ? "" : "none";
      lobbyBox.style.display = s && s.status !== "done" ? "none" : "";
      if (s) {

        if (big) for (const b of big.children) {

      leaveBtn.style.display = active() ? "" : "none";
      leaveBtn.textContent = live() ? "Forfeit match" : "Cancel table";

      } else if (st.info && !st.info.enabled) {
        openBtn.disabled = true;
      }
    }

    /* --- server --- */
    function apply(d) {
      st.info = d.info;
      st.tables = d.tables || [];
      st.history = d.history || [];
      st.loggedIn = !!d.logged_in;
      const prev = st.state;
      st.state = d.state || null;
      if (typeof d.balance === "number") setBalance(d.balance);
      // a new round (or a new match) clears the throw we were holding
      if (!st.state || !prev || st.state.match_id !== prev.match_id ||
          st.state.round !== prev.round || !st.state.you_picked) {
        if (!st.state || !st.state.you_picked) st.pending = null;
      }
      if (window.sfx && st.state && st.state.status === "done" &&
          st.lastMatch !== st.state.match_id && st.state.result &&
          !st.state.result.cancelled) {
        st.lastMatch = st.state.match_id;
        st.state.result.outcome === "win" ? sfx.win() : sfx.lose();
      }
      render();
    }

    async function refresh() {
      try {
        apply(await (await fetch("/api/rps/state")).json());
      } catch (e) {}
    }

    async function post(url, body) {
      if (st.busy) return null;
      st.busy = true;
      try {
        const r = await fetch(url, {
          method: "POST", headers: { "Content-Type": "application/json" },
          body: JSON.stringify(body || {}),
        });
        const d = await r.json();
        if (!r.ok) { msg.textContent = d.error || "something went wrong"; return null; }
        msg.textContent = "";
        if (d.state) { st.state = d.state; render(); }
        if (typeof d.balance === "number") setBalance(d.balance);
        return d;
      } catch (e) {
        return null;
      } finally { st.busy = false; }
    }

    async function openTable() {
      const stake = Math.floor(Number($("rp-stake").value));
      const target = $("rp-target").value.trim();
      if (await post("/api/rps/table", { stake, target })) {
        if (window.sfx) sfx.click();
        $("rp-target").value = "";
        refresh();
      }
    }

    async function join(matchId) {
      if (await post("/api/rps/join", { match_id: matchId })) {
        if (window.sfx) sfx.click();
      } else {
        refresh();  // the table was probably taken a moment before we clicked
      }
    }

    async function throwPick(name) {
      if (!live() || st.state.you_picked || st.pending) return;
      st.pending = name;
      render();
      const d = await post("/api/rps/pick", { throw: name });
      if (!d) { st.pending = null; refresh(); return; }
      if (window.sfx) sfx.click();
    }

    async function leave(silent) {
      if (!silent && live() &&
          !confirm("Forfeit the match? Your opponent takes the pot.")) return;
      await post("/api/rps/leave", {});
      st.state = null;
      st.pending = null;
      render();
      refresh();
    }

    openBtn.onclick = openTable;
    leaveBtn.onclick = () => leave(false);
    $("rp-stake-half").onclick = () => {
      $("rp-stake").value = Math.max(1, Math.floor(Number($("rp-stake").value || 0) / 2));
    };
    $("rp-stake-double").onclick = () => doubleBet($("rp-stake"));

    return {
      show(on) {
        stage.style.display = on ? "" : "none";
        clearInterval(st.poll);
        clearInterval(st.timer);
        if (!on) return;
        refresh();
        // one poll a second while a round is running (the clock is ticking
        // and the opponent could throw at any moment), slower when we're
        // only watching the table list
        st.poll = setInterval(() => {
          if (!document.hidden) refresh();
        }, 1000);
        // the countdown itself is redrawn locally between polls so it doesn't
        // visibly jump a second at a time
        st.timer = setInterval(() => {
          if (!live() || document.hidden) return;
          st.state.deadline_in = Math.max(0, st.state.deadline_in - 0.25);
          const s = st.state;
          const el = matchBox.querySelector(".rp-clock");
          const bar = matchBox.querySelector(".rp-bar i");
          if (!el) return;
          el.textContent = `Round ${s.round} · ${s.deadline_in.toFixed(1)}s`;
          el.classList.toggle("urgent", s.deadline_in <= 5);
          if (bar && st.info) {
            bar.style.width = Math.max(0, Math.min(100, s.deadline_in / st.info.pick_seconds * 100)) + "%";
          }
        }, 250);
      },
      apply(info) {
        st.loggedIn = info.logged_in;
        if (!info.logged_in) {
          hint.textContent = "Log in on the home page to play.";
          openBtn.disabled = true;
        }
      },
      rules(info) {
        if (!info) return;
        hint.textContent = info.enabled
          ? `Best of ${info.best_of} — ties replay. ${info.pick_seconds}s a round; miss it and you forfeit. `
            + `Winner takes the pot${info.rake_pct ? ` less a ${info.rake_pct}% rake` : ""}.`
          : "Multiplayer is switched off right now — the house game still works.";
        openBtn.disabled = !info.enabled || !st.loggedIn;
      },
    };
  })();

  /* ---------- mode switch ---------- */
  function mode(name) {
    const isPvp = name === "pvp";
    $("rp-mode-house").classList.toggle("active", !isPvp);
    $("rp-mode-pvp").classList.toggle("active", isPvp);
    $("rp-house-controls").style.display = isPvp ? "none" : "";
    $("rp-pvp-controls").style.display = isPvp ? "" : "none";
    $("rp-house-stage").style.display = isPvp ? "none" : "";
    pvp.show(isPvp && panel.style.display !== "none");
  }
  $("rp-mode-house").onclick = () => mode("house");
  $("rp-mode-pvp").onclick = () => mode("pvp");

  // Polling stops when the panel isn't the open game — every game's module
  // loads on this page, and only one of them is ever on screen.
  const watchPanel = () => {
    const open = panel.style.display !== "none";
    pvp.show(open && $("rp-mode-pvp").classList.contains("active"));
  };
  window.addEventListener("hashchange", watchPanel);
  document.addEventListener("visibilitychange", watchPanel);

  (async () => {
    const info = await (await fetch("/api/games/info")).json();
    applyCasinoState(info.casino_enabled);
    house.apply(info);
    pvp.apply(info);
    try {
      const d = await (await fetch("/api/rps/state")).json();
      pvp.rules(d.info);
      // a table or a live match survived a reload — drop straight back into
      // it. A finished one doesn't hijack the page: it's a result, not a game.
      if (d.state && d.state.status !== "done") mode("pvp");
      else watchPanel();
    } catch (e) { watchPanel(); }
  })();
})();
