<script>
const $ = (id) => document.getElementById(id);

/* ================= TABS ================= */
(() => {
  const buttons = document.querySelectorAll(".tabs button");
  const names = [...buttons].map(b => b.dataset.tab);
  function show(name) {
    buttons.forEach(b => b.classList.toggle("active", b.dataset.tab === name));
    document.querySelectorAll("[data-tab-panel]").forEach(s => {
      s.style.display = s.dataset.tabPanel === name ? "" : "none";
    });
    history.replaceState(null, "", "#" + name);
  }
  buttons.forEach(b => b.onclick = () => show(b.dataset.tab));
  const initial = location.hash.slice(1);
  show(names.includes(initial) ? initial : names[0]);
  // hash edited while already on the page (e.g. a #tab link) still switches
  window.addEventListener("hashchange", () => {
    const name = location.hash.slice(1);
    if (names.includes(name)) show(name);
  });
})();

/* ================= STOCKS ================= */
const st = {
  sel: null, side: "buy",
  quotes: [], pf: [], bal: 0, fee: 1, depth: 250000, logged: false, admin: false,
  sellAll: false, tf: 60, tradingEnabled: true,
};

// chart timeframes (label, minutes)
const TIMEFRAMES = [["30m", 30], ["1h", 60], ["4h", 240], ["12h", 720], ["24h", 1440]];
(() => {
  const box = $("tf");
  for (const [label, minutes] of TIMEFRAMES) {
    const b = document.createElement("button");
    b.textContent = label;
    b.classList.toggle("active", minutes === st.tf);
    b.onclick = () => {
      st.tf = minutes;
      [...box.children].forEach(x => x.classList.toggle("active", x === b));
      refresh(); // re-fetch quotes so change % matches the new timeframe, then redraws
    };
    box.appendChild(b);
  }
})();

const quote = () => st.quotes.find(q => q.symbol === st.sel);
const holding = () => st.pf.find(h => h.symbol === st.sel);
const fmt = (n) => Math.floor(n).toLocaleString();

async function refresh() {
  const tf = st.tf;
  const d = await (await fetch("/api/stocks?minutes=" + tf)).json();
  if (tf !== st.tf) return; // timeframe switched mid-fetch; a fresher refresh is coming
  st.logged = d.logged_in;
  st.admin = d.is_admin;
  st.tradingEnabled = d.trading_enabled;
  st.quotes = d.quotes;
  st.fee = d.fee_pct !== undefined ? d.fee_pct : 1;
  if (d.depth) st.depth = d.depth;
  st.news = d.news || [];
  st.pf = d.portfolio || [];
  st.nw = d.networth_history || [];
  if (d.balance !== undefined) {
    st.bal = d.balance;
    $("balance").textContent = d.balance.toLocaleString() + " credits";
  }
  if (!st.sel || !quote()) st.sel = st.quotes.length ? st.quotes[0].symbol : null;
  renderAll();
  drawChart();
}

function select(sym) {
  if (sym === st.sel) return;
  st.sel = sym;
  st.sellAll = false;
  renderAll();
  drawChart();
}

function renderAll() {
  renderPfChart();
  renderHead();
  renderTickers();
  renderNews();
  renderPortfolio();
  renderTrade();
  renderMktAdmin();
}

/* ---------- portfolio value over time (net-worth snapshots per market tick) ---------- */
function renderPfChart() {
  const card = $("pf-chart-card");
  const cutoff = Date.now() / 1000 - st.tf * 60;
  const nw = (st.nw || []).filter(p => p[0] >= cutoff);
  if (!st.logged || nw.length < 2) { card.style.display = "none"; return; }
  card.style.display = "";

  const vals = nw.map(p => p[1]);
  const first = vals[0], last = vals[vals.length - 1];
  $("pf-value").textContent = fmt(last);
  changePill($("pf-change"), first > 0 ? (last - first) / first * 100 : 0);

  const cv = $("pf-chart"), ctx = cv.getContext("2d");
  const W = cv.width, H = cv.height;
  ctx.clearRect(0, 0, W, H);
  let min = Math.min(...vals), max = Math.max(...vals);
  const pad = (max - min) * 0.1 || max * 0.01 || 1;
  min = Math.max(0, min - pad); max += pad;
  const span = max - min;
  const PL = 10, PR = 72, PT = 10, PB = 10;
  const px = (i) => PL + i / (vals.length - 1) * (W - PL - PR);
  const py = (v) => PT + (1 - (v - min) / span) * (H - PT - PB);

  ctx.font = "11px ui-monospace, monospace";
  ctx.textBaseline = "middle";
  for (let g = 0; g <= 3; g++) {
    const v = min + span * g / 3;
    const y = py(v);
    ctx.strokeStyle = "#24242a";
    ctx.beginPath(); ctx.moveTo(PL, y); ctx.lineTo(W - PR + 6, y); ctx.stroke();
    ctx.fillStyle = "#8f99a8";
    ctx.fillText(Math.round(v).toLocaleString(), W - PR + 10, y);
  }

  const rising = last >= first;
  const color = rising ? "#32a467" : "#e76a6e";
  const grad = ctx.createLinearGradient(0, PT, 0, H - PB);
  grad.addColorStop(0, rising ? "rgba(50,164,103,0.18)" : "rgba(231,106,110,0.18)");
  grad.addColorStop(1, "rgba(0,0,0,0)");
  ctx.fillStyle = grad;
  ctx.beginPath();
  vals.forEach((v, i) => i ? ctx.lineTo(px(i), py(v)) : ctx.moveTo(px(0), py(v)));
  ctx.lineTo(px(vals.length - 1), H - PB);
  ctx.lineTo(px(0), H - PB);
  ctx.closePath();
  ctx.fill();

  ctx.strokeStyle = color;
  ctx.lineWidth = 2;
  ctx.lineJoin = "round";
  ctx.beginPath();
  vals.forEach((v, i) => i ? ctx.lineTo(px(i), py(v)) : ctx.moveTo(px(0), py(v)));
  ctx.stroke();

  ctx.fillStyle = color;
  ctx.beginPath(); ctx.arc(px(vals.length - 1), py(last), 3.5, 0, Math.PI * 2); ctx.fill();
}

function ago(ts) {
  const s = Math.max(0, Math.floor(Date.now() / 1000 - ts));
  if (s < 60) return s + "s";
  if (s < 3600) return Math.floor(s / 60) + "m";
  if (s < 86400) return Math.floor(s / 3600) + "h";
  return Math.floor(s / 86400) + "d";
}

  $("side-sell").classList.toggle("active", !buy);
  // Master switch: when trading is off, lock every control so users can't
  // buy or sell. The backend rejects trades too, this just matches the UI.
  $("side-buy").disabled = tradingOff;
  $("side-sell").disabled = tradingOff;
  $("trade-amount").disabled = tradingOff;
  $("amt-label").textContent = buy ? "Amount (credits)" : "Shares";
  $("avail").textContent = buy
    ? "Balance: " + st.bal.toLocaleString()
    : "You own: " + (h ? h.shares.toFixed(2) : "0");

  const quick = $("quick");
  quick.innerHTML = "";
  const opts = buy
    ? [["10%", 0.10], ["25%", 0.25], ["50%", 0.50], ["Max", 1]]
    : [["25%", 0.25], ["50%", 0.50], ["75%", 0.75], ["All", 1]];
  for (const [label, frac] of opts) {
    const b = document.createElement("button");
    b.textContent = label;
    b.disabled = tradingOff;
    b.onclick = () => {
      st.sellAll = !buy && frac === 1;
      if (buy) {
        $("trade-amount").value = Math.floor(st.bal * frac);
      } else {
        $("trade-amount").value = h ? +(h.shares * frac).toFixed(2) : 0;
      }
      updateEst();
    };
    quick.appendChild(b);
  }

  const btn = $("trade-btn");
  btn.textContent = (buy ? "Buy" : "Sell") + (st.sel ? " $" + st.sel : "");
  btn.className = "submit " + (buy ? "buy" : "sell");
  btn.disabled = tradingOff || (!buy && !h);
  if (tradingOff) $("trade-hint").textContent = "🚫 The stock market is currently disabled.";
  updateEst();
}

function updateEst() {
  const q = quote();
  const v = Number($("trade-amount").value);
  const est = $("est");
  if (!q || !(v > 0)) { est.innerHTML = "&nbsp;"; return; }
  // mirrors the server's AMM math: trades execute through their own impact
  const D = st.depth, p = q.price;
  const priceLine = (np) => {
    const pct = (np - p) / p * 100;
    return `price ${p.toFixed(2)} → ${np.toFixed(2)} ` +
           `<span class="${pct >= 0 ? "up" : "down"}">(${pct >= 0 ? "+" : ""}${pct.toFixed(2)}%)</span>`;
  };
  if (st.side === "buy") {
    const shares = D / p * (1 - Math.exp(-Math.min(v / D, 50)));
    const newPrice = p * Math.exp(Math.min(v / D, 50));
    est.innerHTML = "≈ " + shares.toFixed(2) + " shares @ " + (v / shares).toFixed(2)
      + " avg<br>" + priceLine(newPrice);
  } else {
    const gross = D * Math.log(1 + v * p / D);
    const newPrice = p / (1 + v * p / D);
    const proceeds = Math.floor(gross * (1 - st.fee / 100));
    est.innerHTML = "≈ " + fmt(proceeds) + " credits after " + st.fee + "% fee<br>"
      + priceLine(newPrice);
  }
}

async function drawChart() {
  const sym = st.sel, tf = st.tf;
  if (!sym) return;
  const data = await (await fetch("/api/stocks/history?symbol=" + sym + "&minutes=" + tf)).json();
  if (sym !== st.sel || tf !== st.tf) return; // user switched ticker/timeframe mid-fetch
  const cv = $("chart"), ctx = cv.getContext("2d");
  const W = cv.width, H = cv.height;
  ctx.clearRect(0, 0, W, H);
  const hist = data.history;
  if (hist.length < 2) {
    ctx.fillStyle = "#8f99a8"; ctx.font = "13px ui-monospace, monospace";
    ctx.fillText("Chart appears after a few price ticks…", 16, 40);
    return;
  }
  const prices = hist.map(p => p[1]);
  let min = Math.min(...prices), max = Math.max(...prices);
  const pad = (max - min) * 0.08 || max * 0.01 || 1;
  min -= pad; max += pad;
  const span = max - min;
  const PL = 10, PR = 64, PT = 10, PB = 10;
  const px = (i) => PL + i / (hist.length - 1) * (W - PL - PR);
  const py = (v) => PT + (1 - (v - min) / span) * (H - PT - PB);

  // gridlines + right-side price labels
  ctx.font = "11px ui-monospace, monospace";
  ctx.textBaseline = "middle";
  for (let g = 0; g <= 3; g++) {
    const v = min + span * g / 3;
    const y = py(v);
    ctx.strokeStyle = "#24242a";
    ctx.beginPath(); ctx.moveTo(PL, y); ctx.lineTo(W - PR + 6, y); ctx.stroke();
    ctx.fillStyle = "#8f99a8";
    ctx.fillText(v.toFixed(2), W - PR + 10, y);
  }

  const rising = prices[prices.length - 1] >= prices[0];
  const color = rising ? "#32a467" : "#e76a6e";

  // gradient fill under the line
  const grad = ctx.createLinearGradient(0, PT, 0, H - PB);
  grad.addColorStop(0, rising ? "rgba(50,164,103,0.18)" : "rgba(231,106,110,0.18)");
  grad.addColorStop(1, "rgba(0,0,0,0)");
  ctx.fillStyle = grad;
  ctx.beginPath();
  hist.forEach((p, i) => i ? ctx.lineTo(px(i), py(p[1])) : ctx.moveTo(px(0), py(p[1])));
  ctx.lineTo(px(hist.length - 1), H - PB);
  ctx.lineTo(px(0), H - PB);
  ctx.closePath();
  ctx.fill();

  // price line
  ctx.strokeStyle = color;
  ctx.lineWidth = 2;
  ctx.lineJoin = "round";
  ctx.beginPath();
  hist.forEach((p, i) => i ? ctx.lineTo(px(i), py(p[1])) : ctx.moveTo(px(0), py(p[1])));
  ctx.stroke();

  // last price marker
  const lastX = px(hist.length - 1), lastY = py(prices[prices.length - 1]);
  ctx.fillStyle = color;
  ctx.beginPath(); ctx.arc(lastX, lastY, 3.5, 0, Math.PI * 2); ctx.fill();
}

function marketCard(p, isOpen) {
  const card = document.createElement("section");
  card.className = "card market" + (isOpen ? "" : " resolved");

  const when = new Date((p.resolved_at || p.created_at) * 1000)
    .toLocaleDateString(undefined, { month: "short", day: "numeric" });
  card.innerHTML = `
    <div class="q">${esc(p.question)}</div>
    <div class="meta">
      <span>${fmt(p.total)} credits bet</span><span>·</span><span>${when}</span>
      ${isOpen && p.locked ? '<span class="tag" style="color:var(--red-text)">🔒 LOCKED</span>' : ""}
      ${isOpen && !p.locked && p.locks_at
        ? `<span>· locks in ${Math.max(1, Math.round((p.locks_at - Date.now() / 1000) / 60))}m</span>` : ""}
      ${isOpen ? "" : '<span class="tag">RESOLVED</span>'}
    </div>`;

  // one bet per prediction: once you've bet, every option locks
  const myOpt = st.logged ? p.options.find((o) => o.your_bet > 0) : null;
  const isLocked = isOpen && p.locked;

  p.options.forEach((o, i) => {
    const isWinner = !isOpen && o.id === p.winning_option_id;
    const bettable = isOpen && st.logged && !myOpt && !isLocked;
    // parimutuel payout if this option wins: the whole pool split pro-rata
    const mult = o.pool > 0 ? (p.total / o.pool).toFixed(2) : null;
    const opt = document.createElement("div");
    opt.className = "opt"
      + (bettable ? " clickable" : "")
      + (isOpen && st.logged && ((myOpt && !o.your_bet) || (isLocked && !o.your_bet)) ? " locked" : "")
      + (!isOpen && !isWinner ? " loser" : "")
      + (st.expanded === p.id + ":" + o.id ? " expanded" : "");
    opt.innerHTML = `
      <div class="opt-top">
        <span class="opt-label">${esc(o.label)}
          ${isWinner ? '<span class="win-tag">✓ WINNER</span>' : ""}

        <span class="opt-pct">${o.pct.toFixed(1)}%</span>
      </div>
      <div class="bar"><div class="fill" style="width:${o.pct}%;background:${COLORS[i % COLORS.length]}"></div></div>

        <span>${fmt(o.pool)} credits</span>
        <span>${o.bettors} bettor${o.bettors === 1 ? "" : "s"}</span>
        ${p.total > 0 ? `<span>pays ×${mult ?? "—"}</span>` : ""}
        ${o.your_bet ? `<span class="you">You: ${fmt(o.your_bet)}</span>` : ""}
      </div>`;
    if (bettable) {
      opt.onclick = (e) => {
        if (e.target.closest(".bet-panel")) return; // clicks inside the panel don't toggle
        const key = p.id + ":" + o.id;
        st.expanded = st.expanded === key ? null : key;
        st.amount = "";
        render();
      };
      if (st.expanded === p.id + ":" + o.id) opt.appendChild(betPanel(p, o));
    }
    card.appendChild(opt);
  });

  if (isOpen && myOpt) {
    const note = document.createElement("div");
    note.className = "hint";
    note.textContent = `You bet ${fmt(myOpt.your_bet)} on “${myOpt.label}” — locked in until it resolves.`;
    card.appendChild(note);

/* what happened to *your* stake on a resolved market */
function resultLine(p) {
  const bet = p.options.reduce((s, o) => s + o.your_bet, 0);
  if (!bet) return null;
  const payout = p.options.reduce((s, o) => s + o.your_payout, 0);
  const el = document.createElement("div");
  el.className = "result-line";
  const won = p.options.some((o) => o.id === p.winning_option_id && o.your_payout > 0);
  if (won) {
    el.innerHTML = `You bet ${fmt(bet)} and won <b class="up">+${fmt(payout)}</b> credits`;
  } else if (payout > 0) {
    el.innerHTML = `Your ${fmt(bet)} credits were <b>refunded</b>`;
  } else {
    el.innerHTML = `You bet ${fmt(bet)} and <b class="down">lost</b>`;
  }
  return el;
}

function betPanel(p, o) {
  const panel = document.createElement("div");
  panel.className = "bet-panel";
  panel.innerHTML = `
    <div class="bet-row">
      <input type="number" id="bet-amount" min="1" step="1" placeholder="Amount"
             value="${esc(st.amount)}">
      <button class="btn success" id="bet-btn">Bet</button>

  const input = panel.querySelector("#bet-amount");
  const est = panel.querySelector("#bet-est");

  const updateEst = () => {
    st.amount = input.value;
    const a = Math.floor(Number(input.value));
    if (!(a > 0)) { est.textContent = "Balance: " + fmt(st.bal); return; }
    // parimutuel: your bet joins the pools, winners split everything pro-rata
    const payout = Math.floor(a * (p.total + a) / (o.pool + a));
    est.innerHTML = `To win: <b>${fmt(payout)}</b> (x${(payout / a).toFixed(2)}) · Balance: ${fmt(st.bal)}`;
  };
  input.addEventListener("input", updateEst);
  updateEst();

  const quick = panel.querySelector("#bet-quick");

    b.onclick = () => {
      input.value = frac === null
        ? (Math.floor(Number(input.value)) || 0) + 100
        : Math.max(1, Math.floor(st.bal * frac));
      updateEst();
    };
    quick.appendChild(b);
  }

  panel.querySelector("#bet-btn").onclick = async () => {
    const amount = Math.floor(Number(input.value));
    const r = await fetch("/api/predictions/bet", {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ prediction_id: p.id, option_id: o.id, amount }),
    });
    const data = await r.json();
    st.hints[p.id] = { ok: r.ok, text: r.ok ? data.message : data.error };
    if (r.ok && window.sfx) sfx.cash();
    if (r.ok) {
      st.expanded = null;
      st.amount = "";
      if (data.balance !== undefined) {
        st.bal = data.balance;
        $("balance").textContent = data.balance.toLocaleString() + " credits";
      }
      setTimeout(() => { delete st.hints[p.id]; render(); }, 4000);
    }
    await refresh();
  };
  return panel;
}

/* ---------- admin ---------- */

async function adminPost(url, body, predictionId) {
  const r = await fetch(url, {
    method: "POST", headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  const data = await r.json();
  if (predictionId) st.hints[predictionId] = { ok: r.ok, text: r.ok ? data.message : data.error };
  await refresh();
  return r.ok;
}

function adminBar(p) {
  const bar = document.createElement("div");
  bar.className = "admin-bar";
  const sel = document.createElement("select");
  for (const o of p.options) {
    const op = document.createElement("option");
    op.value = o.id;
    op.textContent = o.label;
    sel.appendChild(op);
  }
  // the 4s poll rebuilds this card — restore the streamer's picked option
  if (st.resolveSel[p.id]) sel.value = st.resolveSel[p.id];
  sel.onchange = () => { st.resolveSel[p.id] = sel.value; };
  const resolveBtn = document.createElement("button");
  resolveBtn.className = "btn success";
  resolveBtn.textContent = "Resolve & pay";
  resolveBtn.onclick = () => {
    const label = sel.options[sel.selectedIndex].textContent;
    if (!confirm(`Resolve “${p.question}” as “${label}” and pay out ${fmt(p.total)} credits?`)) return;
    adminPost("/api/predictions/resolve",
      { prediction_id: p.id, option_id: Number(sel.value) }, p.id);
  };
  const whoBtn = document.createElement("button");
  whoBtn.className = "btn";
  whoBtn.textContent = "Bettors";
  whoBtn.onclick = async (e) => {
    const card = e.target.closest(".market");
    const old = card.querySelector(".bettors-list");
    if (old) { old.remove(); return; }
    const r = await fetch("/api/admin/predictions/bettors?prediction_id=" + p.id);
    if (!r.ok) return;
    const d = await r.json();
    const list = document.createElement("div");
    list.className = "bettors-list";
    if (!d.bettors.length) list.textContent = "No bets yet.";
    let lastOpt = null;
    for (const b of d.bettors) {
      if (b.option !== lastOpt) {
        lastOpt = b.option;
        const h = document.createElement("div");
        h.className = "b-opt";
        h.textContent = b.option;
        list.appendChild(h);
      }
      const row = document.createElement("div");
      row.className = "b-row";
      const who = document.createElement("span");
      who.textContent = b.username;
      const amt = document.createElement("span");
      amt.textContent = fmt(b.amount);
      row.append(who, amt);
      list.appendChild(row);
    }
    card.appendChild(list);
  };

  const lockBtn = document.createElement("button");
  lockBtn.className = "btn";
  lockBtn.textContent = p.locked ? "Unlock bets" : "Lock bets";
  lockBtn.onclick = () => {
    adminPost("/api/predictions/lock",
      { prediction_id: p.id, locked: !p.locked }, p.id);
  };
  const delBtn = document.createElement("button");
  delBtn.className = "btn danger";
  delBtn.textContent = "Delete & refund";
  delBtn.onclick = () => {
    if (!confirm(`Delete “${p.question}” and refund all ${fmt(p.total)} credits bet?`)) return;
    adminPost("/api/predictions/delete", { prediction_id: p.id }, p.id);
  };
  bar.append(sel, resolveBtn, lockBtn, whoBtn, delBtn);
  return bar;
}

function adminDeleteBar(p) {
