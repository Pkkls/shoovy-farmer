const signed = (n) => (n > 0 ? "+" : "") + fmt(n);

  function show(name) {
    buttons.forEach(b => b.classList.toggle("active", b.dataset.tab === name));
    document.querySelectorAll("[data-tab-panel]").forEach(s => {
      s.style.display = s.dataset.tabPanel === name ? "" : "none";
    });
    history.replaceState(null, "", "#" + name);
  }
  buttons.forEach(b => b.onclick = () => show(b.dataset.tab));
  const initial = location.hash.slice(1);
  show(names.includes(initial) ? initial : names[0]);
  // hash edited while already on the page (e.g. a #tab link) still switches
  window.addEventListener("hashchange", () => {
    const name = location.hash.slice(1);
    if (names.includes(name)) show(name);
  });
})();

function ago(ts) {
  const s = Math.max(0, Math.floor(Date.now() / 1000 - ts));
  if (s < 60) return s + "s ago";
  if (s < 3600) return Math.floor(s / 60) + "m ago";
  if (s < 86400) return Math.floor(s / 3600) + "h ago";
  return Math.floor(s / 86400) + "d ago";
}

function row(cells) {
  const tr = document.createElement("tr");
  for (const c of cells) tr.appendChild(c);
  return tr;
}
function td(text, cls) {
  const el = document.createElement("td");
  if (cls) el.className = cls;
  el.textContent = text;
  return el;
}
function tdAmt(n, opts = {}) {
  const el = td((opts.plain ? fmt(n) : signed(n)), "amt");
  if (!opts.plain) el.classList.add(n > 0 ? "up" : n < 0 ? "down" : "x");
  return el;
}

async function load() {
  const r = await fetch("/api/stats?days=" + statDays);
  if (r.status === 401) { $("login-box").style.display = "block"; return; }
  if (!r.ok) return;
  const d = await r.json();
  $("stats-box").style.display = "block";
  $("range").style.display = "flex";

  $("t-balance").textContent = fmt(d.balance);
  $("balance").textContent = fmt(d.balance) + " credits";
  $("t-since").textContent = d.member_since
    ? "since " + new Date(d.member_since * 1000).toLocaleDateString() : "";
  $("t-earned").textContent = fmt(d.earned_total);
  document.querySelector("#stats-box .tile:nth-child(2) .sub").textContent =
    d.days ? `last ${d.days} days` : "from the stream & chat";
  const net = $("t-net");
  net.textContent = signed(d.gambling_net);
  net.className = d.gambling_net > 0 ? "up" : d.gambling_net < 0 ? "down" : "";
  $("t-net-sub").textContent = d.games.length

    : "no bets yet";
  $("t-best").textContent = d.best_win.amount ? fmt(d.best_win.amount) : "—";
  $("t-best-sub").textContent = d.best_win.game ? "on " + d.best_win.game : "";

  const earnings = $("earnings");
  earnings.innerHTML = "";
  $("earnings-empty").style.display = d.earnings.length ? "none" : "block";
  for (const e of d.earnings)
    earnings.appendChild(row([td(e.source), td(e.count.toLocaleString() + "×", "n"), tdAmt(e.total)]));

  const games = $("games");
  games.innerHTML = "";
  $("games-empty").style.display = d.games.length ? "none" : "block";
  for (const g of d.games)
    games.appendChild(row([
      td(g.game), td(g.plays.toLocaleString(), "n"),
      td(g.winrate_pct !== null && g.winrate_pct !== undefined ? g.winrate_pct + "%" : "—", "n"),
      tdAmt(g.wagered, { plain: true }), tdAmt(g.won, { plain: true }), tdAmt(g.net),

  const o = d.other;

  const lines = [

/* net worth over time — snapshots recorded on every market tick */
function drawNetWorth(nw) {
  const card = $("nw-card"), cv = $("nw-chart");
  const show = nw.length >= 2;
  card.style.display = show ? "" : "none";
  if (!show) return;
  const ctx = cv.getContext("2d");
  const W = cv.width, H = cv.height, PR = 64, PT = 8, PB = 8;
  ctx.clearRect(0, 0, W, H);
  const vals = nw.map(p => p[1]);
  let min = Math.min(...vals), max = Math.max(...vals);
  const pad = (max - min) * 0.1 || max * 0.05 || 1;
  min = Math.max(0, min - pad); max += pad;
  const x = (i) => 4 + i / (vals.length - 1) * (W - PR - 8);
  const y = (v) => PT + (1 - (v - min) / (max - min)) * (H - PT - PB);
  ctx.font = "10px ui-monospace, monospace";
  ctx.textBaseline = "middle";
  for (const g of [0, 1]) {
    const v = min + (max - min) * g, gy = y(v);
    ctx.strokeStyle = "#24242a";
    ctx.beginPath(); ctx.moveTo(0, gy); ctx.lineTo(W - PR + 4, gy); ctx.stroke();
    ctx.fillStyle = "#8f99a8";
    ctx.fillText(Math.round(v).toLocaleString(), W - PR + 8, gy);
  }
  const rising = vals[vals.length - 1] >= vals[0];
  const color = rising ? "#32a467" : "#e76a6e";
  const grad = ctx.createLinearGradient(0, PT, 0, H - PB);
  grad.addColorStop(0, rising ? "rgba(50,164,103,0.18)" : "rgba(231,106,110,0.18)");
  grad.addColorStop(1, "rgba(0,0,0,0)");
  ctx.fillStyle = grad;
  ctx.beginPath();
  vals.forEach((v, i) => i ? ctx.lineTo(x(i), y(v)) : ctx.moveTo(x(0), y(v)));
  ctx.lineTo(x(vals.length - 1), H - PB); ctx.lineTo(x(0), H - PB);
  ctx.closePath(); ctx.fill();
  ctx.strokeStyle = color;
  ctx.lineWidth = 2;
  ctx.beginPath();
  vals.forEach((v, i) => i ? ctx.lineTo(x(i), y(v)) : ctx.moveTo(x(0), y(v)));
  ctx.stroke();
}

/* streamer-only: freeze / hide / stake badge / wipe straight from the modal */
function modBox(u, name) {
  const box = document.createElement("div");
  box.className = "mod-box";
  const label = document.createElement("div");
  label.className = "label";
  label.textContent = "Moderation";
  const row = document.createElement("div");
  row.className = "row";
  const state = document.createElement("div");
  state.className = "state";

  function paintState(m) {
    state.innerHTML = "";
    if (m.frozen) state.innerHTML += '<span class="tag" style="color:var(--red-text)">FROZEN</span>';
    if (m.hidden) state.innerHTML += '<span class="tag">HIDDEN FROM LEADERBOARD</span>';
    if (m.stake) state.innerHTML += '<span class="tag" style="color:var(--green-text)">STAKE AFFILIATE</span>';
  }

  /* the badges next to the name in this modal, kept in sync with the toggle.
     Both are repainted so the stake badge can't end up behind the fishing
     one after a toggle */
  function paintName(on) {
    name.querySelectorAll(".stake-badge, .fish-badge").forEach(b => b.remove());
    if (on && window.stakeBadge) name.appendChild(stakeBadge("lg"));
    if (u.fish_prestige && window.fishBadge)
      name.appendChild(fishBadge("lg", u.fish_prestige));
  }

  async function act(action) {
    const r = await fetch("/api/admin/moderate", {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ username: u.username, action }),
    });
    const d = await r.json();
    if (!r.ok) { state.innerHTML = ""; state.textContent = d.error || "failed"; return; }
    u.moderation = { frozen: d.frozen, hidden: d.hidden, stake: d.stake };
    freeze.textContent = d.frozen ? "Unfreeze" : "Freeze";
    freeze.className = "btn " + (d.frozen ? "success" : "danger");
    hide.textContent = d.hidden ? "Show on leaderboard" : "Hide from leaderboard";
    stake.textContent = d.stake ? "Remove stake badge" : "Give stake badge";
    paintName(d.stake);
    paintState(u.moderation);
    document.querySelectorAll(".modal .points").forEach(p =>
      p.textContent = d.balance.toLocaleString() + " credits");
    loadBoard();  // the board behind the modal reflects hides/wipes immediately
  }

  const freeze = document.createElement("button");
  freeze.className = "btn " + (u.moderation.frozen ? "success" : "danger");
  freeze.textContent = u.moderation.frozen ? "Unfreeze" : "Freeze";
  freeze.onclick = () => act(u.moderation.frozen ? "unfreeze" : "freeze");

  const hide = document.createElement("button");
  hide.className = "btn";
  hide.textContent = u.moderation.hidden ? "Show on leaderboard" : "Hide from leaderboard";
  hide.onclick = () => act(u.moderation.hidden ? "unhide" : "hide");

  const stake = document.createElement("button");
  stake.className = "btn";
  stake.textContent = u.moderation.stake ? "Remove stake badge" : "Give stake badge";
  stake.onclick = () => act(u.moderation.stake ? "unstake" : "stake");

  const wipe = document.createElement("button");
  wipe.className = "btn danger";
  wipe.textContent = "Wipe balance";
  wipe.onclick = () => {
    if (confirm(`Wipe ${u.username}'s entire balance? This goes through the ledger and shows in your logs.`))
      act("wipe");
  };

  row.append(freeze, hide, stake, wipe);
  paintState(u.moderation);
  box.append(label, row, state);
  return box;
}

function fillBoard(el, rows, me, value, emptyText) {
  el.innerHTML = "";
  if (!rows.length) {
    const li = document.createElement("li");
    li.className = "lb-empty";
    li.textContent = emptyText;
    el.appendChild(li);
    return;
  }
  for (const row of rows) {
    const li = document.createElement("li");
    if (me.logged_in && row.username === me.username) li.className = "me";
    let face;
    if (row.picture) {
      face = document.createElement("img");
      face.src = row.picture;
      face.alt = "";
      face.loading = "lazy";
    } else {
      face = document.createElement("span");
      face.textContent = (row.username[0] || "?").toUpperCase();
    }
    face.className = "face";
    const name = document.createElement("span");
    name.className = "name";
    const who = document.createElement("span");
    who.className = "who";
    who.textContent = row.username;
    name.appendChild(who);
    // badges are decoration and live in a separately-cached script, so a
    // stale or missing badges.js must cost a badge, never the whole board
    if (row.stake && window.stakeBadge) name.appendChild(stakeBadge());
    if (row.fish_prestige && window.fishBadge)
      name.appendChild(fishBadge("", row.fish_prestige));
    const bal = document.createElement("span");
    bal.className = "bal";
    bal.textContent = value(row);
    li.append(face, name, bal);
    li.addEventListener("click", () => openUser(row.username));
    el.appendChild(li);
  }
}

async function loadBoard() {
  const [me, lb] = await Promise.all([
    fetch("/api/me").then(r => r.json()),
    fetch("/api/leaderboards").then(r => r.json()),
  ]);
  $("my-rank").textContent = lb.me
    ? `Your rank: #${lb.me.rank} · ${lb.me.balance.toLocaleString()} credits`
    : "";
  fillBoard($("board"), lb.richest, me,
            r => r.balance.toLocaleString(), "No credits earned yet. Go chat!");
  fillBoard($("board-portfolios"), lb.portfolios || [], me,
            r => r.value.toLocaleString(), "Nobody holds any stocks yet.");
  fillBoard($("board-traders"), lb.traders, me,
            r => "+" + r.profit.toLocaleString(), "No trading profits yet.");
  fillBoard($("board-gamblers"), lb.gamblers, me,
            r => "+" + r.profit.toLocaleString(), "No gambling profits yet.");
  fillBoard($("board-criminals"), lb.criminals, me,
            r => "+" + r.stolen.toLocaleString(), "No successful robberies yet.");
}

loadBoard();
setInterval(loadBoard, 30000);
