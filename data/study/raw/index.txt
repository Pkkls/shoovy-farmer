async function load() {
  const me = await (await fetch("/api/me")).json();
  if (!me.logged_in) {
    document.getElementById("login-box").style.display = "flex";
  } else {
    const bal = document.getElementById("m-bal");
    document.getElementById("m-bal-v").textContent = me.balance.toLocaleString();
    document.getElementById("m-bal-s").textContent =
      me.daily_ready ? "daily bonus ready" : "credits on hand";
    bal.className = "metric" + (me.daily_ready ? " ok" : "");
  }

  const lb = await (await fetch("/api/leaderboard?limit=10")).json();
  const note = document.getElementById("stream-note");
  if (lb.channel) {
    const player = document.getElementById("player");
    player.src = "https://player.kick.com/" + lb.channel + "?muted=true&autoplay=true";
    player.style.display = "block";
    note.style.display = "none";
    document.getElementById("stream-ch").textContent = "kick.com/" + lb.channel;
    document.getElementById("m-stream-v").textContent = lb.channel;
    const footer = document.getElementById("channel-name");
    footer.innerHTML = "";
    const a = document.createElement("a");
    a.href = "https://kick.com/" + lb.channel;
    a.target = "_blank";
    a.rel = "noopener";
    a.textContent = "kick.com/" + lb.channel;
    footer.appendChild(a);
  } else {
    note.textContent = "No channel connected yet — the streamer needs to visit /auth/connect.";
  }

  if (me.live !== undefined) {
    const s = document.getElementById("m-stream-s");
    s.textContent = me.live ? "● live now" : "offline";
    s.style.color = me.live ? "var(--red-text)" : "";
    document.getElementById("m-stream").className = "metric" + (me.live ? " crit" : "");
  }

  renderBoard(lb, me);

  if (lb.channel) {
    initShocker(me);
    initTTS(me);
    loadFeed();
    setInterval(loadFeed, 30000);
  }
}

/* Top holders: the same board as /stats, trimmed to the ten that fit here.
   Rows are built with textContent so usernames can't inject markup. */
function renderBoard(lb, me) {
  const rows = lb.leaderboard || [];
  if (!rows.length) return;
  document.getElementById("board-card").style.display = "block";
  const body = document.querySelector("#board tbody");
  body.innerHTML = "";
  rows.forEach((u, i) => {
    const tr = document.createElement("tr");
    if (me.logged_in && u.username === me.username) tr.className = "me";
    const cell = (cls, text) => {
      const td = document.createElement("td");
      td.className = cls;
      td.textContent = text;
      tr.appendChild(td);
    };
    cell("rk", String(i + 1).padStart(2, "0"));
    cell("who", u.username);
    cell("bal num", u.balance.toLocaleString());
    body.appendChild(tr);
  });
  document.getElementById("m-top-v").textContent = rows[0].username;
  document.getElementById("m-top-s").textContent = rows[0].balance.toLocaleString() + " credits";
  if (lb.me) {
    document.getElementById("m-rank-v").textContent = "#" + lb.me.rank;
    document.getElementById("m-rank-s").textContent = "of the tracked ledger";
  }
}

/* Live activity feed: recent notable events, refreshed every 30s. Rows are
   built with textContent (never innerHTML) so usernames and shop item names
   can't inject markup. */
function timeAgo(ts) {
  const s = Math.max(0, Math.floor(Date.now() / 1000 - ts));
  if (s < 60) return "just now";
  if (s < 3600) return Math.floor(s / 60) + "m ago";
  if (s < 86400) return Math.floor(s / 3600) + "h ago";
  return Math.floor(s / 86400) + "d ago";
}

async function loadFeed() {
  let data;
  try {
    data = await (await fetch("/api/feed?limit=14")).json();
  } catch (e) { return; }
  document.getElementById("feed-card").style.display = "flex";
  const list = document.getElementById("feed-list");
  list.innerHTML = "";
  document.getElementById("feed-empty").style.display =
    data.events.length ? "none" : "block";
  for (const ev of data.events) {
    const li = document.createElement("li");
    const msg = document.createElement("span");
    msg.className = "msg";
    const add = (parent, cls, text) => {
      const span = document.createElement("span");
      span.className = cls;
      span.textContent = text;
      parent.appendChild(span);
    };
    add(li, "ico", ev.icon);
    // who + what live in one span so the text wraps under the username
    add(msg, "who", ev.user);
    add(msg, "what", ev.text);
    li.appendChild(msg);
    add(li, "when", timeAgo(ev.ts));
    list.appendChild(li);
  }
}

/* Shocker: spend credits to buzz/shock the streamer's PiShock device.
   Every button is two-step — the first click arms it, the second fires —
   because a mis-click here costs real credits and hits a real person.
   Intensity and duration come from the server, which reads them live from
   the streamer's PiShock share code; nothing about the limits is decided
   here. */
let shockMe = null, shockArmed = null, shockArmedBtn = null, shockArmedLabel = "", shockArmTimer = null;

function shockDisarm() {
  if (shockArmTimer) { clearTimeout(shockArmTimer); shockArmTimer = null; }
  if (shockArmedBtn) {
    shockArmedBtn.classList.remove("armed");
    const name = shockArmedBtn.querySelector(".name");
    if (name) name.textContent = shockArmedLabel;
    shockArmedBtn = null;
  }
  shockArmed = null;
}

function shockWait(seconds) {
  if (seconds < 60) return seconds + "s";
  return Math.floor(seconds / 60) + "m " + (seconds % 60) + "s";
}

function initShocker(me) {
  shockMe = me;
  loadShocker();
  // the queue, the cooldown and the device's own limits all move on their
  // own — but never re-render mid-confirmation
  setInterval(() => { if (!shockArmed) loadShocker(); }, 20000);
}

async function loadShocker() {
  let info;
  try {
    info = await (await fetch("/api/shocker/info")).json();
  } catch (e) { return; }
  const card = document.getElementById("shock-card");
  const status = document.getElementById("shock-status");
  const box = document.getElementById("shock-tiers");
  card.style.display = "block";
  box.innerHTML = "";
  shockDisarm();

  if (!info.enabled) {
    status.textContent = info.reason || "The shocker is unavailable right now.";
    return;
  }

  const bits = ["Limits come from the streamer's own PiShock account — max intensity "
    + info.max_intensity + ", max " + info.max_duration + "s"];
  if (info.cooldown) bits.push("one per viewer every " + shockWait(info.cooldown));
  if (info.remaining_this_hour !== undefined) bits.push(info.remaining_this_hour + " left this hour");
  if (info.queued) bits.push(info.queued + " queued");
  status.textContent = bits.join(" · ");
  if (!shockMe || !shockMe.logged_in) {
    status.appendChild(document.createTextNode(" · "));
    const a = document.createElement("a");
    a.href = "/auth/login";
    a.style.color = "var(--accent)";
    a.textContent = "log in with Kick";
    status.appendChild(a);
    status.appendChild(document.createTextNode(" to use it"));
  }

  const onCooldown = info.cooldown_left > 0;
  for (const t of info.tiers) {
    const btn = document.createElement("button");
    btn.className = "btn tier";
    const name = document.createElement("span");
    name.className = "name";
    name.textContent = t.emoji + " " + t.label;
    const cost = document.createElement("span");
    cost.className = "cost";
    cost.textContent = t.cost.toLocaleString() + " credits";
    const spec = document.createElement("span");
    spec.className = "spec";
    spec.textContent = t.op === "beep"
      ? "beep · " + t.duration + "s"
      : (t.op === "vibrate" ? "vibrate " : "shock ") + t.intensity + " · " + t.duration + "s";
    btn.append(name, cost, spec);
    btn.disabled = !shockMe || !shockMe.logged_in || onCooldown;
    btn.onclick = () => {
      if (shockArmed !== t.key) {
        shockDisarm();
        shockArmed = t.key;
        shockArmedBtn = btn;
        shockArmedLabel = name.textContent;
        btn.classList.add("armed");
        name.textContent = "Confirm?";
        shockArmTimer = setTimeout(shockDisarm, 4000);
        return;
      }
      shockDisarm();
      fireShock(t, btn);
    };
    box.appendChild(btn);
  }

  const hint = document.getElementById("shock-hint");
  if (onCooldown) {
    hint.className = "hint";
    hint.textContent = "You can go again in " + shockWait(info.cooldown_left) + ".";
  }
}

async function fireShock(tier, btn) {
  const hint = document.getElementById("shock-hint");
  btn.disabled = true;
  let ok = false, message = "Something went wrong — try again.";
  try {
    const r = await fetch("/api/shocker/fire", {
      method: "POST", headers: {"Content-Type": "application/json"},
      body: JSON.stringify({ tier: tier.key })
    });
    const data = await r.json();
    ok = r.ok;
    message = (r.ok ? data.message : data.error) || message;
    const balance = document.getElementById("balance");
    if (r.ok && balance) balance.textContent = data.balance.toLocaleString() + " credits";
    if (r.ok) syncBalance(data.balance);
  } catch (e) { /* keep the fallback message */ }
  // re-render first — it picks up the new cooldown and queue depth, and it
  // writes its own hint, which would otherwise swallow this one
  await loadShocker();
  hint.className = "hint " + (ok ? "ok" : "err");
  hint.textContent = message;
}

/* keep the readout strip in step with the command bar after a spend */
function syncBalance(balance) {
  const v = document.getElementById("m-bal-v");
  if (v) v.textContent = balance.toLocaleString();
}

/* TTS send box */
async function initTTS(me) {
  const card = document.getElementById("tts-card");
  const infoEl = document.getElementById("tts-info");
  card.style.display = "block";
  const info = await (await fetch("/api/tts/info")).json();
  if (!info.enabled) {
    infoEl.textContent = "TTS is currently disabled.";
    return;
  }
  infoEl.innerHTML = "Costs " + info.cost.toLocaleString() + " credits · max "
    + info.max_chars + " characters · plays out loud on stream"
    + (info.paused ? " · queue is paused right now — your message will play once it resumes" : "")
    + (me.logged_in ? "" : ' · <a href="/auth/login" style="color:var(--accent);">log in with Kick</a> to send one');
  if (!me.logged_in) return;

  const input = document.getElementById("tts-input");
  const count = document.getElementById("tts-count");
  input.maxLength = info.max_chars;
  const updateCount = () => { count.textContent = input.value.length + " / " + info.max_chars; };
  input.addEventListener("input", updateCount);
  updateCount();
  document.getElementById("tts-box").style.display = "block";

  document.getElementById("tts-send").onclick = async () => {
    const hint = document.getElementById("tts-hint");
    const btn = document.getElementById("tts-send");
    btn.disabled = true;
    try {
      const r = await fetch("/api/tts", {
        method: "POST", headers: {"Content-Type": "application/json"},
        body: JSON.stringify({ message: input.value })
      });
      const data = await r.json();
      hint.className = "hint " + (r.ok ? "ok" : "err");
      if (r.ok) {
        hint.textContent = data.message;
        input.value = "";
        updateCount();
        document.getElementById("balance").textContent = data.balance.toLocaleString() + " credits";
        syncBalance(data.balance);
      } else {
        hint.textContent = data.error;
      }
    } finally {
      btn.disabled = false;
    }
  };
}
