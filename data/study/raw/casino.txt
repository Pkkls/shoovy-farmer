    <script>
    // Tiles are ordered by play count, busiest first — but the counts come
    // from /api/casino/lobby, which lands a moment after the page paints, and
    // reshuffling the grid under someone who just hit refresh reads as broken.
    // So the last order we saw is kept in localStorage and applied here,
    // inline, while the grid is still being parsed and before it has ever been
    // painted; the fetch below then corrects it and saves the new one. A 24h
    // count barely moves between two page loads, so in practice the cached
    // order is already the right one and nothing visibly shifts. A first-ever
    // visit gets the markup order — the same fallback a failed fetch leaves.
    (() => {
      const CACHE_KEY = "casino:lobby-order";
      const lobby = document.querySelector(".lobby");

      // [{tile, game}, ...] in current DOM order
      const tiles = () => [...lobby.querySelectorAll(".tile")].map(tile => ({
        tile, game: tile.querySelector("[data-plays]")?.dataset.plays || "",
      }));

      // A game the order doesn't mention — a tile added to the markup since
      // the order was cached — sorts to the back rather than vanishing.
      const apply = (order) => {
        const rank = (game) => {
          const i = order.indexOf(game);
          return i === -1 ? order.length : i;
        };
        tiles().sort((a, b) => rank(a.game) - rank(b.game))
               .forEach(({ tile }) => lobby.appendChild(tile));
      };

      try {
        const cached = JSON.parse(localStorage.getItem(CACHE_KEY) || "null");
        if (Array.isArray(cached)) apply(cached);
      } catch (e) {}

      // Live-ish activity under each tile. Counts are bets placed in the last
      // 24h, not people in the room — we have no presence tracking, and a
      // made-up "playing" number would be a lie. Tiles ship with "—" so a
      // failed fetch leaves the lobby usable. Only the counts are cached-free:
      // a stale number would be a wrong number, a stale order is just an order.
      (async () => {
        try {
          const d = await (await fetch("/api/casino/lobby")).json();
          document.body.classList.toggle("casino-off", !d.casino_enabled);
          const plays = d.plays || {};
          for (const [game, n] of Object.entries(plays)) {
            const el = document.querySelector(`[data-plays="${game}"]`);
            if (el) el.textContent = n.toLocaleString();
          }
          // Stable, so games tied on plays (all of them, on a quiet day) keep
          // whatever order they're already in instead of jumping around.
          const order = tiles().map(t => ({ ...t, n: plays[t.game] || 0 }))
                               .sort((a, b) => b.n - a.n)
                               .map(t => t.game);
          apply(order);
          localStorage.setItem(CACHE_KEY, JSON.stringify(order));
        } catch (e) {}
      })();
    })();
    </script>
